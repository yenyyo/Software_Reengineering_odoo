402 def test_address_get(self): 

403 """ Test address_get address resolution mechanism: it should first go down through descendants, 

404 stopping when encountering another is_copmany entity, then go up, stopping again at the first 

405 is_company entity or the root ancestor and if nothing matches, it should use the provided partner 

406 itself """ 

407 res_partner = self.env['res.partner'] 

408 elmtree = res_partner.browse(res_partner.name_create('Elmtree')[0]) 

409 branch1 = res_partner.create({'name': 'Branch 1', 

410 'parent_id': elmtree.id, 

411 'is_company': True}) 

412 leaf10 = res_partner.create({'name': 'Leaf 10', 

413 'parent_id': branch1.id, 

414 'type': 'invoice'}) 

415 branch11 = res_partner.create({'name': 'Branch 11', 

416 'parent_id': branch1.id, 

417 'type': 'other'}) 

418 leaf111 = res_partner.create({'name': 'Leaf 111', 

419 'parent_id': branch11.id, 

420 'type': 'delivery'}) 

421 branch11.write({'is_company': False}) # force is_company after creating 1rst child 

422 branch2 = res_partner.create({'name': 'Branch 2', 

423 'parent_id': elmtree.id, 

424 'is_company': True}) 

425 leaf21 = res_partner.create({'name': 'Leaf 21', 

426 'parent_id': branch2.id, 

427 'type': 'delivery'}) 

428 leaf22 = res_partner.create({'name': 'Leaf 22', 

429 'parent_id': branch2.id}) 

430 leaf23 = res_partner.create({'name': 'Leaf 23', 

431 'parent_id': branch2.id, 

432 'type': 'contact'}) 

433 

434 # go up, stop at branch1 

435 self.assertEqual(leaf111.address_get(['delivery', 'invoice', 'contact', 'other']), 

436 {'delivery': leaf111.id, 

437 'invoice': leaf10.id, 

438 'contact': branch1.id, 

439 'other': branch11.id}, 'Invalid address resolution') 

440 self.assertEqual(branch11.address_get(['delivery', 'invoice', 'contact', 'other']), 

441 {'delivery': leaf111.id, 

442 'invoice': leaf10.id, 

443 'contact': branch1.id, 

444 'other': branch11.id}, 'Invalid address resolution') 

445 

446 # go down, stop at at all child companies 

447 self.assertEqual(elmtree.address_get(['delivery', 'invoice', 'contact', 'other']), 

448 {'delivery': elmtree.id, 

449 'invoice': elmtree.id, 

450 'contact': elmtree.id, 

451 'other': elmtree.id}, 'Invalid address resolution') 

452 

453 # go down through children 

454 self.assertEqual(branch1.address_get(['delivery', 'invoice', 'contact', 'other']), 

455 {'delivery': leaf111.id, 

456 'invoice': leaf10.id, 

457 'contact': branch1.id, 

458 'other': branch11.id}, 'Invalid address resolution') 

459 

460 self.assertEqual(branch2.address_get(['delivery', 'invoice', 'contact', 'other']), 

461 {'delivery': leaf21.id, 

462 'invoice': branch2.id, 

463 'contact': branch2.id, 

464 'other': branch2.id}, 'Invalid address resolution. Company is the first encountered contact, therefore default for unfound addresses.') 

465 

466 # go up then down through siblings 

467 self.assertEqual(leaf21.address_get(['delivery', 'invoice', 'contact', 'other']), 

468 {'delivery': leaf21.id, 

469 'invoice': branch2.id, 

470 'contact': branch2.id, 

471 'other': branch2.id}, 'Invalid address resolution, should scan commercial entity ancestor and its descendants') 

472 self.assertEqual(leaf22.address_get(['delivery', 'invoice', 'contact', 'other']), 

473 {'delivery': leaf21.id, 

474 'invoice': leaf22.id, 

475 'contact': leaf22.id, 

476 'other': leaf22.id}, 'Invalid address resolution, should scan commercial entity ancestor and its descendants') 

477 self.assertEqual(leaf23.address_get(['delivery', 'invoice', 'contact', 'other']), 

478 {'delivery': leaf21.id, 

479 'invoice': leaf23.id, 

480 'contact': leaf23.id, 

481 'other': leaf23.id}, 'Invalid address resolution, `default` should only override if no partner with specific type exists')
