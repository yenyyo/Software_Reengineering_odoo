124 attributes = el.attrib 

125 styling = attributes.get('style') 

126 if styling:

128 styles = self._style_re.findall(styling)

131 valid_styles[style[0].lower()] = style[1]

156def tag_quote(el): 

157 def _create_new_node(tag, text, tail=None, attrs=None): 

158 new_node = etree.Element(tag) 

159 new_node.text = text 

160 new_node.tail = tail 

161 if attrs: 

162 for key, val in attrs.items(): 

163 new_node.set(key, val) 

164 return new_node

166 def _tag_matching_regex_in_text(regex, node, tag='span', attrs=None): 

167 text = node.text or '' 

168 if not re.search(regex, text): 

169 return 

170 

171 child_node = None 

172 idx, node_idx = 0, 0 

173 for item in re.finditer(regex, text): 

174 new_node = _create_new_node(tag, text[item.start():item.end()], None, attrs) 

175 if child_node is None: 

176 node.text = text[idx:item.start()] 

177 new_node.tail = text[item.end():] 

178 node.insert(node_idx, new_node) 

179 else: 

180 child_node.tail = text[idx:item.start()] 

181 new_node.tail = text[item.end():] 

182 node.insert(node_idx, new_node) 

183 child_node = new_node 

184 idx = item.end() 

185 node_idx = node_idx + 1 

186 

187 el_class = el.get('class', '') or '' 

188 el_id = el.get('id', '') or '' 

189 

190 # gmail or yahoo // # outlook, html // # msoffice 

191 if 'gmail_extra' in el_class or \ 

192 ('SkyDrivePlaceholder' in el_class or 'SkyDrivePlaceholder' in el_class): 

193 el.set('data-o-mail-quote', '1') 

194 if el.getparent() is not None: 

195 el.getparent().set('data-o-mail-quote-container', '1') 

196 

197 if (el.tag == 'hr' and ('stopSpelling' in el_class or 'stopSpelling' in el_id)) or \ 

198 'yahoo_quoted' in el_class: 

199 # Quote all elements after this one 

200 el.set('data-o-mail-quote', '1') 

201 for sibling in el.itersiblings(preceding=False): 

202 sibling.set('data-o-mail-quote', '1') 

203 

204 # odoo, gmail and outlook automatic signature wrapper 

205 is_signature_wrapper = 'odoo_signature_wrapper' in el_class or 'gmail_signature' in el_class or el_id == "Signature" 

206 is_outlook_auto_message = 'appendonsend' in el_id 

207 # gmail and outlook reply quote 

208 is_outlook_reply_quote = 'divRplyFwdMsg' in el_id 

209 is_gmail_quote = 'gmail_quote' in el_class 

210 is_quote_wrapper = is_signature_wrapper or is_gmail_quote or is_outlook_reply_quote 

211 if is_quote_wrapper: 

212 el.set('data-o-mail-quote-container', '1') 

213 el.set('data-o-mail-quote', '1') 

214 

215 # outlook reply wrapper is preceded with <hr> and a div containing recipient info 

216 if is_outlook_reply_quote: 

217 hr = el.getprevious() 

218 reply_quote = el.getnext() 

219 if hr is not None and hr.tag == 'hr': 

220 hr.set('data-o-mail-quote', '1') 

221 if reply_quote is not None: 

222 reply_quote.set('data-o-mail-quote-container', '1') 

223 reply_quote.set('data-o-mail-quote', '1') 

224 

225 if is_outlook_auto_message: 

226 if not el.text or not el.text.strip(): 

227 el.set('data-o-mail-quote-container', '1') 

228 el.set('data-o-mail-quote', '1') 

229 

230 # html signature (-- <br />blah) 

231 signature_begin = re.compile(r"((?:(?:^|\n)[-]{2}[\s]?$))") 

232 if el.text and el.find('br') is not None and re.search(signature_begin, el.text): 

233 el.set('data-o-mail-quote', '1') 

234 if el.getparent() is not None: 

235 el.getparent().set('data-o-mail-quote-container', '1') 

236 

237 # text-based quotes (>, >>) and signatures (-- Signature) 

238 text_complete_regex = re.compile(r"((?:\n[>]+[^\n\r]*)+|(?:(?:^|\n)[-]{2}[\s]?[\r\n]{1,2}[\s\S]+))") 

239 if not el.get('data-o-mail-quote'): 

240 _tag_matching_regex_in_text(text_complete_regex, el, 'span', {'data-o-mail-quote': '1'}) 

241 

242 if el.tag == 'blockquote': 

243 # remove single node 

244 el.set('data-o-mail-quote-node', '1') 

245 el.set('data-o-mail-quote', '1') 

246 if el.getparent() is not None and not el.getparent().get('data-o-mail-quote-node'): 

247 if el.getparent().get('data-o-mail-quote'): 

248 el.set('data-o-mail-quote', '1') 

249 # only quoting the elements following the first quote in the container 

250 # avoids issues with repeated calls to html_normalize 

251 elif el.getparent().get('data-o-mail-quote-container'): 

252 if (first_sibling_quote := el.getparent().find("*[@data-o-mail-quote]")) is not None: 

253 siblings = el.getparent().getchildren() 

254 quote_index = siblings.index(first_sibling_quote) 

255 element_index = siblings.index(el) 

256 if quote_index < element_index: 

257 el.set('data-o-mail-quote', '1') 

258 if el.getprevious() is not None and el.getprevious().get('data-o-mail-quote') and not el.text_content().strip(): 

259 el.set('data-o-mail-quote', '1')

262def html_normalize(src, filter_callback=None, output_method="html"): 

263 """ Normalize `src` for storage as an html field value. 

264 

265 The string is parsed as an html tag soup, made valid, then decorated for 

266 "email quote" detection, and prepared for an optional filtering. 

267 The filtering step (e.g. sanitization) should be performed by the 

268 `filter_callback` function (to avoid multiple parsing operations, and 

269 normalize the result). 

270 

271 :param src: the html string to normalize 

272 :param filter_callback: optional callable taking a single `etree._Element` 

273 document parameter, to be called during normalization in order to 

274 filter the output document 

275 :param output_method: defines the output method to pass to `html.tostring`. 

276 It defaults to 'html', but can also be 'xml' for xhtml output. 

277 """ 

278 if not src: 

279 return src 

280 

281 # html: remove encoding attribute inside tags 

282 src = re.sub(r'(<[^>]*\s)(encoding=(["\'][^"\']*?["\']|[^\s\n\r>]+)(\s[^>]*|/)?>)', "", src, flags=re.IGNORECASE | re.DOTALL) 

283 

284 src = src.replace('--!>', '-->') 

285 src = re.sub(r'(<!-->|<!--->)', '<!-- -->', src) 

286 # On the specific case of Outlook desktop it adds unnecessary '<o:.*></o:.*>' tags which are parsed 

287 # in '<p></p>' which may alter the appearance (eg. spacing) of the mail body 

288 src = re.sub(r'</?o:.*?>', '', src) 

289 

290 try: 

291 doc = html.fromstring(src) 

292 except etree.ParserError as e: 

293 # HTML comment only string, whitespace only.. 

294 if 'empty' in str(e): 

295 return "" 

296 raise 

297 

298 # perform quote detection before cleaning and class removal 

299 if doc is not None: 

300 for el in doc.iter(tag=etree.Element): 

301 tag_quote(el) 

302 

303 if filter_callback: 

304 doc = filter_callback(doc) 

305 

306 src = html.tostring(doc, encoding='unicode', method=output_method) 

307 

308 # this is ugly, but lxml/etree tostring want to put everything in a 

309 # 'div' that breaks the editor -> remove that 

310 if src.startswith('<div>') and src.endswith('</div>'): 

311 src = src[5:-6] 

312 

313 # html considerations so real html content match database value 

314 src = src.replace(u'\xa0', u'&nbsp;') 

315 

316 return src

319def html_sanitize(src, silent=True, sanitize_tags=True, sanitize_attributes=False, sanitize_style=False, sanitize_form=True, sanitize_conditional_comments=True, strip_style=False, strip_classes=False, output_method="html"): 

320 if not src: 

321 return src 

322 

323 logger = logging.getLogger(__name__ + '.html_sanitize') 

324 

325 def sanitize_handler(doc): 

326 kwargs = { 

327 'page_structure': True, 

328 'style': strip_style, # True = remove style tags/attrs 

329 'sanitize_style': sanitize_style, # True = sanitize styling 

330 'forms': sanitize_form, # True = remove form tags 

331 'remove_unknown_tags': False, 

332 'comments': False, 

333 'conditional_comments': sanitize_conditional_comments, # True = remove conditional comments 

334 'processing_instructions': False 

335 } 

336 if sanitize_tags: 

337 kwargs.update(SANITIZE_TAGS) 

338 

339 if sanitize_attributes: # We keep all attributes in order to keep "style" 

340 if strip_classes: 

341 current_safe_attrs = safe_attrs - frozenset(['class']) 

342 else: 

343 current_safe_attrs = safe_attrs 

344 kwargs.update({ 

345 'safe_attrs_only': True, 

346 'safe_attrs': current_safe_attrs, 

347 }) 

348 else: 

349 kwargs.update({ 

350 'safe_attrs_only': False, # keep oe-data attributes + style 

351 'strip_classes': strip_classes, # remove classes, even when keeping other attributes 

352 }) 

353 

354 cleaner = _Cleaner(**kwargs) 

355 cleaner(doc) 

356 return doc 

357 

358 try: 

359 sanitized = html_normalize(src, filter_callback=sanitize_handler, output_method=output_method) 

360 except etree.ParserError: 

361 if not silent: 

362 raise 

363 logger.warning(u'ParserError obtained when sanitizing %r', src, exc_info=True) 

364 sanitized = '<p>ParserError when sanitizing</p>' 

365 except Exception: 

366 if not silent: 

367 raise 

368 logger.warning(u'unknown error obtained when sanitizing %r', src, exc_info=True) 

369 sanitized = '<p>Unknown error when sanitizing</p>' 

370 

371 return markupsafe.Markup(sanitized)

393def is_html_empty(html_content): 

394 """Check if a html content is empty. If there are only formatting tags with style 

395 attributes or a void content return True. Famous use case if a 

396 '<p style="..."><br></p>' added by some web editor. 

397 

398 :param str html_content: html content, coming from example from an HTML field 

399 :returns: bool, True if no content found or if containing only void formatting tags 

400 """ 

401 if not html_content: 

402 return True 

403 icon_re = r'<\s*(i|span)\b(\s+[A-Za-z_-][A-Za-z0-9-_]*(\s*=\s*[\'"][^"\']*[\'"])?)*\s*\bclass\s*=\s*["\'][^"\']*\b(fa|fab|fad|far|oi)\b' 

404 tag_re = r'<\s*\/?(?:p|div|section|span|br|b|i|font)\b(?:(\s+[A-Za-z_-][A-Za-z0-9-_]*(\s*=\s*[\'"][^"\']*[\'"]))*)(?:\s*>|\s*\/\s*>)' 

405 return not bool(re.sub(tag_re, '', html_content).strip()) and not re.search(icon_re, html_content)

408def html_keep_url(text): 

409 """ Transform the url into clickable link with <a/> tag """ 

410 idx = 0 

411 final = '' 

412 link_tags = re.compile(r"""(?<!["'])((ftp|http|https):\/\/(\w+:{0,1}\w*@)?([^\s<"']+)(:[0-9]+)?(\/|\/([^\s<"']))?)(?![^\s<"']*["']|[^\s<"']*</a>)""") 

413 for item in re.finditer(link_tags, text): 

414 final += text[idx:item.start()] 

415 final += create_link(item.group(0), item.group(0)) 

416 idx = item.end() 

417 final += text[idx:] 

418 return final

421def html_to_inner_content(html): 

422 """Returns unformatted text after removing html tags and excessive whitespace from a 

423 string/Markup. Passed strings will first be sanitized. 

424 """ 

425 if is_html_empty(html): 

426 return '' 

427 if not isinstance(html, markupsafe.Markup): 

428 html = html_sanitize(html) 

429 processed = re.sub(HTML_NEWLINES_REGEX, ' ', html) 

430 processed = re.sub(HTML_TAGS_REGEX, '', processed) 

431 processed = re.sub(r' {2,}|\t', ' ', processed) 

432 processed = htmllib.unescape(processed) 

433 processed = processed.strip() 

434 return processed

441def html2plaintext(html, body_id=None, encoding='utf-8', include_references=True): 

442 """ From an HTML text, convert the HTML to plain text. 

443 If @param body_id is provided then this is the tag where the 

444 body (not necessarily <body>) starts. 

445 :param include_references: If False, numbered references and 

446 URLs for links and images will not be included. 

447 """ 

448 ## (c) Fry-IT, www.fry-it.com, 2007 

449 ## <peter@fry-it.com> 

450 ## download here: http://www.peterbe.com/plog/html2plaintext 

451 if not (html and html.strip()): 

452 return '' 

453 

454 if isinstance(html, bytes): 

455 html = html.decode(encoding) 

456 else: 

457 assert isinstance(html, str), f"expected str got {html.__class__.__name__}" 

458 

459 tree = etree.fromstring(html, parser=etree.HTMLParser()) 

460 

461 if body_id is not None: 

462 source = tree.xpath('//*[@id=%s]' % (body_id,)) 

463 else: 

464 source = tree.xpath('//body') 

465 if len(source): 

466 tree = source[0] 

467 

468 url_index = [] 

469 linkrefs = itertools.count(1) 

470 if include_references: 

471 for link in tree.findall('.//a'): 

472 if url := link.get('href'): 

473 link.tag = 'span' 

474 link.text = f'{link.text} [{next(linkrefs)}]' 

475 url_index.append(url) 

476 

477 for img in tree.findall('.//img'): 

478 if src := img.get('src'): 

479 img.tag = 'span' 

480 img_name = re.search(r'[^/]+(?=\.[a-zA-Z]+(?:\?|$))', src) 

481 img.text = '%s [%s]' % (img_name[0] if img_name else 'Image', next(linkrefs)) 

482 url_index.append(src) 

483 

484 html = etree.tostring(tree, encoding="unicode") 

485 # \r char is converted into &#13;, must remove it 

486 html = html.replace('&#13;', '') 

487 

488 html = html.replace('<strong>', '*').replace('</strong>', '*') 

489 html = html.replace('<b>', '*').replace('</b>', '*') 

490 html = html.replace('<h3>', '*').replace('</h3>', '*') 

491 html = html.replace('<h2>', '**').replace('</h2>', '**') 

492 html = html.replace('<h1>', '**').replace('</h1>', '**') 

493 html = html.replace('<em>', '/').replace('</em>', '/') 

494 html = html.replace('<tr>', '\n') 

495 html = html.replace('</p>', '\n') 

496 html = re.sub(r'<br\s*/?>', '\n', html) 

497 html = re.sub('<.*?>', ' ', html) 

498 html = html.replace(' ' * 2, ' ') 

499 html = html.replace('&gt;', '>') 

500 html = html.replace('&lt;', '<') 

501 html = html.replace('&amp;', '&') 

502 html = html.replace('&nbsp;', '\N{NO-BREAK SPACE}') 

503 

504 # strip all lines 

505 html = '\n'.join([x.strip() for x in html.splitlines()]) 

506 html = html.replace('\n' * 2, '\n') 

507 

508 if url_index: 

509 html += '\n\n' 

510 for i, url in enumerate(url_index, start=1): 

511 html += f'[{i}] {url}\n' 

512 

513 return html.strip()

515def plaintext2html(text, container_tag=None): 

516 r"""Convert plaintext into html. Content of the text is escaped to manage 

517 html entities, using :func:`~odoo.tools.misc.html_escape`. 

518 

519 - all ``\n``, ``\r`` are replaced by ``<br/>`` 

520 - enclose content into ``<p>`` 

521 - convert url into clickable link 

522 - 2 or more consecutive ``<br/>`` are considered as paragraph breaks 

523 

524 :param str text: plaintext to convert 

525 :param str container_tag: container of the html; by default the content is 

526 embedded into a ``<div>`` 

527 :rtype: markupsafe.Markup 

528 """ 

529 assert isinstance(text, str) 

530 text = misc.html_escape(text) 

531 

532 # 1. replace \n and \r 

533 text = re.sub(r'(\r\n|\r|\n)', '<br/>', text) 

534 

535 # 2. clickable links 

536 text = html_keep_url(text) 

537 

538 # 3-4: form paragraphs 

539 idx = 0 

540 final = '<p>' 

541 br_tags = re.compile(r'(([<]\s*[bB][rR]\s*/?[>]\s*){2,})') 

542 for item in re.finditer(br_tags, text): 

543 final += text[idx:item.start()] + '</p><p>' 

544 idx = item.end() 

545 final += text[idx:] + '</p>' 

546 

547 # 5. container 

548 if container_tag: # FIXME: validate that container_tag is just a simple tag? 

549 final = '<%s>%s</%s>' % (container_tag, final, container_tag) 

550 return markupsafe.Markup(final)

552def append_content_to_html(html, content, plaintext=True, preserve=False, container_tag=None): 

553 """ Append extra content at the end of an HTML snippet, trying 

554 to locate the end of the HTML document (</body>, </html>, or 

555 EOF), and converting the provided content in html unless ``plaintext`` 

556 is ``False``. 

557 

558 Content conversion can be done in two ways: 

559 

560 - wrapping it into a pre (``preserve=True``) 

561 - use plaintext2html (``preserve=False``, using ``container_tag`` to 

562 wrap the whole content) 

563 

564 A side-effect of this method is to coerce all HTML tags to 

565 lowercase in ``html``, and strip enclosing <html> or <body> tags in 

566 content if ``plaintext`` is False. 

567 

568 :param str html: html tagsoup (doesn't have to be XHTML) 

569 :param str content: extra content to append 

570 :param bool plaintext: whether content is plaintext and should 

571 be wrapped in a <pre/> tag. 

572 :param bool preserve: if content is plaintext, wrap it into a <pre> 

573 instead of converting it into html 

574 :param str container_tag: tag to wrap the content into, defaults to `div`. 

575 :rtype: markupsafe.Markup 

576 """ 

577 if plaintext and preserve: 

578 content = '\n<pre>%s</pre>\n' % misc.html_escape(content) 

579 elif plaintext: 

580 content = '\n%s\n' % plaintext2html(content, container_tag) 

581 else: 

582 content = re.sub(r'(?i)(</?(?:html|body|head|!\s*DOCTYPE)[^>]*>)', '', content) 

583 content = '\n%s\n' % content 

584 # Force all tags to lowercase 

585 html = re.sub(r'(</?)(\w+)([ >])', 

586 lambda m: '%s%s%s' % (m[1], m[2].lower(), m[3]), html) 

587 insert_location = html.find('</body>') 

588 if insert_location == -1: 

589 insert_location = html.find('</html>') 

590 if insert_location == -1: 

591 return markupsafe.Markup('%s%s' % (html, content)) 

592 return markupsafe.Markup('%s%s%s' % (html[:insert_location], content, html[insert_location:]))

596 """Prepend some HTML content at the beginning of an other HTML content.""" 

597 replacement = re.sub(r'(?i)(</?(?:html|body|head|!\s*DOCTYPE)[^>]*>)', '', html_content) 

598 html_content = markupsafe.Markup(replacement) if isinstance(html_content, markupsafe.Markup) else replacement 

599 html_content = html_content.strip()

601 body_match = re.search(r'<body[^>]*>', html_body) or re.search(r'<html[^>]*>', html_body) 

602 insert_index = body_match.end() if body_match else 0 

603

633def email_split_tuples(text): 

634 """ Return a list of (name, email) address tuples found in ``text`` . Note 

635 that text should be an email header or a stringified email list as it may 

636 give broader results than expected on actual text. """ 

637 def _parse_based_on_spaces(pair): 

638 """ With input 'name email@domain.com' (missing quotes for a formatting) 

639 getaddresses returns ('', 'name email@domain.com). This when having no 

640 name and an email a fallback to enhance parsing is to redo a getaddresses 

641 by replacing spaces by commas. The new email will be split into sub pairs 

642 allowing to find the email and name parts, allowing to make a new name / 

643 email pair. Emails should not contain spaces thus this is coherent with 

644 email formation. """ 

645 name, email = pair 

646 if not name and email and ' ' in email: 

647 inside_pairs = getaddresses([email.replace(' ', ',')]) 

648 name_parts, found_email = [], False 

649 for pair in inside_pairs: 

650 if pair[1] and '@' not in pair[1]: 

651 name_parts.append(pair[1]) 

652 if pair[1] and '@' in pair[1]: 

653 found_email = pair[1] 

654 name, email = (' '.join(name_parts), found_email) if found_email else (name, email) 

655 return (name, email) 

656 

657 if not text: 

658 return [] 

659 

660 # found valid pairs, filtering out failed parsing 

661 valid_pairs = [ 

662 (addr[0], addr[1]) for addr in getaddresses([text]) 

663 # getaddresses() returns '' when email parsing fails, and 

664 # sometimes returns emails without at least '@'. The '@' 

665 # is strictly required in RFC2822's `addr-spec`. 

666 if addr[1] and '@' in addr[1] 

667 ] 

668 # corner case: returning '@gmail.com'-like email (see test_email_split) 

669 if any(pair[1].startswith('@') for pair in valid_pairs): 

670 filtered = [ 

671 found_email for found_email in email_re.findall(text) 

672 if found_email and not found_email.startswith('@') 

673 ] 

674 if filtered: 

675 valid_pairs = [('', found_email) for found_email in filtered] 

676 

677 return list(map(_parse_based_on_spaces, valid_pairs))

700def email_normalize(text, strict=True): 

701 """ Sanitize and standardize email address entries. As of rfc5322 section 

702 3.4.1 local-part is case-sensitive. However most main providers do consider 

703 the local-part as case insensitive. With the introduction of smtp-utf8 

704 within odoo, this assumption is certain to fall short for international 

705 emails. We now consider that 

706 

707 * if local part is ascii: normalize still 'lower' ; 

708 * else: use as it, SMTP-UF8 is made for non-ascii local parts; 

709 

710 Concerning domain part of the address, as of v14 international domain (IDNA) 

711 are handled fine. The domain is always lowercase, lowering it is fine as it 

712 is probably an error. With the introduction of IDNA, there is an encoding 

713 that allow non-ascii characters to be encoded to ascii ones, using 'idna.encode'. 

714 

715 A normalized email is considered as : 

716 - having a left part + @ + a right part (the domain can be without '.something') 

717 - having no name before the address. Typically, having no 'Name <>' 

718 Ex: 

719 - Possible Input Email : 'Name <NaMe@DoMaIn.CoM>' 

720 - Normalized Output Email : 'name@domain.com' 

721 

722 :param boolean strict: if True, text should contain a single email 

723 (default behavior in stable 14+). If more than one email is found no 

724 normalized email is returned. If False the first found candidate is used 

725 e.g. if email is 'tony@e.com, "Tony2" <tony2@e.com>', result is either 

726 False (strict=True), either 'tony@e.com' (strict=False). 

727 

728 :return: False if no email found (or if more than 1 email found when being 

729 in strict mode); normalized email otherwise; 

730 """ 

731 emails = email_split(text) 

732 if not emails or (strict and len(emails) != 1): 

733 return False 

734 

735 return _normalize_email(emails[0])

751def _normalize_email(email): 

752 """ As of rfc5322 section 3.4.1 local-part is case-sensitive. However most 

753 main providers do consider the local-part as case insensitive. With the 

754 introduction of smtp-utf8 within odoo, this assumption is certain to fall 

755 short for international emails. We now consider that 

756 

757 * if local part is ascii: normalize still 'lower' ; 

758 * else: use as it, SMTP-UF8 is made for non-ascii local parts; 

759 

760 Concerning domain part of the address, as of v14 international domain (IDNA) 

761 are handled fine. The domain is always lowercase, lowering it is fine as it 

762 is probably an error. With the introduction of IDNA, there is an encoding 

763 that allow non-ascii characters to be encoded to ascii ones, using 'idna.encode'. 

764 

765 A normalized email is considered as : 

766 - having a left part + @ + a right part (the domain can be without '.something') 

767 - having no name before the address. Typically, having no 'Name <>' 

768 Ex: 

769 - Possible Input Email : 'Name <NaMe@DoMaIn.CoM>' 

770 - Normalized Output Email : 'name@domain.com' 

771 """ 

772 local_part, at, domain = email.rpartition('@') 

773 try: 

774 local_part.encode('ascii') 

775 except UnicodeEncodeError: 

776 pass 

777 else: 

778 local_part = local_part.lower() 

779 return local_part + at + domain.lower()

781def email_anonymize(normalized_email, *, redact_domain=False): 

782 """ 

783 Replace most charaters in the local part of the email address with 

784 '*' to hide the recipient, but keep enough characters for debugging 

785 purpose. 

786 

787 The email address must be normalized already. 

788 

789 >>> email_anonymize('admin@example.com') 

790 'a****@example.com' 

791 >>> email_anonymize('portal@example.com') 

792 'p***al@example.com' 

793 >>> email_anonymize('portal@example.com', redact_domain=True) 

794 'p***al@e******.com' 

795 """ 

796 if not normalized_email: 

797 return normalized_email 

798 

799 local, at, domain = normalized_email.partition('@') 

800 if len(local) <= 5: 

801 anon_local = local[:1] + '*' * (len(local) - 1) 

802 else: 

803 anon_local = local[:1] + '*' * (len(local) - 3) + local[-2:] 

804 

805 host, dot, tld = domain.rpartition('.') 

806 if redact_domain and not domain.startswith('[') and all((host, dot, tld)): 

807 anon_host = host[0] + '*' * (len(host) - 1) 

808 else: 

809 anon_host = host 

810 

811 return f'{anon_local}{at}{anon_host}{dot}{tld}'

851def formataddr(pair, charset='utf-8'): 

852 """Pretty format a 2-tuple of the form (realname, email_address). 

853 

854 If the first element of pair is falsy then only the email address 

855 is returned. 

856 

857 Set the charset to ascii to get a RFC-2822 compliant email. The 

858 realname will be base64 encoded (if necessary) and the domain part 

859 of the email will be punycode encoded (if necessary). The local part 

860 is left unchanged thus require the SMTPUTF8 extension when there are 

861 non-ascii characters. 

862 

863 >>> formataddr(('John Doe', 'johndoe@example.com')) 

864 '"John Doe" <johndoe@example.com>' 

865 

866 >>> formataddr(('', 'johndoe@example.com')) 

867 'johndoe@example.com' 

868 """ 

869 name, address = pair 

870 local, _, domain = address.rpartition('@') 

871 

872 try: 

873 domain.encode(charset) 

874 except UnicodeEncodeError: 

875 # rfc5890 - Internationalized Domain Names for Applications (IDNA) 

876 domain = idna.encode(domain).decode('ascii') 

877 

878 if name: 

879 try: 

880 name.encode(charset) 

881 except UnicodeEncodeError: 

882 # charset mismatch, encode as utf-8/base64 

883 # rfc2047 - MIME Message Header Extensions for Non-ASCII Text 

884 name = base64.b64encode(name.encode('utf-8')).decode('ascii') 

885 return f"=?utf-8?b?{name}?= <{local}@{domain}>" 

886 else: 

887 # ascii name, escape it if needed 

888 # rfc2822 - Internet Message Format 

889 # #section-3.4 - Address Specification 

890 name = email_addr_escapes_re.sub(r'\\\g<0>', name) 

891 return f'"{name}" <{local}@{domain}>' 

892 return f"{local}@{domain}"

899 * New From: notifications@odoo.com 

900 * Output: "Admin" <notifications@odoo.com>

904 return old_email 

905

908 return

910 old_name, old_email = old_email_split[0] 

911 if old_name: 

912 name_part = old_name 

913 else: 

914 name_part = old_email.split("@")[0] 

915 

916 return formataddr(( 

917 name_part, 

918 new_email_split[0][1],
