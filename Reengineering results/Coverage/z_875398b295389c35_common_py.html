122 def _prepare_provider(cls, code='none', company=None, update_values=None): 

123 """ Prepare and return the first provider matching the given provider and company. 

124 

125 If no provider is found in the given company, we duplicate the one from the base company. 

126 

127 All other providers belonging to the same company are disabled to avoid any interferences. 

128 

129 :param str code: The code of the provider to prepare 

130 :param recordset company: The company of the provider to prepare, as a `res.company` record 

131 :param dict update_values: The values used to update the provider 

132 :return: The provider to prepare, if found 

133 :rtype: recordset of `payment.provider` 

134 """ 

135 company = company or cls.env.company 

136 update_values = update_values or {} 

137 provider_domain = cls._get_provider_domain(code) 

138 

139 provider = cls.env['payment.provider'].sudo().search( 

140 AND([provider_domain, [('company_id', '=', company.id)]]), limit=1 

141 ) 

142 if not provider: 

143 if code != 'none': 

144 base_provider = cls.env['payment.provider'].sudo().search(provider_domain, limit=1) 

145 else: 

146 base_provider = cls.provider 

147 if not base_provider: 

148 _logger.error("no payment.provider found for code %s", code) 

149 return cls.env['payment.provider'] 

150 else: 

151 provider = base_provider.copy({'company_id': company.id}) 

152 

153 update_values['state'] = 'test' 

154 provider.write(update_values) 

155 return provider

197 def _generate_test_access_token(self, *values): 

198 """ Generate an access token based on the provided values for testing purposes. 

199 

200 This methods returns a token identical to that generated by 

201 payment.utils.generate_access_token but uses the test class environment rather than the 

202 environment of odoo.http.request. 

203 

204 See payment.utils.generate_access_token for additional details. 

205 

206 :param list values: The values to use for the generation of the token 

207 :return: The generated access token 

208 :rtype: str 

209 """ 

210 token_str = '|'.join(str(val) for val in values) 

211 access_token = hmac_tool(self.env(su=True), 'generate_access_token', token_str) 

212 return access_token

214 def _extract_values_from_html_form(self, html_form): 

215 """ Extract the transaction rendering values from an HTML form. 

216 

217 :param str html_form: The HTML form 

218 :return: The extracted information (action & inputs) 

219 :rtype: dict[str:str] 

220 """ 

221 html_tree = objectify.fromstring(html_form) 

222 if hasattr(html_tree, 'input'): 

223 inputs = {input_.get('name'): input_.get('value') for input_ in html_tree.input} 

224 else: 

225 inputs = {} 

226 return { 

227 'action': html_tree.get('action'), 

228 'method': html_tree.get('method'), 

229 'inputs': inputs, 

230 }

232 def _assert_does_not_raise(self, exception_class, func, *args, **kwargs): 

233 """ Fail if an exception of the provided class is raised when calling the function. 

234 

235 If an exception of any other class is raised, it is caught and silently ignored. 

236 

237 This method cannot be used with functions that make requests. Any exception raised in the 

238 scope of the new request will not be caught and will make the test fail. 

239 

240 :param class exception_class: The class of the exception to monitor. 

241 :param function fun: The function to call when monitoring for exceptions. 

242 :param list args: The positional arguments passed as-is to the called function. 

243 :param dict kwargs: The keyword arguments passed as-is to the called function. 

244 :return: None 

245 """ 

246 try: 

247 func(*args, **kwargs) 

248 except exception_class: 

249 self.fail(f"{func.__name__} should not raise error of class {exception_class.__name__}") 

250 except Exception: 

251 pass # Any exception whose class is not monitored is caught and ignored.
