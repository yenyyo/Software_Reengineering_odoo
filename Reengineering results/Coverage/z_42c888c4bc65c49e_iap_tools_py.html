64def mail_prepare_for_domain_search(email, min_email_length=0): 

65 """ Return an email address to use for a domain-based search. For generic 

66 email providers like gmail (see ``_MAIL_DOMAIN_BLACKLIST``) we consider 

67 each email as being independant (and return the whole email). Otherwise 

68 we return only the right-part of the email (aka "mydomain.com" if email is 

69 "Raoul Lachignole" <raoul@mydomain.com>). 

70 

71 :param integer min_email_length: skip if email has not the sufficient minimal 

72 length, indicating a probably fake / wrong value (skip if 0); 

73 """ 

74 if not email: 

75 return False 

76 email_tocheck = email_normalize(email, strict=False) 

77 if not email_tocheck: 

78 email_tocheck = email.casefold() 

79 

80 if email_tocheck and min_email_length and len(email_tocheck) < min_email_length: 

81 return False 

82 

83 parts = email_tocheck.rsplit('@', maxsplit=1) 

84 if len(parts) == 1: 

85 return email_tocheck 

86 email_domain = parts[1] 

87 if email_domain not in _MAIL_DOMAIN_BLACKLIST: 

88 return '@' + email_domain 

89 return email_tocheck

112def iap_jsonrpc(url, method='call', params=None, timeout=15): 

113 """ 

114 Calls the provided JSON-RPC endpoint, unwraps the result and 

115 returns JSON-RPC errors as exceptions. 

116 """ 

117 if hasattr(threading.current_thread(), 'testing') and threading.current_thread().testing: 

118 raise exceptions.AccessError("Unavailable during tests.") 

119 

120 payload = { 

121 'jsonrpc': '2.0', 

122 'method': method, 

123 'params': params, 

124 'id': uuid.uuid4().hex, 

125 } 

126 

127 _logger.info('iap jsonrpc %s', url) 

128 try: 

129 req = requests.post(url, json=payload, timeout=timeout) 

130 req.raise_for_status() 

131 response = req.json() 

132 _logger.info("iap jsonrpc %s responded in %.3f seconds", url, req.elapsed.total_seconds()) 

133 if 'error' in response: 

134 name = response['error']['data'].get('name').rpartition('.')[-1] 

135 if name == 'InsufficientCreditError': 

136 credit_error = InsufficientCreditError(response['error']['data'].get('message')) 

137 credit_error.data = response['error']['data'] 

138 raise credit_error 

139 else: 

140 raise IAPServerError("An error occurred on the IAP server") 

141 return response.get('result') 

142 except requests.exceptions.Timeout: 

143 _logger.warning("iap jsonrpc %s timed out", url) 

144 raise exceptions.AccessError( 

145 _('The request to the service timed out. Please contact the author of the app. The URL it tried to contact was %s', url) 

146 ) 

147 except (requests.exceptions.RequestException, IAPServerError) as e: 

148 _logger.warning("iap jsonrpc %s failed, %s: %s", url, e.__class__.__name__, exception_to_unicode(e)) 

149 raise exceptions.AccessError( 

150 _("An error occurred while reaching %s. Please contact Odoo support if this error persists.", url) 

151 )

163def iap_authorize(env, key, account_token, credit, dbuuid=False, description=None, credit_template=None, ttl=4320): 

164 endpoint = iap_get_endpoint(env) 

165 params = { 

166 'account_token': account_token, 

167 'credit': credit, 

168 'key': key, 

169 'description': description, 

170 'ttl': ttl, 

171 } 

172 if dbuuid: 

173 params.update({'dbuuid': dbuuid}) 

174 try: 

175 transaction_token = iap_jsonrpc(endpoint + '/iap/1/authorize', params=params) 

176 except InsufficientCreditError as e: 

177 if credit_template: 

178 arguments = json.loads(e.args[0]) 

179 arguments['body'] = env['ir.qweb']._render(credit_template) 

180 e.args = (json.dumps(arguments),) 

181 raise e 

182 return transaction_token

206@contextlib.contextmanager 

207def iap_charge(env, key, account_token, credit, dbuuid=False, description=None, credit_template=None, ttl=4320): 

208 """ 

209 Account charge context manager: takes a hold for ``credit`` 

210 amount before executing the body, then captures it if there 

211 is no error, or cancels it if the body generates an exception. 

212 

213 :param str key: service identifier 

214 :param str account_token: user identifier 

215 :param int credit: cost of the body's operation 

216 :param description: a description of the purpose of the charge, 

217 the user will be able to see it in their 

218 dashboard 

219 :type description: str 

220 :param credit_template: a QWeb template to render and show to the 

221 user if their account does not have enough 

222 credits for the requested operation 

223 :param int ttl: transaction time to live in hours. 

224 If the credit are not captured when the transaction 

225 expires, the transaction is canceled 

226 :type credit_template: str 

227 """ 

228 transaction_token = iap_authorize(env, key, account_token, credit, dbuuid, description, credit_template, ttl) 

229 try: 

230 transaction = IapTransaction() 

231 transaction.credit = credit 

232 yield transaction 

233 except Exception as e: 

234 r = iap_cancel(env,transaction_token, key) 

235 raise e 

236 else: 

237 r = iap_capture(env,transaction_token, key, transaction.credit)
