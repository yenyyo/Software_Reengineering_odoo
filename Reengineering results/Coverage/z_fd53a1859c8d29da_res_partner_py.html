87 @api.model 

88 @api.returns('self', lambda value: value.id) 

89 def find_or_create(self, email, assert_valid_email=False): 

90 """ Override to use the email_normalized field. """ 

91 if not email: 

92 raise ValueError(_('An email is required for find_or_create to work')) 

93 

94 parsed_name, parsed_email_normalized = tools.parse_contact_from_email(email) 

95 if not parsed_email_normalized and assert_valid_email: 

96 raise ValueError(_('%(email)s is not recognized as a valid email. This is required to create a new customer.')) 

97 if parsed_email_normalized: 

98 partners = self.search([('email_normalized', '=', parsed_email_normalized)], limit=1) 

99 if partners: 

100 return partners 

101 

102 # We don't want to call `super()` to avoid searching twice on the email 

103 # Especially when the search `email =ilike` cannot be as efficient as 

104 # a search on email_normalized with a btree index 

105 # If you want to override `find_or_create()` your module should depend on `mail` 

106 create_values = {self._rec_name: parsed_name or parsed_email_normalized} 

107 if parsed_email_normalized: # otherwise keep default_email in context 

108 create_values['email'] = parsed_email_normalized 

109 return self.create(create_values)

111 @api.model 

112 def _find_or_create_from_emails(self, emails, additional_values=None): 

113 """ Based on a list of emails, find or create partners. Additional values 

114 can be given to newly created partners. If an email is not unique (e.g. 

115 multi-email input), only the first found email is considered. 

116 

117 Additional values allow to customize the created partner when context 

118 allows to give more information. It data is based on email normalized 

119 as it is the main information used in this method to distinguish or 

120 find partners. 

121 

122 If no valid email is found for a given item, the given value is used to 

123 find partners with same invalid email or create a new one with the wrong 

124 value. It allows updating it afterwards. Notably with notifications 

125 resend it is possible to update emails, if only a typo prevents from 

126 having a real email for example. 

127 

128 :param list emails: list of emails that may be formatted (each input 

129 will be parsed and normalized); 

130 :param dict additional_values: additional values per normalized email 

131 given to create if the partner is not found. Typically used to 

132 propagate a company_id and customer information from related record. 

133 Values for key 'False' are used when creating partner for invalid 

134 emails; 

135 

136 :return: res.partner records in a list, following order of emails. It 

137 is not a recordset, to keep Falsy values. 

138 """ 

139 additional_values = additional_values if additional_values else {} 

140 partners, tocreate_vals_list = self.env['res.partner'], [] 

141 name_emails = [tools.parse_contact_from_email(email) for email in emails] 

142 

143 # find valid emails_normalized, filtering out false / void values, and search 

144 # for existing partners based on those emails 

145 emails_normalized = {email_normalized 

146 for _name, email_normalized in name_emails 

147 if email_normalized} 

148 # find partners for invalid (but not void) emails, aka either invalid email 

149 # either no email and a name that will be used as email 

150 names = { 

151 name.strip() 

152 for name, email_normalized in name_emails 

153 if not email_normalized and name.strip() 

154 } 

155 if emails_normalized or names: 

156 domains = [] 

157 if emails_normalized: 

158 domains.append([('email_normalized', 'in', list(emails_normalized))]) 

159 if names: 

160 domains.append([('email', 'in', list(names))]) 

161 partners += self.search(expression.OR(domains)) 

162 

163 # create partners for valid email without any existing partner. Keep 

164 # only first found occurrence of each normalized email, aka: ('Norbert', 

165 # 'norbert@gmail.com'), ('Norbert With Surname', 'norbert@gmail.com')' 

166 # -> a single partner is created for email 'norbert@gmail.com' 

167 seen = set() 

168 notfound_emails = (emails_normalized - set(partners.mapped('email_normalized'))) if partners else emails_normalized 

169 notfound_name_emails = [ 

170 name_email 

171 for name_email in name_emails 

172 if name_email[1] in notfound_emails and name_email[1] not in seen 

173 and not seen.add(name_email[1]) 

174 ] 

175 tocreate_vals_list += [ 

176 { 

177 self._rec_name: name or email_normalized, 

178 'email': email_normalized, 

179 **additional_values.get(email_normalized, {}), 

180 } 

181 for name, email_normalized in notfound_name_emails 

182 ] 

183 

184 # create partners for invalid emails (aka name and not email_normalized) 

185 # without any existing partner 

186 tocreate_vals_list += [ 

187 { 

188 self._rec_name: name, 

189 'email': name, 

190 **additional_values.get(False, {}), 

191 } 

192 for name in names if name not in partners.mapped('email') 

193 ] 

194 

195 # create partners once 

196 if tocreate_vals_list: 

197 partners += self.create(tocreate_vals_list) 

198 

199 return [ 

200 next( 

201 (partner for partner in partners 

202 if (email_normalized and partner.email_normalized == email_normalized) 

203 or (not email_normalized and email and partner.email == email) 

204 or (not email_normalized and name and partner.name == name) 

205 ), 

206 self.env['res.partner'] 

207 ) 

208 for (name, email_normalized), email in zip(name_emails, emails) 

209 ]

215 def _to_store(self, store: Store, /, *, fields=None, main_user_by_partner=None): 

216 if fields is None: 

217 fields = ["active", "avatar_128", "email", "im_status", "is_company", "name", "user"] 

218 if not self.env.user._is_internal() and "email" in fields: 

219 fields.remove("email") 

220 for partner in self: 

221 data = partner._read_format( 

222 [ 

223 field 

224 for field in fields 

225 if field 

226 not in [ 

227 "avatar_128", 

228 "country", 

229 "display_name", 

230 "isAdmin", 

231 "notification_type", 

232 "signature", 

233 "user", 

234 ] 

235 ], 

236 load=False, 

237 )[0] 

238 if "avatar_128" in fields: 

239 data["avatar_128_access_token"] = limited_field_access_token(partner, "avatar_128") 

240 data["write_date"] = partner.write_date 

241 if "country" in fields: 

242 c = partner.country_id 

243 data["country"] = {"code": c.code, "id": c.id, "name": c.name} if c else False 

244 if "display_name" in fields: 

245 data["displayName"] = partner.display_name 

246 if 'user' in fields: 

247 main_user = main_user_by_partner and main_user_by_partner.get(partner) 

248 if not main_user: 

249 users = partner.with_context(active_test=False).user_ids 

250 internal_users = users - users.filtered("share") 

251 main_user = ( 

252 internal_users[0] 

253 if len(internal_users) > 0 

254 else users[0] if len(users) > 0 else self.env["res.users"] 

255 ) 

256 data['userId'] = main_user.id 

257 data["isInternalUser"] = not main_user.share if main_user else False 

258 if "isAdmin" in fields:

260 if "notification_type" in fields: 

261 data["notification_preference"] = main_user.notification_type 

262 if "signature" in fields: 

263 data["signature"] = main_user.signature 

264 store.add(partner, data)

268 def get_mention_suggestions(self, search, limit=8): 

269 """ Return 'limit'-first partners' such that the name or email matches a 'search' string. 

270 Prioritize partners that are also (internal) users, and then extend the research to all partners. 

271 The return format is a list of partner data (as per returned by `_to_store()`). 

272 """ 

273 domain = self._get_mention_suggestions_domain(search) 

274 partners = self._search_mention_suggestions(domain, limit) 

275 return Store(partners).get_result()

287 @api.model 

288 def _search_mention_suggestions(self, domain, limit, extra_domain=None): 

289 domain_is_user = expression.AND([[('user_ids', '!=', False)], [('user_ids.active', '=', True)], domain]) 

290 priority_conditions = [ 

291 expression.AND([domain_is_user, [('partner_share', '=', False)]]), # Search partners that are internal users 

292 domain_is_user, # Search partners that are users 

293 domain, # Search partners that are not users 

294 ] 

295 if extra_domain: 

296 priority_conditions.append(extra_domain) 

297 partners = self.env['res.partner'] 

298 for domain in priority_conditions: 

299 remaining_limit = limit - len(partners) 

300 if remaining_limit <= 0: 

301 break 

302 # We are using _search to avoid the default order that is 

303 # automatically added by the search method. "Order by" makes the query 

304 # really slow. 

305 query = self._search(expression.AND([[('id', 'not in', partners.ids)], domain]), limit=remaining_limit) 

306 partners |= self.browse(query) 

307 return partners

309 @api.readonly 

310 @api.model 

311 def im_search(self, name, limit=20, excluded_ids=None): 

312 """ Search partner with a name and return its id, name and im_status. 

313 Note : the user must be logged 

314 :param name : the partner name to search 

315 :param limit : the limit of result to return 

316 :param excluded_ids : the ids of excluded partners 

317 """ 

318 # This method is supposed to be used only in the context of channel creation or 

319 # extension via an invite. As both of these actions require the 'create' access 

320 # right, we check this specific ACL. 

321 if excluded_ids is None: 

322 excluded_ids = [] 

323 users = self.env['res.users'].search([ 

324 ('id', '!=', self.env.user.id), 

325 ('name', 'ilike', name), 

326 ('active', '=', True), 

327 ('share', '=', False), 

328 ('partner_id', 'not in', excluded_ids) 

329 ], order='name, id', limit=limit) 

330 return Store(users.partner_id).get_result()
