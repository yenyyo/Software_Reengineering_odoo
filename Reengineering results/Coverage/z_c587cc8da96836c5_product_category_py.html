37 def _compute_product_count(self): 

38 read_group_res = self.env['product.template']._read_group([('categ_id', 'child_of', self.ids)], ['categ_id'], ['__count']) 

39 group_data = {categ.id: count for categ, count in read_group_res} 

40 for categ in self: 

41 product_count = 0 

42 for sub_categ_id in categ.search([('id', 'child_of', categ.ids)]).ids: 

43 product_count += group_data.get(sub_categ_id, 0) 

44 categ.product_count = product_count
