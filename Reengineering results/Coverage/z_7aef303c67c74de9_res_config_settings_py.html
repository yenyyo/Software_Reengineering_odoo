215 @api.depends('country_code') 

216 def _compute_is_account_peppol_eligible(self): 

217 # we want to show Peppol settings only to customers that are eligible for Peppol, 

218 # except countries that are not in Europe 

219 for config in self: 

220 config.is_account_peppol_eligible = config.country_code in PEPPOL_LIST

272 @api.onchange('tax_exigibility') 

273 def _onchange_tax_exigibility(self): 

274 res = {} 

275 tax = self.env['account.tax'].search([ 

276 *self.env['account.tax']._check_company_domain(self.env.company), 

277 ('tax_exigibility', '=', 'on_payment'), 

278 ], limit=1) 

279 if not self.tax_exigibility and tax: 

280 self.tax_exigibility = True 

281 res['warning'] = { 

282 'title': _('Error!'), 

283 'message': _('You cannot disable this setting because some of your taxes are cash basis. ' 

284 'Modify your taxes first before disabling this setting.') 

285 } 

286 return res
