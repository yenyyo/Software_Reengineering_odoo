20 def _compute_name(self): 

21 # Avoid computing the name for reward lines 

22 reward = self.filtered('reward_id') 

23 super(SaleOrderLine, self - reward)._compute_name()

25 @api.depends('reward_id') 

26 def _compute_is_reward_line(self): 

27 for line in self: 

28 line.is_reward_line = bool(line.reward_id)

30 def _compute_tax_id(self): 

31 reward_lines = self.filtered('is_reward_line') 

32 super(SaleOrderLine, self - reward_lines)._compute_tax_id() 

33 # Discount reward line is split per tax, the discount is set on the line but not on the product 

34 # as the product is the generic discount line. 

35 # In case of a free product, retrieving the tax on the line instead of the product won't affect the behavior. 

36 for line in reward_lines: 

37 line = line.with_company(line.company_id) 

38 fpos = line.order_id.fiscal_position_id or line.order_id.fiscal_position_id._get_fiscal_position(line.order_partner_id) 

39 # If company_id is set, always filter taxes by the company 

40 taxes = line.tax_id.filtered(lambda r: not line.company_id or r.company_id == line.company_id) 

41 line.tax_id = fpos.map_tax(taxes)

56 def _reset_loyalty(self, complete=False): 

57 """ 

58 Reset the line(s) to a state which does not impact reward computation. 

59 If complete is set to True we also remove the coupon and reward from the line(s). 

60 This option should be used when the line will be unlinked. 

61 

62 Returns self 

63 """ 

64 vals = { 

65 'points_cost': 0, 

66 'price_unit': 0, 

67 } 

68 if complete: 

69 vals.update({ 

70 'coupon_id': False, 

71 'reward_id': False, 

72 }) 

73 self.write(vals) 

74 return self

76 @api.model_create_multi 

77 def create(self, vals_list): 

78 res = super().create(vals_list) 

79 # Update our coupon points if the order is in a confirmed state 

80 for line in res: 

81 if line.coupon_id and line.points_cost and line.state == 'sale': 

82 line.coupon_id.points -= line.points_cost 

83 line.order_id._update_loyalty_history(line.coupon_id, line.points_cost) 

84 return res

86 def write(self, vals): 

87 cost_in_vals = 'points_cost' in vals 

88 if cost_in_vals: 

89 previous_cost = {l: l.points_cost for l in self} 

90 res = super().write(vals) 

91 if cost_in_vals: 

92 # Update our coupon points if the order is in a confirmed state 

93 for line in self: 

94 if previous_cost[line] != line.points_cost and line.state == 'sale': 

95 line.coupon_id.points += (previous_cost[line] - line.points_cost) 

96 return res

98 def unlink(self): 

99 # Remove related reward lines 

100 reward_coupon_set = {(l.reward_id, l.coupon_id, l.reward_identifier_code) for l in self if l.reward_id} 

101 related_lines = self.env['sale.order.line'] 

102 related_lines |= self.order_id.order_line.filtered(lambda l: (l.reward_id, l.coupon_id, l.reward_identifier_code) in reward_coupon_set) 

103 # Remove the line's coupon from order if it is the last line using that coupon 

104 coupons_to_unlink = self.env['loyalty.card'] 

105 for line in self: 

106 if line.coupon_id: 

107 # 2 cases: 

108 # case 1: coupon has been applied directly 

109 # case 2: coupon was created from a program 

110 if line.coupon_id in line.order_id.applied_coupon_ids: 

111 line.order_id.applied_coupon_ids -= line.coupon_id 

112 elif line.coupon_id.order_id == line.order_id and line.coupon_id.program_id.applies_on == 'current' and\ 

113 not any(oLine.coupon_id == line.coupon_id and oLine not in related_lines for oLine in line.order_id.order_line): 

114 # ondelete='restrict' would prevent deletion of the coupon unlink after unlinking lines 

115 coupons_to_unlink |= line.coupon_id 

116 line.order_id.code_enabled_rule_ids = line.order_id.code_enabled_rule_ids.filtered(lambda r: r.program_id != line.coupon_id.program_id) 

117 # Give back the points if the order is confirmed, points are given back if the order is cancelled but in this case we need to do it directly 

118 for line in related_lines: 

119 if line.state == 'sale': 

120 line.coupon_id.points += line.points_cost 

121 res = super(SaleOrderLine, self | related_lines).unlink() 

122 coupons_to_unlink.sudo().unlink() 

123 return res
