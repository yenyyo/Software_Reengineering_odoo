28 def _make_http_get_request(self, url, params=None): 

29 """ Make an HTTP GET request to the provided URL. 

30 

31 :param str url: The URL to make the request to 

32 :param dict params: The parameters to be sent in the query string 

33 :return: The response of the request 

34 :rtype: :class:`requests.models.Response` 

35 """ 

36 formatted_params = self._format_http_request_payload(payload=params) 

37 return self.opener.get(url, params=formatted_params)

39 def _make_http_post_request(self, url, data=None): 

40 """ Make an HTTP POST request to the provided URL. 

41 

42 :param str url: The URL to make the request to 

43 :param dict data: The data to be send in the request body 

44 :return: The response of the request 

45 :rtype: :class:`requests.models.Response`

47 formatted_data = self._format_http_request_payload(payload=data) 

48 return self.opener.post(url, data=formatted_data)

50 def _format_http_request_payload(self, payload=None): 

51 """ Format a request payload to replace float values by their string representation. 

52 

53 :param dict payload: The payload to format 

54 :return: The formatted payload 

55 :rtype: dict 

56 """ 

57 formatted_payload = {} 

58 if payload is not None: 

59 for k, v in payload.items(): 

60 formatted_payload[k] = str(v) if isinstance(v, float) else v 

61 return formatted_payload

79 def _get_payment_context(self, response): 

80 """Extracts the payment context & other form info (provider & token ids) 

81 from a payment response 

82 

83 :param response: http Response, with a payment form as text 

84 :return: Transaction context (+ provider_ids & token_ids) 

85 :rtype: dict 

86 """ 

87 # Need to specify an HTML parser as parser 

88 # Otherwise void elements (<img>, <link> without a closing / tag) 

89 # are considered wrong and trigger a lxml.etree.XMLSyntaxError 

90 html_tree = objectify.fromstring( 

91 response.text, 

92 parser=etree.HTMLParser(), 

93 ) 

94 payment_form = html_tree.xpath('//form[@id="o_payment_form"]')[0] 

95 values = {} 

96 for key, val in payment_form.items(): 

97 if key.startswith("data-"): 

98 formatted_key = key[5:].replace('-', '_') 

99 if formatted_key.endswith('_id'): 

100 formatted_val = int(val) 

101 elif formatted_key == 'amount': 

102 formatted_val = float(val) 

103 else: 

104 formatted_val = val 

105 values[formatted_key] = formatted_val 

106 

107 payment_options_inputs = html_tree.xpath("//input[@name='o_payment_radio']") 

108 token_ids = [] 

109 payment_method_ids = [] 

110 for p_o_input in payment_options_inputs: 

111 data = dict() 

112 for key, val in p_o_input.items(): 

113 if key.startswith('data-'): 

114 data[key[5:]] = val 

115 if data['payment-option-type'] == 'token': 

116 token_ids.append(int(data['payment-option-id'])) 

117 else: # 'payment_method' 

118 payment_method_ids.append(int(data['payment-option-id'])) 

119 

120 values.update({ 

121 'token_ids': token_ids, 

122 'payment_method_ids': payment_method_ids, 

123 }) 

124 

125 return values
