12class ProductPackaging(models.Model):

31 @api.constrains('barcode') 

32 def _check_barcode_uniqueness(self): 

33 """ With GS1 nomenclature, products and packagings use the same pattern. Therefore, we need 

34 to ensure the uniqueness between products' barcodes and packagings' ones""" 

35 domain = [('barcode', 'in', [b for b in self.mapped('barcode') if b])] 

36 if self.env['product.product'].search_count(domain, limit=1): 

37 raise ValidationError(_("A product already uses the barcode"))

39 def _check_qty(self, product_qty, uom_id, rounding_method="HALF-UP"): 

40 """Check if product_qty in given uom is a multiple of the packaging qty. 

41 If not, rounding the product_qty to closest multiple of the packaging qty 

42 according to the rounding_method "UP", "HALF-UP or "DOWN". 

43 """ 

44 self.ensure_one() 

45 default_uom = self.product_id.uom_id 

46 packaging_qty = default_uom._compute_quantity(self.qty, uom_id) 

47 # We do not use the modulo operator to check if qty is a mltiple of q. Indeed the quantity 

48 # per package might be a float, leading to incorrect results. For example: 

49 # 8 % 1.6 = 1.5999999999999996 

50 # 5.4 % 1.8 = 2.220446049250313e-16 

51 if product_qty and packaging_qty: 

52 rounded_qty = float_round(product_qty / packaging_qty, precision_rounding=1.0, 

53 rounding_method=rounding_method) * packaging_qty 

54 return rounded_qty if float_compare(rounded_qty, product_qty, precision_rounding=default_uom.rounding) else product_qty 

55 return product_qty

57 def _find_suitable_product_packaging(self, product_qty, uom_id): 

58 """ try find in `self` if a packaging's qty in given uom is a divisor of 

59 the given product_qty. If so, return the one with greatest divisor. 

60 """ 

61 packagings = self.sorted(lambda p: p.qty, reverse=True) 

62 for packaging in packagings: 

63 new_qty = packaging._check_qty(product_qty, uom_id) 

64 if new_qty == product_qty: 

65 return packaging 

66 return self.env['product.packaging']

68 def _compute_qty(self, qty, qty_uom=False): 

69 """Returns the qty of this packaging that qty converts to. 

70 A float is returned because there are edge cases where some users use 

71 "part" of a packaging 

72 

73 :param qty: float of product quantity (given in product UoM if no qty_uom provided) 

74 :param qty_uom: Optional uom of quantity 

75 :returns: float of packaging qty 

76 """ 

77 self.ensure_one() 

78 if qty_uom: 

79 qty = qty_uom._compute_quantity(qty, self.product_uom_id) 

80 return float_round(qty / self.qty, precision_rounding=self.product_uom_id.rounding)
