305 def _compute_reference(self, provider_code, prefix=None, separator='-', **kwargs): 

306 """ Compute a unique reference for the transaction. 

307 

308 The reference corresponds to the prefix if no other transaction with that prefix already 

309 exists. Otherwise, it follows the pattern `{computed_prefix}{separator}{sequence_number}` 

310 where: 

311 

312 - `{computed_prefix}` is: 

313 

314 - The provided custom prefix, if any. 

315 - The computation result of :meth:`_compute_reference_prefix` if the custom prefix is not 

316 filled, but the kwargs are. 

317 - `'tx-{datetime}'` if neither the custom prefix nor the kwargs are filled. 

318 

319 - `{separator}` is the string that separates the prefix from the sequence number. 

320 - `{sequence_number}` is the next integer in the sequence of references sharing the same 

321 prefix. The sequence starts with `1` if there is only one matching reference. 

322 

323 .. example:: 

324 

325 - Given the custom prefix `'example'` which has no match with an existing reference, the 

326 full reference will be `'example'`. 

327 - Given the custom prefix `'example'` which matches the existing reference `'example'`, 

328 and the custom separator `'-'`, the full reference will be `'example-1'`. 

329 - Given the kwargs `{'invoice_ids': [1, 2]}`, the custom separator `'-'` and no custom 

330 prefix, the full reference will be `'INV1-INV2'` (or similar) if no existing reference 

331 has the same prefix, or `'INV1-INV2-n'` if `n` existing references have the same 

332 prefix. 

333 

334 :param str provider_code: The code of the provider handling the transaction. 

335 :param str prefix: The custom prefix used to compute the full reference. 

336 :param str separator: The custom separator used to separate the prefix from the suffix. 

337 :param dict kwargs: Optional values passed to :meth:`_compute_reference_prefix` if no custom 

338 prefix is provided. 

339 :return: The unique reference for the transaction. 

340 :rtype: str 

341 """ 

342 # Compute the prefix. 

343 if prefix: 

344 # Replace special characters by their ASCII alternative (é -> e ; ä -> a ; ...) 

345 prefix = unicodedata.normalize('NFKD', prefix).encode('ascii', 'ignore').decode('utf-8') 

346 if not prefix: # Prefix not provided or voided above, compute it based on the kwargs. 

347 prefix = self.sudo()._compute_reference_prefix(provider_code, separator, **kwargs) 

348 if not prefix: # Prefix not computed from the kwargs, fallback on time-based value 

349 prefix = payment_utils.singularize_reference_prefix() 

350 

351 # Compute the sequence number. 

352 reference = prefix # The first reference of a sequence has no sequence number. 

353 if self.sudo().search_count([('reference', '=', prefix)], limit=1): # The reference already has a match 

354 # We now execute a second search on `payment.transaction` to fetch all the references 

355 # starting with the given prefix. The load of these two searches is mitigated by the 

356 # index on `reference`. Although not ideal, this solution allows for quickly knowing 

357 # whether the sequence for a given prefix is already started or not, usually not. An SQL 

358 # query wouldn't help either as the selector is arbitrary and doing that would be an 

359 # open-door to SQL injections. 

360 same_prefix_references = self.sudo().search( 

361 [('reference', '=like', f'{prefix}{separator}%')] 

362 ).with_context(prefetch_fields=False).mapped('reference') 

363 

364 # A final regex search is necessary to figure out the next sequence number. The previous 

365 # search could not rely on alphabetically sorting the reference to infer the largest 

366 # sequence number because both the prefix and the separator are arbitrary. A given 

367 # prefix could happen to be a substring of the reference from a different sequence. 

368 # For instance, the prefix 'example' is a valid match for the existing references 

369 # 'example', 'example-1' and 'example-ref', in that order. Trusting the order to infer 

370 # the sequence number would lead to a collision with 'example-1'. 

371 search_pattern = re.compile(rf'^{re.escape(prefix)}{separator}(\d+)$') 

372 max_sequence_number = 0 # If no match is found, start the sequence with this reference. 

373 for existing_reference in same_prefix_references: 

374 search_result = re.search(search_pattern, existing_reference) 

375 if search_result: # The reference has the same prefix and is from the same sequence 

376 # Find the largest sequence number, if any. 

377 current_sequence = int(search_result.group(1)) 

378 if current_sequence > max_sequence_number: 

379 max_sequence_number = current_sequence 

380 

381 # Compute the full reference. 

382 reference = f'{prefix}{separator}{max_sequence_number + 1}' 

383 return reference

588 """ Create a new transaction with the current transaction as its parent transaction.

590 This happens only in case of a refund or a partial capture (where the initial transaction is 

591 split between smaller transactions, either captured or voided). 

592 

593 Note: self.ensure_one() 

594 

595 :param float amount: The strictly positive amount of the child transaction, in the same 

596 currency as the source transaction. 

597 :param bool is_refund: Whether the child transaction is a refund. 

598 :return: The created child transaction. 

599 :rtype: payment.transaction 

600 """ 

601 self.ensure_one() 

602 

603 if is_refund: 

604 reference_prefix = f'R-{self.reference}' 

605 amount = -amount 

606 operation = 'refund' 

607 else: # Partial capture or void. 

608 reference_prefix = f'P-{self.reference}' 

609 operation = self.operation 

610 

611 return self.create({ 

612 'provider_id': self.provider_id.id, 

613 'payment_method_id': self.payment_method_id.id, 

614 'reference': self._compute_reference(self.provider_code, prefix=reference_prefix), 

615 'amount': amount, 

616 'currency_id': self.currency_id.id, 

617 'token_id': self.token_id.id, 

618 'operation': operation, 

619 'source_transaction_id': self.id, 

620 'partner_id': self.partner_id.id, 

621 **custom_create_values, 

622 })

752 def _update_state(self, allowed_states, target_state, state_message): 

753 """ Update the transactions' state to the target state if the current state allows it. 

754 

755 If the current state is the same as the target state, the transaction is skipped and a log 

756 with level INFO is created. 

757 

758 :param tuple[str] allowed_states: The allowed source states for the target state. 

759 :param str target_state: The target state. 

760 :param str state_message: The message to set as `state_message`. 

761 :return: The recordset of transactions whose state was updated. 

762 :rtype: recordset of `payment.transaction` 

763 """ 

764 def classify_by_state(transactions_): 

765 """ Classify the transactions according to their current state. 

766 

767 For each transaction of the current recordset, if: 

768 

769 - The state is an allowed state: the transaction is flagged as `to process`. 

770 - The state is equal to the target state: the transaction is flagged as `processed`. 

771 - The state matches none of above: the transaction is flagged as `in wrong state`. 

772 

773 :param recordset transactions_: The transactions to classify, as a `payment.transaction` 

774 recordset. 

775 :return: A 3-items tuple of recordsets of classified transactions, in this order: 

776 transactions `to process`, `processed`, and `in wrong state`. 

777 :rtype: tuple(recordset) 

778 """ 

779 txs_to_process_ = transactions_.filtered(lambda _tx: _tx.state in allowed_states) 

780 txs_already_processed_ = transactions_.filtered(lambda _tx: _tx.state == target_state) 

781 txs_wrong_state_ = transactions_ - txs_to_process_ - txs_already_processed_ 

782 

783 return txs_to_process_, txs_already_processed_, txs_wrong_state_ 

784 

785 txs_to_process, txs_already_processed, txs_wrong_state = classify_by_state(self) 

786 for tx in txs_already_processed: 

787 _logger.info( 

788 "tried to write on transaction with reference %s with the same value for the " 

789 "state: %s", 

790 tx.reference, tx.state, 

791 ) 

792 for tx in txs_wrong_state: 

793 _logger.warning( 

794 "tried to write on transaction with reference %(ref)s with illegal value for the " 

795 "state (previous state: %(tx_state)s, target state: %(target_state)s, expected " 

796 "previous state to be in: %(allowed_states)s)", 

797 { 

798 'ref': tx.reference, 

799 'tx_state': tx.state, 

800 'target_state': target_state, 

801 'allowed_states': allowed_states, 

802 }, 

803 ) 

804 txs_to_process.write({ 

805 'state': target_state, 

806 'state_message': state_message, 

807 'last_state_change': fields.Datetime.now(), 

808 'is_post_processed': False, # Reset to allow post-processing again for other states. 

809 }) 

810 return txs_to_process

812 def _update_source_transaction_state(self): 

813 """ Update the state of the source transactions for which all child transactions have 

814 reached a final state. 

815 

816 :return: None 

817 """ 

818 for child_tx in self.filtered('source_transaction_id'): 

819 sibling_txs = child_tx.source_transaction_id.child_transaction_ids.filtered( 

820 lambda tx: tx.state in ['done', 'cancel'] and tx.operation == child_tx.operation 

821 ) 

822 processed_amount = round( 

823 sum(tx.amount for tx in sibling_txs), child_tx.currency_id.decimal_places 

824 ) 

825 if child_tx.source_transaction_id.amount == processed_amount: 

826 state_message = _( 

827 "This transaction has been confirmed following the processing of its partial " 

828 "capture and partial void transactions (%(provider)s).", 

829 provider=child_tx.provider_id.name, 

830 ) 

831 # Call `_update_state` directly instead of `_set_authorized` to avoid looping. 

832 child_tx.source_transaction_id._update_state(('authorized',), 'done', state_message)

836 def _cron_post_process(self): 

837 """ Trigger the post-processing of the transactions that were not handled by the client in 

838 the `poll_status` controller method. 

839 

840 :return: None 

841 """ 

842 txs_to_post_process = self 

843 if not txs_to_post_process: 

844 # Don't try forever to post-process a transaction that doesn't go through. Set the limit 

845 # to 4 days because some providers (PayPal) need that much for the payment verification. 

846 retry_limit_date = datetime.now() - relativedelta.relativedelta(days=4) 

847 # Retrieve all transactions matching the criteria for post-processing 

848 txs_to_post_process = self.search( 

849 [('is_post_processed', '=', False), ('last_state_change', '>=', retry_limit_date)] 

850 ) 

851 for tx in txs_to_post_process: 

852 try: 

853 tx._post_process() 

854 self.env.cr.commit() 

855 except psycopg2.OperationalError: 

856 self.env.cr.rollback() # Rollback and try later. 

857 except Exception as e: 

858 _logger.exception( 

859 "encountered an error while post-processing transaction with reference %s:\n%s", 

860 tx.reference, e 

861 ) 

862 self.env.cr.rollback()

869 apply its specific logic to the transactions, optionally after filtering them based on their

914 def _get_sent_message(self): 

915 """ Return the message stating that the transaction has been requested. 

916 

917 Note: `self.ensure_one()` 

918 

919 :return: The 'transaction sent' message. 

920 :rtype: str 

921 """ 

922 self.ensure_one() 

923 

924 # Choose the message based on the payment flow. 

925 if self.operation in ('online_redirect', 'online_direct'): 

926 message = _( 

927 "A transaction with reference %(ref)s has been initiated (%(provider_name)s).", 

928 ref=self.reference, provider_name=self.provider_id.name 

929 ) 

930 elif self.operation == 'refund': 

931 formatted_amount = format_amount(self.env, -self.amount, self.currency_id) 

932 message = _( 

933 "A refund request of %(amount)s has been sent. The payment will be created soon. " 

934 "Refund transaction reference: %(ref)s (%(provider_name)s).", 

935 amount=formatted_amount, ref=self.reference, provider_name=self.provider_id.name 

936 ) 

937 elif self.operation in ('online_token', 'offline'): 

938 message = _( 

939 "A transaction with reference %(ref)s has been initiated using the payment method " 

940 "%(token)s (%(provider_name)s).", 

941 ref=self.reference, 

942 token=self.token_id._build_display_name(), 

943 provider_name=self.provider_id.name 

944 ) 

945 else: # 'validation' 

946 message = _( 

947 "A transaction with reference %(ref)s has been initiated to save a new payment " 

948 "method (%(provider_name)s)", 

949 ref=self.reference, 

950 provider_name=self.provider_id.name, 

951 ) 

952 return message

954 def _get_received_message(self): 

955 """ Return the message stating that the transaction has been received by the provider. 

956 

957 Note: `self.ensure_one()` 

958 

959 :return: The 'transaction received' message. 

960 :rtype: str 

961 """ 

962 self.ensure_one() 

963 

964 formatted_amount = format_amount(self.env, self.amount, self.currency_id) 

965 if self.state == 'pending': 

966 message = _( 

967 ("The transaction with reference %(ref)s for %(amount)s " 

968 "is pending (%(provider_name)s)."), 

969 ref=self.reference, 

970 amount=formatted_amount, 

971 provider_name=self.provider_id.name 

972 ) 

973 elif self.state == 'authorized': 

974 message = _( 

975 "The transaction with reference %(ref)s for %(amount)s has been authorized " 

976 "(%(provider_name)s).", ref=self.reference, amount=formatted_amount, 

977 provider_name=self.provider_id.name 

978 ) 

979 elif self.state == 'done': 

980 message = _( 

981 "The transaction with reference %(ref)s for %(amount)s has been confirmed " 

982 "(%(provider_name)s).", ref=self.reference, amount=formatted_amount, 

983 provider_name=self.provider_id.name 

984 ) 

985 elif self.state == 'error': 

986 message = _( 

987 "The transaction with reference %(ref)s for %(amount)s encountered an error" 

988 " (%(provider_name)s).", 

989 ref=self.reference, amount=formatted_amount, provider_name=self.provider_id.name 

990 ) 

991 if self.state_message: 

992 message += Markup("<br/>") + _("Error: %s", self.state_message) 

993 else: 

994 message = _( 

995 ("The transaction with reference %(ref)s for %(amount)s is canceled " 

996 "(%(provider_name)s)."), 

997 ref=self.reference, 

998 amount=formatted_amount, 

999 provider_name=self.provider_id.name 

1000 ) 

1001 if self.state_message: 

1002 message += Markup("<br/>") + _("Reason: %s", self.state_message) 

1003 return message
