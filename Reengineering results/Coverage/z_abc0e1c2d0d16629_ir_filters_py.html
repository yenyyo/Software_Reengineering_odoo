63 """Return a domain component for matching filters that are visible in the

65 action_condition = ('action_id', 'in', [action_id, False]) if action_id else ('action_id', '=', False) 

66 embedded_condition = ('embedded_action_id', '=', embedded_action_id) if embedded_action_id else ('embedded_action_id', '=', False)

71 @api.model 

72 def get_filters(self, model, action_id=None, embedded_action_id=None, embedded_parent_res_id=None): 

73 """Obtain the list of filters available for the user on the given model. 

74 

75 :param int model: id of model to find filters for 

76 :param action_id: optional ID of action to restrict filters to this action 

77 plus global filters. If missing only global filters are returned. 

78 The action does not have to correspond to the model, it may only be 

79 a contextual action. 

80 :return: list of :meth:`~osv.read`-like dicts containing the 

81 ``name``, ``is_default``, ``domain``, ``user_id`` (m2o tuple), 

82 ``action_id`` (m2o tuple), ``embedded_action_id`` (m2o tuple), ``embedded_parent_res_id`` 

83 and ``context`` of the matching ``ir.filters``. 

84 """ 

85 # available filters: private filters (user_id=uid) and public filters (uid=NULL), 

86 # and filters for the action (action_id=action_id) or global (action_id=NULL) 

87 user_context = self.env['res.users'].context_get() 

88 action_domain = self._get_action_domain(action_id, embedded_action_id, embedded_parent_res_id) 

89 return self.with_context(user_context).search_read( 

90 action_domain + [('model_id', '=', model), ('user_id', 'in', [self._uid, False])], 

91 ['name', 'is_default', 'domain', 'context', 'user_id', 'sort', 'embedded_action_id', 'embedded_parent_res_id'], 

92 )

94 @api.model 

95 def _check_global_default(self, vals, matching_filters): 

96 """ _check_global_default(dict, list(dict), dict) -> None 

97 

98 Checks if there is a global default for the model_id requested. 

99 

100 If there is, and the default is different than the record being written 

101 (-> we're not updating the current global default), raise an error 

102 to avoid users unknowingly overwriting existing global defaults (they 

103 have to explicitly remove the current default before setting a new one) 

104 

105 This method should only be called if ``vals`` is trying to set 

106 ``is_default`` 

107 

108 :raises odoo.exceptions.UserError: if there is an existing default and 

109 we're not updating it 

110 """ 

111 domain = self._get_action_domain(vals.get('action_id'), vals.get('embedded_action_id'), vals.get('embedded_parent_res_id')) 

112 defaults = self.search(domain + [ 

113 ('model_id', '=', vals['model_id']), 

114 ('user_id', '=', False), 

115 ('is_default', '=', True), 

116 ]) 

117 

118 if not defaults: 

119 return 

120 if matching_filters and (matching_filters[0]['id'] == defaults.id): 

121 return 

122 

123 raise UserError(self.env._("There is already a shared filter set as default for %(model)s, delete or change it before setting a new default", model=vals.get('model_id')))

126 @api.returns('self', lambda value: value.id) 

127 def create_or_replace(self, vals): 

128 action_id = vals.get('action_id') 

129 embedded_action_id = vals.get('embedded_action_id') 

130 if not embedded_action_id and 'embedded_parent_res_id' in vals: 

131 del vals['embedded_parent_res_id'] 

132 embedded_parent_res_id = vals.get('embedded_parent_res_id') 

133 current_filters = self.get_filters(vals['model_id'], action_id, embedded_action_id, embedded_parent_res_id) 

134 matching_filters = [f for f in current_filters 

135 if f['name'].lower() == vals['name'].lower() 

136 # next line looks for matching user_ids (specific or global), i.e. 

137 # f.user_id is False and vals.user_id is False or missing, 

138 # or f.user_id.id == vals.user_id 

139 if (f['user_id'] and f['user_id'][0]) == vals.get('user_id')] 

140 

141 if vals.get('is_default'): 

142 if vals.get('user_id'): 

143 # Setting new default: any other default that belongs to the user 

144 # should be turned off 

145 domain = self._get_action_domain(action_id, embedded_action_id, embedded_parent_res_id) 

146 defaults = self.search(domain + [ 

147 ('model_id', '=', vals['model_id']), 

148 ('user_id', '=', vals['user_id']), 

149 ('is_default', '=', True), 

150 ]) 

151 if defaults: 

152 defaults.write({'is_default': False}) 

153 else: 

154 self._check_global_default(vals, matching_filters) 

155 

156 # When a filter exists for the same (name, model, user) triple, we simply 

157 # replace its definition (considering action_id irrelevant here) 

158 if matching_filters: 

159 matching_filter = self.browse(matching_filters[0]['id']) 

160 matching_filter.write(vals) 

161 return matching_filter 

162 

163 return self.create(vals)
