189def assert_no_dunder_name(code_obj, expr): 

190 """ assert_no_dunder_name(code_obj, expr) -> None 

191 

192 Asserts that the code object does not refer to any "dunder name" 

193 (__$name__), so that safe_eval prevents access to any internal-ish Python 

194 attribute or method (both are loaded via LOAD_ATTR which uses a name, not a 

195 const or a var). 

196 

197 Checks that no such name exists in the provided code object (co_names). 

198 

199 :param code_obj: code object to name-validate 

200 :type code_obj: CodeType 

201 :param str expr: expression corresponding to the code object, for debugging 

202 purposes 

203 :raises NameError: in case a forbidden name (containing two underscores) 

204 is found in ``code_obj`` 

205 

206 .. note:: actually forbids every name containing 2 underscores 

207 """ 

208 for name in code_obj.co_names: 

209 if "__" in name or name in _UNSAFE_ATTRIBUTES: 

210 raise NameError('Access to forbidden name %r (%r)' % (name, expr))

212def assert_valid_codeobj(allowed_codes, code_obj, expr): 

213 """ Asserts that the provided code object validates against the bytecode 

214 and name constraints. 

215 

216 Recursively validates the code objects stored in its co_consts in case 

217 lambdas are being created/used (lambdas generate their own separated code 

218 objects and don't live in the root one) 

219 

220 :param allowed_codes: list of permissible bytecode instructions 

221 :type allowed_codes: set(int) 

222 :param code_obj: code object to name-validate 

223 :type code_obj: CodeType 

224 :param str expr: expression corresponding to the code object, for debugging 

225 purposes 

226 :raises ValueError: in case of forbidden bytecode in ``code_obj`` 

227 :raises NameError: in case a forbidden name (containing two underscores) 

228 is found in ``code_obj`` 

229 """ 

230 assert_no_dunder_name(code_obj, expr) 

231 

232 # set operations are almost twice as fast as a manual iteration + condition 

233 # when loading /web according to line_profiler 

234 code_codes = {i.opcode for i in dis.get_instructions(code_obj)} 

235 if not allowed_codes >= code_codes: 

236 raise ValueError("forbidden opcode(s) in %r: %s" % (expr, ', '.join(opname[x] for x in (code_codes - allowed_codes)))) 

237 

238 for const in code_obj.co_consts: 

239 if isinstance(const, CodeType): 

240 assert_valid_codeobj(allowed_codes, const, 'lambda')

242def test_expr(expr, allowed_codes, mode="eval", filename=None): 

243 """test_expr(expression, allowed_codes[, mode[, filename]]) -> code_object 

244 

245 Test that the expression contains only the allowed opcodes. 

246 If the expression is valid and contains only allowed codes, 

247 return the compiled code object. 

248 Otherwise raise a ValueError, a Syntax Error or TypeError accordingly. 

249 

250 :param filename: optional pseudo-filename for the compiled expression, 

251 displayed for example in traceback frames 

252 :type filename: string 

253 """ 

254 try: 

255 if mode == 'eval': 

256 # eval() does not like leading/trailing whitespace 

257 expr = expr.strip() 

258 code_obj = compile(expr, filename or "", mode) 

259 except (SyntaxError, TypeError, ValueError): 

260 raise 

261 except Exception as e: 

262 raise ValueError('%r while compiling\n%r' % (e, expr)) 

263 assert_valid_codeobj(allowed_codes, code_obj, expr) 

264 return code_obj

267def const_eval(expr): 

268 """const_eval(expression) -> value 

269 

270 Safe Python constant evaluation 

271 

272 Evaluates a string that contains an expression describing 

273 a Python constant. Strings that are not valid Python expressions 

274 or that contain other code besides the constant raise ValueError. 

275 

276 >>> const_eval("10") 

277 10 

278 >>> const_eval("[1,2, (3,4), {'foo':'bar'}]") 

279 [1, 2, (3, 4), {'foo': 'bar'}] 

280 >>> const_eval("1+2") 

281 Traceback (most recent call last): 

282 ... 

283 ValueError: opcode BINARY_ADD not allowed 

284 """ 

285 c = test_expr(expr, _CONST_OPCODES) 

286 return unsafe_eval(c)

347def safe_eval(expr, globals_dict=None, locals_dict=None, mode="eval", nocopy=False, locals_builtins=False, filename=None): 

348 """safe_eval(expression[, globals[, locals[, mode[, nocopy]]]]) -> result 

349 

350 System-restricted Python expression evaluation 

351 

352 Evaluates a string that contains an expression that mostly 

353 uses Python constants, arithmetic expressions and the 

354 objects directly provided in context. 

355 

356 This can be used to e.g. evaluate 

357 an OpenERP domain expression from an untrusted source. 

358 

359 :param filename: optional pseudo-filename for the compiled expression, 

360 displayed for example in traceback frames 

361 :type filename: string 

362 :throws TypeError: If the expression provided is a code object 

363 :throws SyntaxError: If the expression provided is not valid Python 

364 :throws NameError: If the expression provided accesses forbidden names 

365 :throws ValueError: If the expression provided uses forbidden bytecode 

366 """ 

367 if type(expr) is CodeType: 

368 raise TypeError("safe_eval does not allow direct evaluation of code objects.") 

369 

370 # prevent altering the globals/locals from within the sandbox 

371 # by taking a copy. 

372 if not nocopy: 

373 # isinstance() does not work below, we want *exactly* the dict class 

374 if (globals_dict is not None and type(globals_dict) is not dict) \ 

375 or (locals_dict is not None and type(locals_dict) is not dict): 

376 _logger.warning( 

377 "Looks like you are trying to pass a dynamic environment, " 

378 "you should probably pass nocopy=True to safe_eval().") 

379 if globals_dict is not None: 

380 globals_dict = dict(globals_dict) 

381 if locals_dict is not None: 

382 locals_dict = dict(locals_dict) 

383 

384 check_values(globals_dict) 

385 check_values(locals_dict) 

386 

387 if globals_dict is None: 

388 globals_dict = {} 

389 

390 globals_dict['__builtins__'] = _BUILTINS 

391 if locals_builtins: 

392 if locals_dict is None: 

393 locals_dict = {} 

394 locals_dict.update(_BUILTINS) 

395 c = test_expr(expr, _SAFE_OPCODES, mode=mode, filename=filename) 

396 try: 

397 return unsafe_eval(c, globals_dict, locals_dict) 

398 except odoo.exceptions.UserError: 

399 raise 

400 except odoo.exceptions.RedirectWarning: 

401 raise 

402 except werkzeug.exceptions.HTTPException: 

403 raise 

404 except OperationalError: 

405 # Do not hide PostgreSQL low-level exceptions, to let the auto-replay 

406 # of serialized transactions work its magic 

407 raise 

408 except ZeroDivisionError: 

409 raise 

410 except Exception as e: 

411 raise ValueError('%r while evaluating\n%r' % (e, expr)) 

412def test_python_expr(expr, mode="eval"): 

413 try: 

414 test_expr(expr, _SAFE_OPCODES, mode=mode) 

415 except (SyntaxError, TypeError, ValueError) as err: 

416 if len(err.args) >= 2 and len(err.args[1]) >= 4: 

417 error = { 

418 'message': err.args[0], 

419 'filename': err.args[1][0], 

420 'lineno': err.args[1][1], 

421 'offset': err.args[1][2], 

422 'error_line': err.args[1][3], 

423 } 

424 msg = "%s : %s at line %d\n%s" % (type(err).__name__, error['message'], error['lineno'], error['error_line']) 

425 else: 

426 msg = str(err) 

427 return msg 

428 return False

431def check_values(d): 

432 if not d: 

433 return d 

434 for v in d.values(): 

435 if isinstance(v, types.ModuleType): 

436 raise TypeError(f"""Module {v} can not be used in evaluation contexts 

437 

438Prefer providing only the items necessary for your intended use. 

439 

440If a "module" is necessary for backwards compatibility, use 

441`odoo.tools.safe_eval.wrap_module` to generate a wrapper recursively 

442whitelisting allowed attributes. 

443 

444Pre-wrapped modules are provided as attributes of `odoo.tools.safe_eval`. 

445""") 

446 return d

448class wrap_module: 

449 def __init__(self, module, attributes): 

450 """Helper for wrapping a package/module to expose selected attributes 

451 

452 :param module: the actual package/module to wrap, as returned by ``import <module>`` 

453 :param iterable attributes: attributes to expose / whitelist. If a dict, 

454 the keys are the attributes and the values 

455 are used as an ``attributes`` in case the 

456 corresponding item is a submodule 

457 """ 

458 # builtin modules don't have a __file__ at all 

459 modfile = getattr(module, '__file__', '(built-in)') 

460 self._repr = f"<wrapped {module.__name__!r} ({modfile})>" 

461 for attrib in attributes: 

462 target = getattr(module, attrib) 

463 if isinstance(target, types.ModuleType): 

464 target = wrap_module(target, attributes[attrib]) 

465 setattr(self, attrib, target)
