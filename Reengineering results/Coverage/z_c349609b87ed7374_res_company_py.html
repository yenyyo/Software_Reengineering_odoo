50 def iap_enrich_auto(self): 

51 """ Enrich company. This method should be called by automatic processes 

52 and a protection is added to avoid doing enrich in a loop. """ 

53 if self.env.user._is_system(): 

54 for company in self.filtered(lambda company: not company.iap_enrich_auto_done): 

55 company._enrich() 

56 self.iap_enrich_auto_done = True 

57 return True

59 def _enrich(self): 

60 """ This method calls the partner autocomplete service from IAP to enrich 

61 partner related fields of the company. 

62 

63 :return bool: either done, either failed """ 

64 self.ensure_one() 

65 _logger.info("Starting enrich of company %s (%s)", self.name, self.id) 

66 

67 company_domain = self._get_company_domain() 

68 if not company_domain: 

69 return False 

70 

71 company_data = self.env['res.partner'].enrich_company(company_domain, False, self.vat, timeout=COMPANY_AC_TIMEOUT) 

72 if company_data.get('error'): 

73 return False 

74 additional_data = company_data.pop('additional_info', False) 

75 

76 # Keep only truthy values that are not already set on the target partner 

77 # Erase image_1920 even if something is in it. Indeed as partner_autocomplete is probably installed as a 

78 # core app (mail -> iap -> partner_autocomplete auto install chain) it is unlikely that people already 

79 # updated their company logo. 

80 self.env['res.partner']._iap_replace_logo(company_data) 

81 company_data = {field: value for field, value in company_data.items() 

82 if field in self.partner_id._fields and value and (field == 'image_1920' or not self.partner_id[field])} 

83 

84 # for company and childs: from state_id / country_id display_name like to IDs 

85 company_data.update(self._enrich_extract_m2o_id(company_data, ['state_id', 'country_id']))

87 company_data['child_ids'] = [

107 def _enrich_extract_m2o_id(self, iap_data, m2o_fields): 

108 """ Extract m2O ids from data (because of res.partner._format_data_company) """ 

109 extracted_data = {} 

110 for m2o_field in m2o_fields: 

111 relation_data = iap_data.get(m2o_field) 

112 if relation_data and isinstance(relation_data, dict): 

113 extracted_data[m2o_field] = relation_data.get('id', False) 

114 return extracted_data

116 def _enrich_replace_o2m_creation(self, iap_data): 

117 for o2m_field, values in iap_data.items(): 

118 if isinstance(values, list): 

119 commands = [( 

120 0, 0, create_value 

121 ) for create_value in values if isinstance(create_value, dict)] 

122 if commands: 

123 iap_data[o2m_field] = commands 

124 else: 

125 iap_data.pop(o2m_field, None) 

126 return iap_data

128 def _get_company_domain(self): 

129 """ Extract the company domain to be used by IAP services. 

130 The domain is extracted from the website or the email information. 

131 e.g: 

132 - www.info.proximus.be -> proximus.be 

133 - info@proximus.be -> proximus.be """ 

134 self.ensure_one() 

135 

136 company_domain = email_domain_extract(self.email) if self.email else False 

137 if company_domain and company_domain not in iap_tools._MAIL_PROVIDERS: 

138 return company_domain 

139 

140 company_domain = url_domain_extract(self.website) if self.website else False 

141 if not company_domain or company_domain in ['localhost', 'example.com']: 

142 return False 

143 

144 return company_domain
