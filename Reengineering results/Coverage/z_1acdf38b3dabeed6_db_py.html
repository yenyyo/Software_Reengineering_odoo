20def initialize(cr): 

21 """ Initialize a database with for the ORM. 

22 

23 This executes base/data/base_data.sql, creates the ir_module_categories 

24 (taken from each module descriptor file), and creates the ir_module_module 

25 and ir_model_data entries. 

26 

27 """ 

28 try: 

29 f = odoo.tools.misc.file_path('base/data/base_data.sql') 

30 except FileNotFoundError: 

31 m = "File not found: 'base.sql' (provided by module 'base')." 

32 _logger.critical(m) 

33 raise IOError(m) 

34 

35 with odoo.tools.misc.file_open(f) as base_sql_file: 

36 cr.execute(base_sql_file.read()) # pylint: disable=sql-injection 

37 

38 for i in odoo.modules.get_modules(): 

39 mod_path = odoo.modules.get_module_path(i) 

40 if not mod_path: 

41 continue 

42 

43 # This will raise an exception if no/unreadable descriptor file. 

44 info = odoo.modules.get_manifest(i) 

45 

46 if not info: 

47 continue 

48 categories = info['category'].split('/') 

49 category_id = create_categories(cr, categories) 

50 

51 if info['installable']: 

52 state = 'uninstalled' 

53 else: 

54 state = 'uninstallable' 

55 

56 cr.execute('INSERT INTO ir_module_module \ 

57 (author, website, name, shortdesc, description, \ 

58 category_id, auto_install, state, web, license, application, icon, sequence, summary) \ 

59 VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s) RETURNING id', ( 

60 info['author'], 

61 info['website'], i, Json({'en_US': info['name']}), 

62 Json({'en_US': info['description']}), category_id, 

63 info['auto_install'] is not False, state, 

64 info['web'], 

65 info['license'], 

66 info['application'], info['icon'], 

67 info['sequence'], Json({'en_US': info['summary']}))) 

68 id = cr.fetchone()[0] 

69 cr.execute('INSERT INTO ir_model_data \ 

70 (name,model,module, res_id, noupdate) VALUES (%s,%s,%s,%s,%s)', ( 

71 'module_'+i, 'ir.module.module', 'base', id, True)) 

72 dependencies = info['depends'] 

73 for d in dependencies: 

74 cr.execute( 

75 'INSERT INTO ir_module_module_dependency (module_id, name, auto_install_required)' 

76 ' VALUES (%s, %s, %s)', 

77 (id, d, d in (info['auto_install'] or ())) 

78 ) 

79 

80 # Install recursively all auto-installing modules 

81 while True: 

82 # this selects all the auto_install modules whose auto_install_required 

83 # deps are marked as to install 

84 cr.execute(""" 

85 SELECT m.name FROM ir_module_module m 

86 WHERE m.auto_install 

87 AND state not in ('to install', 'uninstallable') 

88 AND NOT EXISTS ( 

89 SELECT 1 FROM ir_module_module_dependency d 

90 JOIN ir_module_module mdep ON (d.name = mdep.name) 

91 WHERE d.module_id = m.id 

92 AND d.auto_install_required 

93 AND mdep.state != 'to install' 

94 )""") 

95 to_auto_install = [x[0] for x in cr.fetchall()] 

96 # however if the module has non-required deps we need to install 

97 # those, so merge-in the modules which have a dependen*t* which is 

98 # *either* to_install or in to_auto_install and merge it in? 

99 cr.execute(""" 

100 SELECT d.name FROM ir_module_module_dependency d 

101 JOIN ir_module_module m ON (d.module_id = m.id) 

102 JOIN ir_module_module mdep ON (d.name = mdep.name) 

103 WHERE (m.state = 'to install' OR m.name = any(%s)) 

104 -- don't re-mark marked modules 

105 AND NOT (mdep.state = 'to install' OR mdep.name = any(%s)) 

106 """, [to_auto_install, to_auto_install]) 

107 to_auto_install.extend(x[0] for x in cr.fetchall()) 

108 

109 if not to_auto_install: break 

110 cr.execute("""UPDATE ir_module_module SET state='to install' WHERE name in %s""", (tuple(to_auto_install),))

112def create_categories(cr, categories): 

113 """ Create the ir_module_category entries for some categories. 

114 

115 categories is a list of strings forming a single category with its 

116 parent categories, like ['Grand Parent', 'Parent', 'Child']. 

117 

118 Return the database id of the (last) category. 

119 

120 """ 

121 p_id = None 

122 category = [] 

123 while categories: 

124 category.append(categories[0]) 

125 xml_id = 'module_category_' + ('_'.join(x.lower() for x in category)).replace('&', 'and').replace(' ', '_') 

126 # search via xml_id (because some categories are renamed) 

127 cr.execute("SELECT res_id FROM ir_model_data WHERE name=%s AND module=%s AND model=%s", 

128 (xml_id, "base", "ir.module.category")) 

129 

130 c_id = cr.fetchone() 

131 if not c_id: 

132 cr.execute('INSERT INTO ir_module_category \ 

133 (name, parent_id) \ 

134 VALUES (%s, %s) RETURNING id', (Json({'en_US': categories[0]}), p_id)) 

135 c_id = cr.fetchone()[0] 

136 cr.execute('INSERT INTO ir_model_data (module, name, res_id, model, noupdate) \ 

137 VALUES (%s, %s, %s, %s, %s)', ('base', xml_id, c_id, 'ir.module.category', True)) 

138 else: 

139 c_id = c_id[0] 

140 p_id = c_id 

141 categories = categories[1:] 

142 return p_id

144class FunctionStatus(IntEnum): 

145 MISSING = 0 # function is not present (falsy) 

146 PRESENT = 1 # function is present but not indexable (not immutable) 

147 INDEXABLE = 2 # function is present and indexable (immutable)

149def has_unaccent(cr): 

150 """ Test whether the database has function 'unaccent' and return its status. 

151 

152 The unaccent is supposed to be provided by the PostgreSQL unaccent contrib 

153 module but any similar function will be picked by OpenERP. 

154 

155 :rtype: FunctionStatus 

156 """ 

157 cr.execute(""" 

158 SELECT p.provolatile 

159 FROM pg_proc p 

160 LEFT JOIN pg_catalog.pg_namespace ns ON p.pronamespace = ns.oid 

161 WHERE p.proname = 'unaccent' 

162 AND p.pronargs = 1 

163 AND ns.nspname = 'public' 

164 """) 

165 result = cr.fetchone() 

166 if not result: 

167 return FunctionStatus.MISSING 

168 # The `provolatile` of unaccent allows to know whether the unaccent function 

169 # can be used to create index (it should be 'i' - means immutable), see 

170 # https://www.postgresql.org/docs/current/catalog-pg-proc.html. 

171 return FunctionStatus.INDEXABLE if result[0] == 'i' else FunctionStatus.PRESENT
