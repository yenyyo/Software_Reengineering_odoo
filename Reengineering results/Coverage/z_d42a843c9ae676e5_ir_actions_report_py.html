15 def _render_qweb_pdf_prepare_streams(self, report_ref, data, res_ids=None): 

16 """Override to add and fill headers, footers and product documents to the sale quotation.""" 

17 result = super()._render_qweb_pdf_prepare_streams(report_ref, data, res_ids=res_ids) 

18 if self._get_report(report_ref).report_name != 'sale.report_saleorder': 

19 return result 

20 

21 orders = self.env['sale.order'].browse(res_ids) 

22 

23 for order in orders: 

24 initial_stream = result[order.id]['stream'] 

25 if initial_stream: 

26 quotation_documents = order.quotation_document_ids 

27 headers = quotation_documents.filtered(lambda doc: doc.document_type == 'header') 

28 footers = quotation_documents - headers 

29 has_product_document = any(line.product_document_ids for line in order.order_line) 

30 

31 if not headers and not has_product_document and not footers: 

32 continue 

33 

34 form_fields_values_mapping = {} 

35 writer = PdfFileWriter() 

36 

37 self_with_order_context = self.with_context( 

38 use_babel=True, lang=order._get_lang() or self.env.user.lang 

39 ) 

40 

41 if headers: 

42 for header in headers: 

43 prefix = f'quotation_document_id_{header.id}__' 

44 self_with_order_context._update_mapping_and_add_pages_to_writer( 

45 writer, header, form_fields_values_mapping, prefix, order 

46 ) 

47 if has_product_document: 

48 for line in order.order_line: 

49 for doc in line.product_document_ids: 

50 # Use both the id of the line and the doc as variants could use the same 

51 # document. 

52 prefix = f'sol_id_{line.id}_product_document_id_{doc.id}__' 

53 self_with_order_context._update_mapping_and_add_pages_to_writer( 

54 writer, doc, form_fields_values_mapping, prefix, order, line 

55 ) 

56 self._add_pages_to_writer(writer, initial_stream.getvalue()) 

57 if footers: 

58 for footer in footers: 

59 prefix = f'quotation_document_id_{footer.id}__' 

60 self_with_order_context._update_mapping_and_add_pages_to_writer( 

61 writer, footer, form_fields_values_mapping, prefix, order 

62 ) 

63 pdf.fill_form_fields_pdf(writer, form_fields=form_fields_values_mapping) 

64 with io.BytesIO() as _buffer: 

65 writer.write(_buffer) 

66 stream = io.BytesIO(_buffer.getvalue()) 

67 result[order.id].update({'stream': stream}) 

68 

69 return result

72 def _update_mapping_and_add_pages_to_writer( 

73 self, writer, document, form_fields_values_mapping, prefix, order, order_line=None 

74 ): 

75 """ Update the mapping with the field-value of the document, and add the doc to the writer. 

76 

77 Note: document.ensure_one(), order.ensure_one(), order_line and order_line.ensure_one() 

78 

79 :param PdfFileWriter writer: the writer to which pages needs to be added 

80 :param recordset document: the document that needs to be added to the writer and get its 

81 form fields mapped. Either a quotation.document or a 

82 product.document. 

83 :param dict form_fields_values_mapping: the existing prefixed form field names - values that 

84 will be updated to add those of the current document 

85 :param str prefix: the prefix needed to update existing form field name, to be able to add 

86 the correct values in fields with the same name but on different 

87 documents, either customizable fields or dynamic fields of different sale 

88 order lines. 

89 :param recordset order: the sale order from where to take the values 

90 :param recordset order_line: the sale order line from where to take the values (optional) 

91 return: None 

92 """ 

93 document.ensure_one() 

94 order.ensure_one() 

95 order_line and order_line.ensure_one() 

96 

97 for form_field in document.form_field_ids: 

98 if form_field.path: # Dynamic field 

99 field_value = self._get_value_from_path(form_field, order, order_line) 

100 else: # Customizable field 

101 field_value = self._get_custom_value_from_order( 

102 document, form_field.name, order, order_line 

103 ) 

104 form_fields_values_mapping[prefix + form_field.name] = field_value 

105 

106 # Avoid useless update of the pdf when no form field and just add the pdf 

107 prefix = prefix if document.form_field_ids else None 

108 decoded_document = base64.b64decode(document.datas) 

109 self._add_pages_to_writer(writer, decoded_document, prefix)

112 def _get_value_from_path(self, form_field, order, order_line=None): 

113 """ Get the string value by following the path indicated in the record form_field. 

114 

115 :param recordset form_field: sale.pdf.form.field that has a valid path. 

116 :param recordset order: sale.order from where the values and timezone need to be taken 

117 :param recordset order_line: sale.order.line from where the values need to be taken 

118 (optional, only for product.document) 

119 :return: value that need to be shown in the final pdf. Multiple values are joined by ', ' 

120 :rtype: str 

121 """ 

122 tz = order.partner_id.tz or order.env.user.tz or 'UTC' 

123 base_record = order_line or order 

124 path = form_field.path 

125 

126 # If path = 'order_id.order_line.product_id.name' 

127 path = path.split('.') # ['order_id', 'order_line', 'product_id', 'name'] 

128 # Sudo to be able to follow the path set by the admin 

129 records = base_record.sudo().mapped('.'.join(path[:-1])) # product.product(id1, id2, ...) 

130 field_name = path[-1] # 'name' 

131 

132 def _get_formatted_value(self): 

133 # self must be named so to be considered in the translation logic 

134 field_ = records._fields[field_name] 

135 field_type_ = field_.type 

136 for record_ in records: 

137 value_ = record_[field_name] 

138 if field_type_ == 'boolean': 

139 formatted_value_ = _("Yes") if value_ else _("No") 

140 elif field_type_ == 'monetary': 

141 currency_id_ = record_[field_.get_currency_field(record_)] 

142 formatted_value_ = format_amount( 

143 self.env, value_, currency_id_ or order.currency_id 

144 ) 

145 elif not value_: 

146 formatted_value_ = '' 

147 elif field_type_ == 'date': 

148 formatted_value_ = format_date(self.env, value_) 

149 elif field_type_ == 'datetime': 

150 formatted_value_ = format_datetime(self.env, value_, tz=tz, dt_format=False) 

151 elif field_type_ == 'selection' and value_: 

152 formatted_value_ = dict(field_._description_selection(self.env))[value_] 

153 elif field_type_ in {'one2many', 'many2one', 'many2many'}: 

154 formatted_value_ = ', '.join([v.display_name for v in value_]) 

155 else: 

156 formatted_value_ = str(value_) 

157 

158 yield formatted_value_ 

159 

160 return ', '.join(_get_formatted_value(self))

163 def _get_custom_value_from_order(self, document, form_field_name, order, order_line): 

164 """ Get the custom value of a form field directly from the order. 

165 

166 :param recordset document: the document that needs to be added to the writer and get its 

167 form fields mapped. Either a quotation.document or a 

168 product.document. 

169 :param str form_field_name: the name of the form field as present in the PDF. 

170 :param recordset order: the sale order from where to take the existing mapping. 

171 :param recordset order_line: the sale order line linked to the document (optional) 

172 :return: value that need to be shown in the final pdf. 

173 :rtype: str 

174 """ 

175 existing_mapping = json.loads(order.customizable_pdf_form_fields) 

176 if order_line: 

177 base_values = existing_mapping.get('line', {}).get(str(order_line.id), {}) 

178 elif document.document_type == 'header': 

179 base_values = existing_mapping.get('header', {}) 

180 else: 

181 base_values = existing_mapping.get('footer', {}) 

182 custom_form_fields = base_values.get(str(document.id), {}).get('custom_form_fields') 

183 return custom_form_fields.get(form_field_name, "")

185 @api.model 

186 def _add_pages_to_writer(self, writer, document, prefix=None): 

187 """Add a PDF doc to the writer and fill the form text fields present in the pages if needed. 

188 

189 :param PdfFileWriter writer: the writer to which pages needs to be added 

190 :param bytes document: the document to add in the final pdf 

191 :param str prefix: the prefix needed to update existing form field name, if any, to be able 

192 to add the correct values in fields with the same name but on different 

193 documents, either customizable fields or dynamic fields of different sale 

194 order lines. (optional) 

195 :return: None 

196 """ 

197 reader = PdfFileReader(io.BytesIO(document), strict=False) 

198 

199 field_names = set() 

200 if prefix: 

201 field_names = reader.getFormTextFields() 

202 

203 for page_id in range(reader.getNumPages()): 

204 page = reader.getPage(page_id) 

205 if prefix and page.get('/Annots'): 

206 # Modifying the annots that hold every information about the form fields 

207 for j in range(len(page['/Annots'])): 

208 reader_annot = page['/Annots'][j].getObject() 

209 if reader_annot.get('/T') in field_names: 

210 # Prefix all form fields in the document with the document identifier. 

211 # This is necessary to know which value needs to be taken when filling the forms. 

212 form_key = reader_annot.get('/T') 

213 new_key = prefix + form_key 

214 

215 # Modifying the form flags to force some characteristics 

216 # 1. make all text fields read-only 

217 # 2. make all text fields support multiline 

218 form_flags = reader_annot.get('/Ff', 0) 

219 readonly_flag = 1 # 1st bit sets readonly 

220 multiline_flag = 1 << 12 # 13th bit sets multiline text 

221 new_flags = form_flags | readonly_flag | multiline_flag 

222 

223 reader_annot.update({ 

224 NameObject("/T"): createStringObject(new_key), 

225 NameObject("/Ff"): NumberObject(new_flags), 

226 }) 

227 writer.addPage(page)
