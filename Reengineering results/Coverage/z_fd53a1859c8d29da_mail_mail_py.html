218 def process_email_queue(self, ids=None, batch_size=1000): 

219 """Send immediately queued messages, committing after each 

220 message is sent - this is not transactional and should 

221 not be called during another transaction! 

222 

223 A maximum of 1K MailMail (configurable using 'mail.mail.queue.batch.size' 

224 optional ICP) are fetched in order to keep time under control. 

225 

226 :param list ids: optional list of emails ids to send. If given only 

227 scheduled and outgoing emails within this ids list 

228 are sent; 

229 :param dict context: if a 'filters' key is present in context, 

230 this value will be used as an additional 

231 filter to further restrict the outgoing 

232 messages to send (by default all 'outgoing' 

233 'scheduled' messages are sent). 

234 """ 

235 domain = [ 

236 '&', 

237 ('state', '=', 'outgoing'), 

238 '|', 

239 ('scheduled_date', '=', False), 

240 ('scheduled_date', '<=', datetime.datetime.utcnow()), 

241 ] 

242 if 'filters' in self._context: 

243 domain.extend(self._context['filters']) 

244 batch_size = int(self.env['ir.config_parameter'].sudo().get_param('mail.mail.queue.batch.size', batch_size)) 

245 send_ids = self.search(domain, limit=batch_size if not ids else batch_size * 10).ids 

246 if not ids: 

247 ids_done = set() 

248 total = len(send_ids) if len(send_ids) < batch_size else self.search_count(domain) 

249 

250 def post_send_callback(ids): 

251 """ Track mail ids that have been sent, and notify cron progress accordingly. """ 

252 ids_done.update(send_ids) 

253 self.env['ir.cron']._notify_progress(done=len(ids_done), remaining=total - len(ids_done)) 

254 else: 

255 send_ids = list(set(send_ids) & set(ids)) 

256 post_send_callback = None 

257 

258 send_ids.sort() 

259 

260 res = None 

261 try: 

262 # auto-commit except in testing mode 

263 auto_commit = not getattr(threading.current_thread(), 'testing', False) 

264 res = self.browse(send_ids).send(auto_commit=auto_commit, post_send_callback=post_send_callback) 

265 except Exception: 

266 _logger.exception("Failed processing mail queue") 

267 

268 return res

309 def _parse_scheduled_datetime(self, scheduled_datetime): 

310 """ Taking an arbitrary datetime (either as a date, a datetime or a string) 

311 try to parse it and return a datetime timezoned to UTC. 

312 

313 If no specific timezone information is given, we consider it as being 

314 given in UTC, as all datetime values given to the server. Trying to 

315 guess its timezone based on user or flow would be strange as this is 

316 not standard. When manually creating datetimes for mail.mail scheduled 

317 date, business code should ensure either a timezone info is set, either 

318 it is converted into UTC. 

319 

320 Using yearfirst when parsing str datetimes eases parser's job when 

321 dealing with the hard-to-parse trio (01/04/09 -> ?). In most use cases 

322 year will be given first as this is the expected default formatting. 

323 

324 :return datetime: parsed datetime (or False if parser failed) 

325 """ 

326 if isinstance(scheduled_datetime, datetime.datetime): 

327 parsed_datetime = scheduled_datetime 

328 elif isinstance(scheduled_datetime, datetime.date): 

329 parsed_datetime = datetime.combine(scheduled_datetime, datetime.time.min) 

330 else: 

331 try: 

332 parsed_datetime = parse(scheduled_datetime, yearfirst=True) 

333 except (ValueError, TypeError): 

334 parsed_datetime = False 

335 if parsed_datetime: 

336 parsed_datetime = parsed_datetime.replace(microsecond=0) 

337 if not parsed_datetime.tzinfo: 

338 parsed_datetime = pytz.utc.localize(parsed_datetime) 

339 else: 

340 try: 

341 parsed_datetime = parsed_datetime.astimezone(pytz.utc) 

342 except Exception: 

343 pass 

344 return parsed_datetime

351 def _estimate_email_size(self, headers, body, attachments_size): 

352 """ Estimate the email size, incorporating a small added security margin. 

353 

354 :param dict headers: email headers 

355 :param str body: email body 

356 :param list attachments_size: list of attachment size in bytes 

357 """ 

358 return ( 

359 len(json.dumps(headers or {}, ensure_ascii=False).encode(errors='ignore')) + 

360 len((body or '').encode(errors='ignore')) + 

361 sum(attachments_size) * 8 / 6 + # 8 / 6 factor because base 64 are 6 bits coded in 8 bits 

362 10 * 1024 # adding 10Ko as security 

363 )

396 def _prepare_outgoing_list(self, mail_server=False, recipients_follower_status=None): 

397 """ Return a list of emails to send based on current mail.mail. Each 

398 is a dictionary for specific email values, depending on a partner, or 

399 generic to the whole recipients given by mail.email_to. 

400 

401 :param mail_server: <ir.mail_server> mail server that will be used to send the mails, 

402 False if it is the default one 

403 :param set recipients_follower_status: see ``Followers._get_mail_recipients_follower_status()`` 

404 :return list: list of dicts used in IrMailServer.build_email() 

405 """ 

406 self.ensure_one() 

407 body = self._prepare_outgoing_body() 

408 

409 # headers 

410 headers = {} 

411 if self.headers: 

412 try: 

413 headers = ast.literal_eval(self.headers) 

414 except (ValueError, TypeError) as e: 

415 _logger.warning( 

416 'Evaluation error when evaluating mail headers (received %r): %s', 

417 self.headers, e, 

418 ) 

419 # global except as we don't want to crash the queue just due to a malformed 

420 # headers value 

421 except Exception as e: 

422 _logger.warning( 

423 'Unknown error when evaluating mail headers (received %r): %s', 

424 self.headers, e, 

425 ) 

426 headers['X-Odoo-Message-Id'] = self.message_id 

427 headers.setdefault('Return-Path', self.record_alias_domain_id.bounce_email or self.env.company.bounce_email) 

428 

429 # prepare recipients: use email_to if defined then check recipient_ids 

430 # that receive a specific email, notably due to link shortening / redirect 

431 # that is recipients-dependent. Keep original email/partner as this is 

432 # used in post-processing to know failures, like missing recipients 

433 email_list = [] 

434 if self.email_to: 

435 email_to_normalized = tools.mail.email_normalize_all(self.email_to) 

436 email_to = tools.mail.email_split_and_format_normalize(self.email_to) 

437 email_list.append({ 

438 'email_cc': [], 

439 'email_to': email_to, 

440 # list of normalized emails to help extract_rfc2822 

441 'email_to_normalized': email_to_normalized, 

442 # keep raw initial value for incoming pre processing of outgoing emails 

443 'email_to_raw': self.email_to or '', 

444 'partner_id': False, 

445 }) 

446 # add all cc once, either to the first "To", either as a single entry (do not mix 

447 # with partner-specific sending) 

448 if self.email_cc: 

449 if email_list: 

450 email_list[0]['email_cc'] = tools.mail.email_split_and_format_normalize(self.email_cc) 

451 email_list[0]['email_to_normalized'] += tools.mail.email_normalize_all(self.email_cc) 

452 else: 

453 email_list.append({ 

454 'email_cc': tools.mail.email_split_and_format_normalize(self.email_cc), 

455 'email_to': [], 

456 'email_to_normalized': tools.mail.email_normalize_all(self.email_cc), 

457 'email_to_raw': False, 

458 'partner_id': False, 

459 }) 

460 # specific behavior to customize the send email for notified partners 

461 for partner in self.recipient_ids: 

462 # check partner email content 

463 email_to_normalized = tools.mail.email_normalize_all(partner.email) 

464 email_to = [ 

465 tools.formataddr((partner.name or "", email or "False")) 

466 for email in email_to_normalized or [partner.email] 

467 ] 

468 email_list.append({ 

469 'email_cc': [], 

470 'email_to': email_to, 

471 # list of normalized emails to help extract_rfc2822 

472 'email_to_normalized': email_to_normalized, 

473 # keep raw initial value for incoming pre processing of outgoing emails 

474 'email_to_raw': partner.email or '', 

475 'partner_id': partner, 

476 }) 

477 

478 attachments = self.attachment_ids 

479 # Prepare attachments: 

480 # Remove attachments if user send the link with the access_token. 

481 if body and attachments: 

482 link_ids = {int(link) for link in re.findall(r'/web/(?:content|image)/([0-9]+)', body)} 

483 if link_ids: 

484 attachments = attachments - self.env['ir.attachment'].browse(list(link_ids)) 

485 # Turn remaining attachments into links if they are too heavy and 

486 # their ownership are business models (i.e. something != mail.message, 

487 # otherwise they will be deleted along with the mail message leading to a 404) 

488 if record_owned_attachments := attachments.sudo().filtered( 

489 lambda a: a.res_model and a.res_id and a.res_model != 'mail.message'): 

490 estimated_email_size_bytes = self._estimate_email_size( 

491 headers, body, [a.file_size for a in attachments.sudo()]) 

492 max_email_size_bytes = (mail_server or self.env['ir.mail_server'] 

493 ).sudo()._get_max_email_size() * 1024 * 1024 

494 if estimated_email_size_bytes > max_email_size_bytes: 

495 # Remove attachments and prepare downloadable links to be added in the body 

496 record_owned_attachments.sudo().generate_access_token() 

497 attachments_links = self.env['ir.qweb']._render('mail.mail_attachment_links', 

498 {'attachments': record_owned_attachments}) 

499 body = tools.mail.append_content_to_html(body, attachments_links, plaintext=False) 

500 attachments -= record_owned_attachments 

501 # Prepare the remaining attachment (those not embedded as link) 

502 # load attachment binary data with a separate read(), as prefetching all 

503 # `datas` (binary field) could bloat the browse cache, triggering 

504 # soft/hard mem limits with temporary data. 

505 email_attachments = [(a['name'], a['raw'], a['mimetype']) 

506 for a in attachments.sudo().read(['name', 'raw', 'mimetype']) 

507 if a['raw'] is not False] 

508 

509 # Build final list of email values with personalized body for recipient 

510 results = [] 

511 for email_values in email_list: 

512 partner_id = email_values['partner_id'] 

513 body_personalized = self._personalize_outgoing_body(body, partner_id, recipients_follower_status) 

514 results.append({ 

515 'attachments': email_attachments, 

516 'body': body_personalized, 

517 'body_alternative': tools.html2plaintext(body_personalized), 

518 'email_cc': email_values['email_cc'], 

519 'email_from': self.email_from, 

520 'email_to': email_values['email_to'], 

521 'email_to_normalized': email_values['email_to_normalized'], 

522 'email_to_raw': email_values['email_to_raw'], 

523 'headers': headers, 

524 'message_id': self.message_id, 

525 'object_id': f'{self.res_id}-{self.model}' if self.res_id else '', 

526 'partner_id': partner_id, 

527 'references': self.references, 

528 'reply_to': self.reply_to, 

529 'subject': self.subject, 

530 }) 

531 

532 return results

534 def _split_by_mail_configuration(self): 

535 """Group the <mail.mail> based on their "email_from", their "alias domain" 

536 and their "mail_server_id". 

537 

538 The <mail.mail> will have the "same sending configuration" if they have the same 

539 mail server, alias domain and mail from. For performance purpose, we can use an SMTP 

540 session in batch and therefore we need to group them by the parameter that will 

541 influence the mail server used. 

542 

543 The same "sending configuration" may repeat in order to limit batch size 

544 according to the `mail.session.batch.size` system parameter. 

545 

546 Return iterators over 

547 mail_server_id, email_from, Records<mail.mail>.ids 

548 """ 

549 mail_values = self.read(['id', 'email_from', 'mail_server_id', 'record_alias_domain_id']) 

550 

551 # First group the <mail.mail> per mail_server_id, per alias_domain (if no server) and per email_from 

552 group_per_email_from = defaultdict(list) 

553 for values in mail_values: 

554 # protect against ill-formatted email_from when formataddr was used on an already formatted email 

555 emails_from = tools.mail.email_split_and_format_normalize(values['email_from']) 

556 email_from = emails_from[0] if emails_from else values['email_from'] 

557 mail_server_id = values['mail_server_id'][0] if values['mail_server_id'] else False 

558 alias_domain_id = values['record_alias_domain_id'][0] if values['record_alias_domain_id'] else False 

559 key = (mail_server_id, alias_domain_id, email_from) 

560 group_per_email_from[key].append(values['id']) 

561 

562 # Then find the mail server for each email_from and group the <mail.mail> 

563 # per mail_server_id and smtp_from 

564 mail_servers = self.env['ir.mail_server'].sudo().search([], order='sequence, id') 

565 group_per_smtp_from = defaultdict(list) 

566 for (mail_server_id, alias_domain_id, email_from), mail_ids in group_per_email_from.items(): 

567 if not mail_server_id: 

568 mail_server = self.env['ir.mail_server'] 

569 if alias_domain_id: 

570 alias_domain = self.env['mail.alias.domain'].sudo().browse(alias_domain_id) 

571 mail_server = mail_server.with_context( 

572 domain_notifications_email=alias_domain.default_from_email, 

573 domain_bounce_address=alias_domain.bounce_email, 

574 ) 

575 mail_server, smtp_from = mail_server._find_mail_server(email_from, mail_servers) 

576 mail_server_id = mail_server.id if mail_server else False 

577 else: 

578 smtp_from = email_from 

579 

580 group_per_smtp_from[(mail_server_id, alias_domain_id, smtp_from)].extend(mail_ids) 

581 

582 batch_size = int(self.env['ir.config_parameter'].sudo().get_param('mail.session.batch.size')) or 1000 

583 for (mail_server_id, alias_domain_id, smtp_from), record_ids in group_per_smtp_from.items(): 

584 for batch_ids in tools.split_every(batch_size, record_ids): 

585 yield mail_server_id, alias_domain_id, smtp_from, batch_ids

588 """Queues the email to be sent after the commit of the current cursor. 

589 

590 Useful to send an email only if a transaction is successful. 

591 """ 

592 # send immediately on same cursor when testing as we don't want 

593 # to actually commit during tests 

594 if getattr(threading.current_thread(), 'testing', False): 

595 self.send() 

596 return

609 def send(self, auto_commit=False, raise_exception=False, post_send_callback=None): 

610 """ Sends the selected emails immediately, ignoring their current 

611 state (mails that have already been sent should not be passed 

612 unless they should actually be re-sent). 

613 Emails successfully delivered are marked as 'sent', and those 

614 that fail to be deliver are marked as 'exception', and the 

615 corresponding error mail is output in the server logs. 

616 

617 :param bool auto_commit: whether to force a commit of the mail status 

618 after sending each mail (meant only for scheduler processing); 

619 should never be True during normal transactions (default: False) 

620 :param bool raise_exception: whether to raise an exception if the 

621 email sending process has failed 

622 :param post_send_callback: an optional function, called as ``post_send_callback(ids, force=False)``, 

623 with the mail ids that have been sent; calls with redundant ids 

624 are possible 

625 :return: True 

626 """ 

627 for mail_server_id, alias_domain_id, smtp_from, batch_ids in self._split_by_mail_configuration(): 

628 smtp_session = None 

629 try: 

630 smtp_session = self.env['ir.mail_server'].connect(mail_server_id=mail_server_id, smtp_from=smtp_from) 

631 except Exception as exc: 

632 if raise_exception: 

633 # To be consistent and backward compatible with mail_mail.send() raised 

634 # exceptions, it is encapsulated into an Odoo MailDeliveryException 

635 raise MailDeliveryException(_('Unable to connect to SMTP Server'), exc) 

636 else: 

637 batch = self.browse(batch_ids) 

638 batch.write({'state': 'exception', 'failure_reason': tools.exception_to_unicode(exc)}) 

639 batch._postprocess_sent_message(success_pids=[], failure_type="mail_smtp") 

640 else: 

641 mail_server = self.env['ir.mail_server'].browse(mail_server_id) 

642 self.browse(batch_ids)._send( 

643 auto_commit=auto_commit, 

644 raise_exception=raise_exception, 

645 smtp_session=smtp_session, 

646 alias_domain_id=alias_domain_id, 

647 mail_server=mail_server, 

648 post_send_callback=post_send_callback, 

649 ) 

650 if not modules.module.current_test: 

651 _logger.info( 

652 'Sent batch %s emails via mail server ID #%s', 

653 len(batch_ids), mail_server_id) 

654 finally: 

655 if smtp_session: 

656 smtp_session.quit()

658 def _send(self, auto_commit=False, raise_exception=False, smtp_session=None, alias_domain_id=False, 

659 mail_server=False, post_send_callback=None): 

660 IrMailServer = self.env['ir.mail_server'] 

661 # Only retrieve recipient followers of the mails if needed 

662 mails_with_unfollow_link = self.filtered(lambda m: m.body_html and '/mail/unfollow' in m.body_html) 

663 recipients_follower_status = ( 

664 None if not mails_with_unfollow_link 

665 else self.env['mail.followers']._get_mail_recipients_follower_status(mails_with_unfollow_link.ids) 

666 ) 

667 

668 for mail_id in self.ids: 

669 success_pids = [] 

670 failure_reason = None 

671 failure_type = None 

672 mail = None 

673 try: 

674 mail = self.browse(mail_id) 

675 if mail.state != 'outgoing': 

676 continue 

677 

678 # Writing on the mail object may fail (e.g. lock on user) which 

679 # would trigger a rollback *after* actually sending the email. 

680 # To avoid sending twice the same email, provoke the failure earlier 

681 mail.write({ 

682 'state': 'exception', 

683 'failure_reason': _('Error without exception. Probably due to sending an email without computed recipients.'), 

684 }) 

685 # Update notification in a transient exception state to avoid concurrent 

686 # update in case an email bounces while sending all emails related to current 

687 # mail record. 

688 notifs = self.env['mail.notification'].search([ 

689 ('notification_type', '=', 'email'), 

690 ('mail_mail_id', 'in', mail.ids), 

691 ('notification_status', 'not in', ('sent', 'canceled')) 

692 ]) 

693 if notifs: 

694 notif_msg = _('Error without exception. Probably due to concurrent access update of notification records. Please see with an administrator.') 

695 notifs.sudo().write({ 

696 'notification_status': 'exception', 

697 'failure_type': 'unknown', 

698 'failure_reason': notif_msg, 

699 }) 

700 # `test_mail_bounce_during_send`, force immediate update to obtain the lock. 

701 # see rev. 56596e5240ef920df14d99087451ce6f06ac6d36 

702 notifs.flush_recordset(['notification_status', 'failure_type', 'failure_reason']) 

703 

704 # protect against ill-formatted email_from when formataddr was used on an already formatted email 

705 emails_from = tools.mail.email_split_and_format_normalize(mail.email_from) 

706 email_from = emails_from[0] if emails_from else mail.email_from 

707 

708 # build an RFC2822 email.message.Message object and send it without queuing 

709 res = None 

710 # TDE note: could be great to pre-detect missing to/cc and skip sending it 

711 # to go directly to failed state update 

712 email_list = mail._prepare_outgoing_list( 

713 mail_server=mail_server or mail.mail_server_id, 

714 recipients_follower_status=recipients_follower_status, 

715 ) 

716 

717 # send each sub-email 

718 for email in email_list: 

719 # give indication to 'send_mail' about emails already considered 

720 # as being valid 

721 email_to_normalized = email.pop('email_to_normalized', []) 

722 # if given, contextualize sending using alias domains 

723 if alias_domain_id: 

724 alias_domain = self.env['mail.alias.domain'].sudo().browse(alias_domain_id) 

725 SendIrMailServer = IrMailServer.with_context( 

726 domain_notifications_email=alias_domain.default_from_email, 

727 domain_bounce_address=email['headers'].get('Return-Path') or alias_domain.bounce_email, 

728 send_validated_to=email_to_normalized, 

729 ) 

730 else: 

731 SendIrMailServer = IrMailServer.with_context(send_validated_to=email_to_normalized) 

732 msg = SendIrMailServer.build_email( 

733 email_from=email_from, 

734 email_to=email['email_to'], 

735 subject=email['subject'], 

736 body=email['body'], 

737 body_alternative=email['body_alternative'], 

738 email_cc=email['email_cc'], 

739 reply_to=email['reply_to'], 

740 attachments=email['attachments'], 

741 message_id=email['message_id'], 

742 references=email['references'], 

743 object_id=email['object_id'], 

744 subtype='html', 

745 subtype_alternative='plain', 

746 headers=email['headers'], 

747 ) 

748 processing_pid = email.pop("partner_id", None) 

749 try: 

750 res = SendIrMailServer.send_email( 

751 msg, mail_server_id=mail.mail_server_id.id, smtp_session=smtp_session) 

752 if processing_pid: 

753 success_pids.append(processing_pid) 

754 processing_pid = None 

755 except AssertionError as error: 

756 if str(error) == IrMailServer.NO_VALID_RECIPIENT: 

757 # if we have a list of void emails for email_list -> email missing, otherwise generic email failure 

758 if not email.get('email_to') and failure_type != "mail_email_invalid": 

759 failure_type = "mail_email_missing" 

760 else: 

761 failure_type = "mail_email_invalid" 

762 # No valid recipient found for this particular 

763 # mail item -> ignore error to avoid blocking 

764 # delivery to next recipients, if any. If this is 

765 # the only recipient, the mail will show as failed. 

766 _logger.info("Ignoring invalid recipients for mail.mail %s: %s", 

767 mail.message_id, email.get('email_to')) 

768 else: 

769 raise 

770 if res: # mail has been sent at least once, no major exception occurred 

771 mail.write({'state': 'sent', 'message_id': res, 'failure_reason': False}) 

772 if not modules.module.current_test: 

773 _logger.info( 

774 "Mail with ID %r and Message-Id %r from %r to (redacted) %r successfully sent", 

775 mail.id, 

776 mail.message_id, 

777 tools.email_normalize(msg['from']), 

778 tools.mail.email_anonymize(tools.email_normalize(msg['to'])) 

779 ) 

780 # /!\ can't use mail.state here, as mail.refresh() will cause an error 

781 # see revid:odo@openerp.com-20120622152536-42b2s28lvdv3odyr in 6.1 

782 mail._postprocess_sent_message(success_pids=success_pids, failure_type=failure_type) 

783 except MemoryError: 

784 # prevent catching transient MemoryErrors, bubble up to notify user or abort cron job 

785 # instead of marking the mail as failed 

786 _logger.exception( 

787 'MemoryError while processing mail with ID %r and Msg-Id %r. Consider raising the --limit-memory-hard startup option', 

788 mail.id, mail.message_id) 

789 # mail status will stay on ongoing since transaction will be rollback 

790 raise 

791 except (psycopg2.Error, smtplib.SMTPServerDisconnected): 

792 # If an error with the database or SMTP session occurs, chances are that the cursor 

793 # or SMTP session are unusable, causing further errors when trying to save the state. 

794 _logger.exception( 

795 'Exception while processing mail with ID %r and Msg-Id %r.', 

796 mail.id, mail.message_id) 

797 raise 

798 except Exception as e: 

799 if isinstance(e, AssertionError): 

800 # Handle assert raised in IrMailServer to try to catch notably from-specific errors. 

801 # Note that assert may raise several args, a generic error string then a specific 

802 # message for logging in failure type 

803 error_code = e.args[0] 

804 if len(e.args) > 1 and error_code == IrMailServer.NO_VALID_FROM: 

805 # log failing email in additional arguments message 

806 failure_reason = str(e.args[1]) 

807 else: 

808 failure_reason = error_code 

809 if error_code == IrMailServer.NO_VALID_FROM: 

810 failure_type = "mail_from_invalid" 

811 elif error_code in (IrMailServer.NO_FOUND_FROM, IrMailServer.NO_FOUND_SMTP_FROM): 

812 failure_type = "mail_from_missing" 

813 # generic (unknown) error as fallback 

814 if not failure_reason: 

815 failure_reason = tools.exception_to_unicode(e) 

816 if not failure_type: 

817 failure_type = "unknown" 

818 

819 _logger.exception('failed sending mail (id: %s) due to %s', mail.id, failure_reason) 

820 mail.write({ 

821 "failure_reason": failure_reason, 

822 "failure_type": failure_type, 

823 "state": "exception", 

824 }) 

825 mail._postprocess_sent_message( 

826 success_pids=success_pids, 

827 failure_reason=failure_reason, failure_type=failure_type 

828 ) 

829 if raise_exception: 

830 if isinstance(e, (AssertionError, UnicodeEncodeError)): 

831 if isinstance(e, UnicodeEncodeError): 

832 value = "Invalid text: %s" % e.object 

833 else: 

834 value = '. '.join(e.args) 

835 raise MailDeliveryException(value) 

836 raise 

837 if auto_commit is True: 

838 if post_send_callback: 

839 post_send_callback([mail_id]) 

840 self._cr.commit() 

841 if post_send_callback: 

842 post_send_callback(self.ids) 

843 return True
