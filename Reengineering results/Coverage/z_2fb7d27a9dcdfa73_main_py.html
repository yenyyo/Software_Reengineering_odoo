30 @http.route(_return_url, type='http', methods=['GET'], auth='public') 

31 def stripe_return(self, **data): 

32 """ Process the notification data sent by Stripe after redirection from payment. 

33 

34 Customers go through this route regardless of whether the payment was direct or with 

35 redirection to Stripe or to an external service (e.g., for strong authentication). 

36 

37 :param dict data: The notification data, including the reference appended to the URL in 

38 `_get_specific_processing_values`. 

39 """ 

40 # Retrieve the transaction based on the reference included in the return url. 

41 tx_sudo = request.env['payment.transaction'].sudo()._get_tx_from_notification_data( 

42 'stripe', data 

43 ) 

44 

45 if tx_sudo.operation != 'validation': 

46 # Fetch the PaymentIntent and PaymentMethod objects from Stripe. 

47 payment_intent = tx_sudo.provider_id._stripe_make_request( 

48 f'payment_intents/{data.get("payment_intent")}', 

49 payload={'expand[]': 'payment_method'}, # Expand all required objects. 

50 method='GET', 

51 ) 

52 _logger.info("Received payment_intents response:\n%s", pprint.pformat(payment_intent)) 

53 self._include_payment_intent_in_notification_data(payment_intent, data) 

54 else: 

55 # Fetch the SetupIntent and PaymentMethod objects from Stripe. 

56 setup_intent = tx_sudo.provider_id._stripe_make_request( 

57 f'setup_intents/{data.get("setup_intent")}', 

58 payload={'expand[]': 'payment_method'}, # Expand all required objects. 

59 method='GET', 

60 ) 

61 _logger.info("Received setup_intents response:\n%s", pprint.pformat(setup_intent)) 

62 self._include_setup_intent_in_notification_data(setup_intent, data) 

63 

64 # Handle the notification data crafted with Stripe API's objects. 

65 tx_sudo._handle_notification_data('stripe', data) 

66 

67 # Redirect the user to the status page. 

68 return request.redirect('/payment/status')

70 @http.route(_webhook_url, type='http', methods=['POST'], auth='public', csrf=False) 

71 def stripe_webhook(self): 

72 """ Process the notification data sent by Stripe to the webhook. 

73 

74 :return: An empty string to acknowledge the notification. 

75 :rtype: str 

76 """ 

77 event = request.get_json_data() 

78 _logger.info("Notification received from Stripe with data:\n%s", pprint.pformat(event)) 

79 try: 

80 if event['type'] in HANDLED_WEBHOOK_EVENTS: 

81 stripe_object = event['data']['object'] # {Payment,Setup}Intent, Charge, or Refund. 

82 

83 # Check the integrity of the event. 

84 data = { 

85 'reference': stripe_object.get('description'), 

86 'event_type': event['type'], 

87 'object_id': stripe_object['id'], 

88 } 

89 tx_sudo = request.env['payment.transaction'].sudo()._get_tx_from_notification_data( 

90 'stripe', data 

91 ) 

92 self._verify_notification_signature(tx_sudo) 

93 

94 # Handle the notification data. 

95 if event['type'].startswith('payment_intent'): # Payment operation. 

96 if tx_sudo.tokenize: 

97 payment_method = tx_sudo.provider_id._stripe_make_request( 

98 f'payment_methods/{stripe_object["payment_method"]}', method='GET' 

99 ) 

100 _logger.info( 

101 "Received payment_methods response:\n%s", pprint.pformat(payment_method) 

102 ) 

103 stripe_object['payment_method'] = payment_method 

104 self._include_payment_intent_in_notification_data(stripe_object, data) 

105 elif event['type'].startswith('setup_intent'): # Validation operation. 

106 # Fetch the missing PaymentMethod object. 

107 payment_method = tx_sudo.provider_id._stripe_make_request( 

108 f'payment_methods/{stripe_object["payment_method"]}', method='GET' 

109 ) 

110 _logger.info( 

111 "Received payment_methods response:\n%s", pprint.pformat(payment_method) 

112 ) 

113 stripe_object['payment_method'] = payment_method 

114 self._include_setup_intent_in_notification_data(stripe_object, data) 

115 elif event['type'] == 'charge.refunded': # Refund operation (refund creation). 

116 refunds = stripe_object['refunds']['data'] 

117 

118 # The refunds linked to this charge are paginated, fetch the remaining refunds. 

119 has_more = stripe_object['refunds']['has_more'] 

120 while has_more: 

121 payload = { 

122 'charge': stripe_object['id'], 

123 'starting_after': refunds[-1]['id'], 

124 'limit': 100, 

125 } 

126 additional_refunds = tx_sudo.provider_id._stripe_make_request( 

127 'refunds', payload=payload, method='GET' 

128 ) 

129 refunds += additional_refunds['data'] 

130 has_more = additional_refunds['has_more'] 

131 

132 # Process the refunds for which a refund transaction has not been created yet. 

133 processed_refund_ids = tx_sudo.child_transaction_ids.filtered( 

134 lambda tx: tx.operation == 'refund' 

135 ).mapped('provider_reference') 

136 for refund in filter(lambda r: r['id'] not in processed_refund_ids, refunds): 

137 refund_tx_sudo = self._create_refund_tx_from_refund(tx_sudo, refund) 

138 self._include_refund_in_notification_data(refund, data) 

139 refund_tx_sudo._handle_notification_data('stripe', data) 

140 # Don't handle the notification data for the source transaction. 

141 return request.make_json_response('') 

142 elif event['type'] == 'charge.refund.updated': # Refund operation (with update). 

143 # A refund was updated by Stripe after it was already processed (possibly to 

144 # cancel it). This can happen when the customer's payment method can no longer 

145 # be topped up (card expired, account closed...). The `tx_sudo` record is the 

146 # refund transaction to update. 

147 self._include_refund_in_notification_data(stripe_object, data) 

148 

149 # Handle the notification data crafted with Stripe API objects 

150 tx_sudo._handle_notification_data('stripe', data) 

151 except ValidationError: # Acknowledge the notification to avoid getting spammed 

152 _logger.exception("unable to handle the notification data; skipping to acknowledge") 

153 return request.make_json_response('')

173 @staticmethod 

174 def _create_refund_tx_from_refund(source_tx_sudo, refund_object): 

175 """ Create a refund transaction based on Stripe data. 

176 

177 :param recordset source_tx_sudo: The source transaction for which a refund is initiated, as 

178 a sudoed `payment.transaction` record. 

179 :param dict refund_object: The Stripe refund object to create the refund from. 

180 :return: The created refund transaction. 

181 :rtype: recordset of `payment.transaction` 

182 """ 

183 amount_to_refund = refund_object['amount'] 

184 converted_amount = payment_utils.to_major_currency_units( 

185 amount_to_refund, source_tx_sudo.currency_id 

186 ) 

187 return source_tx_sudo._create_child_transaction(converted_amount, is_refund=True)

189 def _verify_notification_signature(self, tx_sudo): 

190 """ Check that the received signature matches the expected one. 

191 

192 See https://stripe.com/docs/webhooks/signatures#verify-manually. 

193 

194 :param recordset tx_sudo: The sudoed transaction referenced by the notification data, as a 

195 `payment.transaction` record 

196 :return: None 

197 :raise: :class:`werkzeug.exceptions.Forbidden` if the timestamp is too old or if the 

198 signatures don't match 

199 """ 

200 webhook_secret = stripe_utils.get_webhook_secret(tx_sudo.provider_id) 

201 if not webhook_secret: 

202 _logger.warning("ignored webhook event due to undefined webhook secret") 

203 return 

204 

205 notification_payload = request.httprequest.data.decode('utf-8') 

206 signature_entries = request.httprequest.headers['Stripe-Signature'].split(',') 

207 signature_data = {k: v for k, v in [entry.split('=') for entry in signature_entries]} 

208 

209 # Retrieve the timestamp from the data 

210 event_timestamp = int(signature_data.get('t', '0')) 

211 if not event_timestamp: 

212 _logger.warning("received notification with missing timestamp") 

213 raise Forbidden() 

214 

215 # Check if the timestamp is not too old 

216 if datetime.utcnow().timestamp() - event_timestamp > self.WEBHOOK_AGE_TOLERANCE: 

217 _logger.warning("received notification with outdated timestamp: %s", event_timestamp) 

218 raise Forbidden() 

219 

220 # Retrieve the received signature from the data 

221 received_signature = signature_data.get('v1') 

222 if not received_signature: 

223 _logger.warning("received notification with missing signature") 

224 raise Forbidden() 

225 

226 # Compare the received signature with the expected signature computed from the data 

227 signed_payload = f'{event_timestamp}.{notification_payload}' 

228 expected_signature = hmac.new( 

229 webhook_secret.encode('utf-8'), signed_payload.encode('utf-8'), hashlib.sha256 

230 ).hexdigest() 

231 if not hmac.compare_digest(received_signature, expected_signature): 

232 _logger.warning("received notification with invalid signature") 

233 raise Forbidden()
