26def nl2br(string: str) -> Markup: 

27 """ Converts newlines to HTML linebreaks in ``string`` after HTML-escaping 

28 it. 

29 """ 

30 return escape_silent(string).replace('\n', Markup('<br>\n'))

33def nl2br_enclose(string: str, enclosure_tag: str = 'div') -> Markup: 

34 """ Like nl2br, but returns enclosed Markup allowing to better manipulate 

35 trusted and untrusted content. New lines added by use are trusted, other 

36 content is escaped. """ 

37 return Markup('<{enclosure_tag}>{converted}</{enclosure_tag}>').format( 

38 enclosure_tag=enclosure_tag, 

39 converted=nl2br(string),

78 def attributes(self, record, field_name, options, values=None): 

79 """ attributes(record, field_name, field, options, values) 

80 

81 Generates the metadata attributes (prefixed by ``data-oe-``) for the 

82 root node of the field conversion. 

83 

84 The default attributes are:

101 return data

104 data['data-oe-id'] = record.id 

105 data['data-oe-field'] = field.name 

106 data['data-oe-type'] = options.get('type') 

107 data['data-oe-expression'] = options.get('expression')

366 @api.model 

367 def value_to_html(self, value, options): 

368 irQweb = self.env['ir.qweb'] 

369 # wrap value inside a body and parse it as HTML 

370 body = etree.fromstring("<body>%s</body>" % value, etree.HTMLParser(encoding='utf-8'))[0] 

371 # use pos processing for all nodes with attributes 

372 for element in body.iter(): 

373 if element.attrib: 

374 attrib = dict(element.attrib) 

375 attrib = irQweb._post_processing_att(element.tag, attrib) 

376 element.attrib.clear() 

377 element.attrib.update(attrib) 

378 return Markup(etree.tostring(body, encoding='unicode', method='html')[6:-7])

430class MonetaryConverter(models.AbstractModel): 

431 """ ``monetary`` converter, has a mandatory option

435

453 display_currency=dict(type='model', params='res.currency', string=_('Display currency'), required="value_to_html"), 

454 date=dict(type='date', string=_('Date'), description=_('Date used for the original currency (only used for t-esc). by default use the current date.')),

459 @api.model 

460 def value_to_html(self, value, options): 

461 display_currency = options['display_currency'] 

462 

463 if not isinstance(value, (int, float)): 

464 raise ValueError(_("The value send to monetary field is not a number.")) 

465 

466 # lang.format mandates a sprintf-style format. These formats are non- 

467 # minimal (they have a default fixed precision instead), and 

468 # lang.format will not set one by default. currency.round will not 

469 # provide one either. So we need to generate a precision value 

470 # (integer > 0) from the currency's rounding (a float generally < 1.0). 

471 fmt = "%.{0}f".format(options.get('decimal_places', display_currency.decimal_places)) 

472 

473 if options.get('from_currency'): 

474 date = options.get('date') or fields.Date.today() 

475 company_id = options.get('company_id') 

476 if company_id: 

477 company = self.env['res.company'].browse(company_id) 

478 else: 

479 company = self.env.company 

480 value = options['from_currency']._convert(value, display_currency, company, date) 

481 

482 lang = self.user_lang() 

483 formatted_amount = lang.format(fmt, display_currency.round(value), grouping=True)\ 

484 .replace(r' ', '\N{NO-BREAK SPACE}').replace(r'-', '-\N{ZERO WIDTH NO-BREAK SPACE}') 

485 

486 pre = post = '' 

487 if display_currency.position == 'before': 

488 pre = '{symbol}\N{NO-BREAK SPACE}'.format(symbol=display_currency.symbol or '') 

489 else: 

490 post = '\N{NO-BREAK SPACE}{symbol}'.format(symbol=display_currency.symbol or '') 

491 

492 if options.get('label_price') and lang.decimal_point in formatted_amount: 

493 sep = lang.decimal_point 

494 integer_part, decimal_part = formatted_amount.split(sep) 

495 integer_part += sep 

496 return Markup('{pre}<span class="oe_currency_value">{0}</span><span class="oe_currency_value" style="font-size:0.5em">{1}</span>{post}').format(integer_part, decimal_part, pre=pre, post=post) 

497 

498 return Markup('{pre}<span class="oe_currency_value">{0}</span>{post}').format(formatted_amount, pre=pre, post=post)

610 default_value='long' 

611 ), 

612 add_direction=dict( 

613 type="boolean", 

614 string=_("Add direction"), 

615 description=_("Add directional information (not used for digital)") 

616 ), 

617 )

620 @api.model 

621 def value_to_html(self, value, options): 

622 units = {unit: duration for unit, label, duration in TIMEDELTA_UNITS} 

623 

624 locale = babel_locale_parse(self.user_lang().code) 

625 factor = units[options.get('unit', 'second')] 

626 round_to = units[options.get('round', 'second')] 

627 

628 if options.get('digital') and round_to > 3600: 

629 round_to = 3600 

630 

631 r = round((value * factor) / round_to) * round_to 

632 

633 sections = [] 

634 sign = '' 

635 if value < 0: 

636 r = -r 

637 sign = '-' 

638 

639 if options.get('digital'): 

640 for unit, label, secs_per_unit in TIMEDELTA_UNITS: 

641 if secs_per_unit > 3600: 

642 continue 

643 v, r = divmod(r, secs_per_unit) 

644 if not v and (secs_per_unit > factor or secs_per_unit < round_to): 

645 continue 

646 sections.append(u"%02.0f" % int(round(v))) 

647 return sign + u':'.join(sections) 

648 

649 for unit, label, secs_per_unit in TIMEDELTA_UNITS: 

650 v, r = divmod(r, secs_per_unit) 

651 if not v: 

652 continue 

653 try: 

654 section = babel.dates.format_timedelta( 

655 v*secs_per_unit, 

656 granularity=round_to, 

657 add_direction=options.get('add_direction'), 

658 format=options.get('format', 'long'), 

659 threshold=1, 

660 locale=locale) 

661 except KeyError: 

662 # in case of wrong implementation of babel, try to fallback on en_US locale. 

663 # https://github.com/python-babel/babel/pull/827/files 

664 # Some bugs already fixed in 2.10 but ubuntu22 is 2.8 

665 localeUS = babel_locale_parse('en_US') 

666 section = babel.dates.format_timedelta( 

667 v*secs_per_unit, 

668 granularity=round_to, 

669 add_direction=options.get('add_direction'), 

670 format=options.get('format', 'long'), 

671 threshold=1, 

672 locale=localeUS) 

673 if section: 

674 sections.append(section) 

675 

676 if sign: 

677 sections.insert(0, sign) 

678 return u' '.join(sections)
