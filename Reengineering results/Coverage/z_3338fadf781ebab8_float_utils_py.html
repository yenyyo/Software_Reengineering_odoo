16def round(f): 

17 # P3's builtin round differs from P2 in the following manner: 

18 # * it rounds half to even rather than up (away from 0) 

19 # * round(-0.) loses the sign (it returns -0 rather than 0) 

20 # * round(x) returns an int rather than a float 

21 # 

22 # this compatibility shim implements Python 2's round in terms of 

23 # Python 3's so that important rounding error under P3 can be 

24 # trivially fixed, assuming the P2 behaviour to be debugged and 

25 # correct. 

26 roundf = builtins.round(f) 

27 if builtins.round(f + 1) - roundf != 1: 

28 return f + math.copysign(0.5, f) 

29 # copysign ensures round(-0.) -> -0 *and* result is a float 

30 return math.copysign(roundf, f)

33def _float_check_precision(precision_digits=None, precision_rounding=None): 

34 if precision_rounding is not None and precision_digits is None: 

35 assert precision_rounding > 0,\ 

36 f"precision_rounding must be positive, got {precision_rounding}" 

37 elif precision_digits is not None and precision_rounding is None: 

38 # TODO: `int`s will also get the `is_integer` method starting from python 3.12 

39 assert float(precision_digits).is_integer() and precision_digits >= 0,\ 

40 f"precision_digits must be a non-negative integer, got {precision_digits}" 

41 precision_rounding = 10 ** -precision_digits 

42 else: 

43 msg = "exactly one of precision_digits and precision_rounding must be specified" 

44 raise AssertionError(msg) 

45 return precision_rounding

48def float_round(value, precision_digits=None, precision_rounding=None, rounding_method='HALF-UP'): 

49 """Return ``value`` rounded to ``precision_digits`` decimal digits, 

50 minimizing IEEE-754 floating point representation errors, and applying 

51 the tie-breaking rule selected with ``rounding_method``, by default 

52 HALF-UP (away from zero). 

53 Precision must be given by ``precision_digits`` or ``precision_rounding``, 

54 not both! 

55 

56 :param float value: the value to round 

57 :param int precision_digits: number of fractional digits to round to. 

58 :param float precision_rounding: decimal number representing the minimum 

59 non-zero value at the desired precision (for example, 0.01 for a 

60 2-digit precision). 

61 :param rounding_method: the rounding method used: 

62 - 'HALF-UP' will round to the closest number with ties going away from zero. 

63 - 'HALF-DOWN' will round to the closest number with ties going towards zero. 

64 - 'HALF_EVEN' will round to the closest number with ties going to the closest 

65 even number. 

66 - 'UP' will always round away from 0. 

67 - 'DOWN' will always round towards 0. 

68 :return: rounded float 

69 """ 

70 rounding_factor = _float_check_precision(precision_digits=precision_digits, 

71 precision_rounding=precision_rounding) 

72 if rounding_factor == 0 or value == 0: 

73 return 0.0 

74 

75 # NORMALIZE - ROUND - DENORMALIZE 

76 # In order to easily support rounding to arbitrary 'steps' (e.g. coin values), 

77 # we normalize the value before rounding it as an integer, and de-normalize 

78 # after rounding: e.g. float_round(1.3, precision_rounding=.5) == 1.5 

79 def normalize(val): 

80 return val / rounding_factor 

81 

82 def denormalize(val): 

83 return val * rounding_factor 

84 

85 # inverting small rounding factors reduces rounding errors 

86 if rounding_factor < 1: 

87 rounding_factor = float_invert(rounding_factor) 

88 normalize, denormalize = denormalize, normalize 

89 

90 normalized_value = normalize(value) 

91 

92 # Due to IEEE-754 float/double representation limits, the approximation of the 

93 # real value may be slightly below the tie limit, resulting in an error of 

94 # 1 unit in the last place (ulp) after rounding. 

95 # For example 2.675 == 2.6749999999999998. 

96 # To correct this, we add a very small epsilon value, scaled to the 

97 # the order of magnitude of the value, to tip the tie-break in the right 

98 # direction. 

99 # Credit: discussion with OpenERP community members on bug 882036 

100 epsilon_magnitude = math.log2(abs(normalized_value)) 

101 # `2**(epsilon_magnitude - 52)` would be the minimal size, but we increase it to be 

102 # more tolerant of inaccuracies accumulated after multiple floating point operations 

103 epsilon = 2**(epsilon_magnitude - 50) 

104 

105 match rounding_method: 

106 case 'HALF-UP': # 0.5 rounds away from 0 

107 result = round(normalized_value + math.copysign(epsilon, normalized_value)) 

108 case 'HALF-EVEN': # 0.5 rounds towards closest even number 

109 integral = math.floor(normalized_value) 

110 remainder = abs(normalized_value - integral) 

111 is_half = abs(0.5 - remainder) < epsilon 

112 # if is_half & integral is odd, add odd bit to make it even 

113 result = integral + (integral & 1) if is_half else round(normalized_value) 

114 case 'HALF-DOWN': # 0.5 rounds towards 0 

115 result = round(normalized_value - math.copysign(epsilon, normalized_value)) 

116 case 'UP': # round to number furthest from zero 

117 result = math.trunc(normalized_value + math.copysign(1 - epsilon, normalized_value)) 

118 case 'DOWN': # round to number closest to zero 

119 result = math.trunc(normalized_value + math.copysign(epsilon, normalized_value)) 

120 case _: 

121 msg = f"unknown rounding method: {rounding_method}" 

122 raise ValueError(msg) 

123 

124 return denormalize(result)

127def float_is_zero(value, precision_digits=None, precision_rounding=None): 

128 """Returns true if ``value`` is small enough to be treated as 

129 zero at the given precision (smaller than the corresponding *epsilon*). 

130 The precision (``10**-precision_digits`` or ``precision_rounding``) 

131 is used as the zero *epsilon*: values less than that are considered 

132 to be zero. 

133 Precision must be given by ``precision_digits`` or ``precision_rounding``, 

134 not both! 

135 

136 Warning: ``float_is_zero(value1-value2)`` is not equivalent to 

137 ``float_compare(value1,value2) == 0``, as the former will round after 

138 computing the difference, while the latter will round before, giving 

139 different results for e.g. 0.006 and 0.002 at 2 digits precision. 

140 

141 :param int precision_digits: number of fractional digits to round to. 

142 :param float precision_rounding: decimal number representing the minimum 

143 non-zero value at the desired precision (for example, 0.01 for a 

144 2-digit precision). 

145 :param float value: value to compare with the precision's zero 

146 :return: True if ``value`` is considered zero 

147 """ 

148 epsilon = _float_check_precision(precision_digits=precision_digits, 

149 precision_rounding=precision_rounding) 

150 return value == 0.0 or abs(float_round(value, precision_rounding=epsilon)) < epsilon

153def float_compare(value1, value2, precision_digits=None, precision_rounding=None): 

154 """Compare ``value1`` and ``value2`` after rounding them according to the 

155 given precision. A value is considered lower/greater than another value 

156 if their rounded value is different. This is not the same as having a 

157 non-zero difference! 

158 Precision must be given by ``precision_digits`` or ``precision_rounding``, 

159 not both! 

160 

161 Example: 1.432 and 1.431 are equal at 2 digits precision, 

162 so this method would return 0 

163 However 0.006 and 0.002 are considered different (this method returns 1) 

164 because they respectively round to 0.01 and 0.0, even though 

165 0.006-0.002 = 0.004 which would be considered zero at 2 digits precision. 

166 

167 Warning: ``float_is_zero(value1-value2)`` is not equivalent to 

168 ``float_compare(value1,value2) == 0``, as the former will round after 

169 computing the difference, while the latter will round before, giving 

170 different results for e.g. 0.006 and 0.002 at 2 digits precision. 

171 

172 :param float value1: first value to compare 

173 :param float value2: second value to compare 

174 :param int precision_digits: number of fractional digits to round to. 

175 :param float precision_rounding: decimal number representing the minimum 

176 non-zero value at the desired precision (for example, 0.01 for a 

177 2-digit precision). 

178 :return: (resp.) -1, 0 or 1, if ``value1`` is (resp.) lower than, 

179 equal to, or greater than ``value2``, at the given precision. 

180 """ 

181 rounding_factor = _float_check_precision(precision_digits=precision_digits, 

182 precision_rounding=precision_rounding) 

183 # equal numbers round equally, so we can skip that step 

184 # doing this after _float_check_precision to validate parameters first 

185 if value1 == value2: 

186 return 0 

187 value1 = float_round(value1, precision_rounding=rounding_factor) 

188 value2 = float_round(value2, precision_rounding=rounding_factor) 

189 delta = value1 - value2 

190 if float_is_zero(delta, precision_rounding=rounding_factor): 

191 return 0 

192 return -1 if delta < 0.0 else 1

195def float_repr(value, precision_digits): 

196 """Returns a string representation of a float with the 

197 given number of fractional digits. This should not be 

198 used to perform a rounding operation (this is done via 

199 :func:`~.float_round`), but only to produce a suitable 

200 string representation for a float. 

201 

202 :param float value: 

203 :param int precision_digits: number of fractional digits to include in the output 

204 """ 

205 # Can't use str() here because it seems to have an intrinsic 

206 # rounding to 12 significant digits, which causes a loss of 

207 # precision. e.g. str(123456789.1234) == str(123456789.123)!! 

208 return "%.*f" % (precision_digits, value)

211def float_split_str(value, precision_digits): 

212 """Splits the given float 'value' in its unitary and decimal parts, 

213 returning each of them as a string, rounding the value using 

214 the provided ``precision_digits`` argument. 

215 

216 The length of the string returned for decimal places will always 

217 be equal to ``precision_digits``, adding zeros at the end if needed. 

218 

219 In case ``precision_digits`` is zero, an empty string is returned for 

220 the decimal places. 

221 

222 Examples: 

223 1.432 with precision 2 => ('1', '43') 

224 1.49 with precision 1 => ('1', '5') 

225 1.1 with precision 3 => ('1', '100') 

226 1.12 with precision 0 => ('1', '') 

227 

228 :param float value: value to split. 

229 :param int precision_digits: number of fractional digits to round to. 

230 :return: returns the tuple(<unitary part>, <decimal part>) of the given value 

231 :rtype: tuple(str, str) 

232 """ 

233 value = float_round(value, precision_digits=precision_digits) 

234 value_repr = float_repr(value, precision_digits) 

235 return tuple(value_repr.split('.')) if precision_digits else (value_repr, '')

238def float_split(value, precision_digits): 

239 """ same as float_split_str() except that it returns the unitary and decimal 

240 parts as integers instead of strings. In case ``precision_digits`` is zero, 

241 0 is always returned as decimal part. 

242 

243 :rtype: tuple(int, int) 

244 """ 

245 units, cents = float_split_str(value, precision_digits) 

246 if not cents: 

247 return int(units), 0 

248 return int(units), int(cents)

251def json_float_round(value, precision_digits, rounding_method='HALF-UP'): 

252 """Not suitable for float calculations! Similar to float_repr except that it 

253 returns a float suitable for json dump 

254 

255 This may be necessary to produce "exact" representations of rounded float 

256 values during serialization, such as what is done by `json.dumps()`. 

257 Unfortunately `json.dumps` does not allow any form of custom float representation, 

258 nor any custom types, everything is serialized from the basic JSON types. 

259 

260 :param int precision_digits: number of fractional digits to round to. 

261 :param rounding_method: the rounding method used: 'HALF-UP', 'UP' or 'DOWN', 

262 the first one rounding up to the closest number with the rule that 

263 number>=0.5 is rounded up to 1, the second always rounding up and the 

264 latest one always rounding down. 

265 :return: a rounded float value that must not be used for calculations, but 

266 is ready to be serialized in JSON with minimal chances of 

267 representation errors. 

268 """ 

269 rounded_value = float_round(value, precision_digits=precision_digits, rounding_method=rounding_method) 

270 rounded_repr = float_repr(rounded_value, precision_digits=precision_digits) 

271 # As of Python 3.1, rounded_repr should be the shortest representation for our 

272 # rounded float, so we create a new float whose repr is expected 

273 # to be the same value, or a value that is semantically identical 

274 # and will be used in the json serialization. 

275 # e.g. if rounded_repr is '3.1750', the new float repr could be 3.175 

276 # but not 3.174999999999322452. 

277 # Cfr. bpo-1580: https://bugs.python.org/issue1580 

278 return float(rounded_repr)

291def float_invert(value): 

292 """Inverts a floating point number with increased accuracy. 

293 

294 :param float value: value to invert. 

295 :param bool store: whether store the result in memory for future calls. 

296 :return: rounded float. 

297 """ 

298 result = _INVERTDICT.get(value) 

299 if result is None: 

300 coefficient, exponent = f'{value:.15e}'.split('e') 

301 # invert exponent by changing sign, and coefficient by dividing by its square 

302 result = float(f'{coefficient}e{-int(exponent)}') / float(coefficient)**2 

303 return result

312 def try_round(amount, expected, precision_digits=3): 

313 result = float_repr(float_round(amount, precision_digits=precision_digits), 

314 precision_digits=precision_digits) 

315 if result != expected: 

316 print('###!!! Rounding error: got %s , expected %s' % (result, expected)) 

317 return complex(1, 1)

320 # Extended float range test, inspired by Cloves Almeida's test on bug #882036. 

321 fractions = [.0, .015, .01499, .675, .67499, .4555, .4555, .45555] 

322 expecteds = ['.00', '.02', '.01', '.68', '.67', '.46', '.456', '.4556'] 

323 precisions = [2, 2, 2, 2, 2, 2, 3, 4] 

324 for magnitude in range(7): 

325 for frac, exp, prec in zip(fractions, expecteds, precisions): 

326 for sign in [-1, 1]: 

327 for x in range(0, 10000, 97): 

328 n = x * 10**magnitude 

329 f = sign * (n + frac) 

330 f_exp = ('-' if f != 0 and sign == -1 else '') + str(n) + exp 

331 count += try_round(f, f_exp, precision_digits=prec) 

332 

333 stop = time.time() 

334 count, errors = int(count.real), int(count.imag)
