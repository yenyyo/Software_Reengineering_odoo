23 def upload_document(self, ufile, sale_order_template_id=False): 

24 # TODO: add `allowed_company_ids` as method param in master 

25 if allowed_company_ids := request.params.get('allowed_company_ids'): 

26 request.update_context(allowed_company_ids=json.loads(allowed_company_ids)) 

27 sale_order_template = request.env['sale.order.template'].browse( 

28 int(sale_order_template_id) 

29 ) 

30 company = sale_order_template.company_id if sale_order_template else request.env.company 

31 files = request.httprequest.files.getlist('ufile') 

32 result = {'success': _("All files uploaded")} 

33 for ufile in files: 

34 try: 

35 mimetype = ufile.content_type 

36 doc = request.env['quotation.document'].create({ 

37 'name': ufile.filename, 

38 'mimetype': mimetype, 

39 'raw': ufile.read(), 

40 'quotation_template_ids': sale_order_template.ids, 

41 'company_id': company.id, 

42 }) 

43 # pypdf will also catch malformed document 

44 utils._ensure_document_not_encrypted(base64.b64decode(doc.datas)) 

45 except Exception as e: 

46 logger.exception("Failed to upload document %s", ufile.filename) 

47 result = {'error': str(e)} 

48 

49 return json.dumps(result)
