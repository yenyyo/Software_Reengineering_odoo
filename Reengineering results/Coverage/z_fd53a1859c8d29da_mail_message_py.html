266 @api.model 

267 def _search(self, domain, offset=0, limit=None, order=None): 

268 """ Override that adds specific access rights of mail.message, to remove 

269 ids uid could not see according to our custom rules. Please refer to 

270 _check_access() for more details about those rules. 

271 

272 Non employees users see only message with subtype (aka do not see 

273 internal logs). 

274 

275 After having received ids of a classic search, keep only: 

276 - if author_id == pid, uid is the author, OR 

277 - uid belongs to a notified channel, OR 

278 - uid is in the specified recipients, OR 

279 - uid has a notification on the message 

280 - otherwise: remove the id 

281 """ 

282 # Rules do not apply to administrator 

283 if self.env.is_superuser(): 

284 return super()._search(domain, offset, limit, order) 

285 

286 # Non-employee see only messages with a subtype and not internal 

287 if not self.env.user._is_internal(): 

288 domain = self._get_search_domain_share() + domain 

289 

290 # make the search query with the default rules 

291 query = super()._search(domain, offset, limit, order) 

292 

293 # retrieve matching records and determine which ones are truly accessible 

294 self.flush_model(['model', 'res_id', 'author_id', 'message_type', 'partner_ids']) 

295 self.env['mail.notification'].flush_model(['mail_message_id', 'res_partner_id']) 

296 

297 pid = self.env.user.partner_id.id 

298 ids = [] 

299 allowed_ids = set() 

300 model_ids = defaultdict(lambda: defaultdict(set)) 

301 

302 rel_alias = query.make_alias(self._table, 'partner_ids') 

303 query.add_join("LEFT JOIN", rel_alias, 'mail_message_res_partner_rel', SQL( 

304 "%s = %s AND %s = %s", 

305 SQL.identifier(self._table, 'id'), 

306 SQL.identifier(rel_alias, 'mail_message_id'), 

307 SQL.identifier(rel_alias, 'res_partner_id'), 

308 pid, 

309 )) 

310 notif_alias = query.make_alias(self._table, 'notification_ids') 

311 query.add_join("LEFT JOIN", notif_alias, 'mail_notification', SQL( 

312 "%s = %s AND %s = %s", 

313 SQL.identifier(self._table, 'id'), 

314 SQL.identifier(notif_alias, 'mail_message_id'), 

315 SQL.identifier(notif_alias, 'res_partner_id'), 

316 pid, 

317 )) 

318 self.env.cr.execute(query.select( 

319 SQL.identifier(self._table, 'id'), 

320 SQL.identifier(self._table, 'model'), 

321 SQL.identifier(self._table, 'res_id'), 

322 SQL.identifier(self._table, 'author_id'), 

323 SQL.identifier(self._table, 'message_type'), 

324 SQL( 

325 "COALESCE(%s, %s)", 

326 SQL.identifier(rel_alias, 'res_partner_id'), 

327 SQL.identifier(notif_alias, 'res_partner_id'), 

328 ), 

329 )) 

330 for id_, model, res_id, author_id, message_type, partner_id in self.env.cr.fetchall(): 

331 ids.append(id_) 

332 if author_id == pid: 

333 allowed_ids.add(id_) 

334 elif partner_id == pid: 

335 allowed_ids.add(id_) 

336 elif model and res_id and message_type != 'user_notification': 

337 model_ids[model][res_id].add(id_) 

338 

339 allowed_ids.update(self._find_allowed_doc_ids(model_ids)) 

340 allowed = self.browse(id_ for id_ in ids if id_ in allowed_ids) 

341 return allowed._as_query(order)

359 def _check_access(self, operation: str) -> tuple | None: 

360 """ Access rules of mail.message: 

361 - read: if 

362 - author_id == pid, uid is the author OR 

363 - create_uid == uid, uid is the creator OR 

364 - uid is in the recipients (partner_ids) OR 

365 - uid has been notified (needaction) OR 

366 - uid have read access to the related document if model, res_id 

367 - otherwise: raise 

368 - create: if 

369 - no model, no res_id (private message) OR 

370 - pid in message_follower_ids if model, res_id OR 

371 - uid can read the parent OR 

372 - uid have write or create access on the related document if model, res_id, OR 

373 - otherwise: raise 

374 - write: if 

375 - author_id == pid, uid is the author, OR 

376 - uid is in the recipients (partner_ids) OR 

377 - uid has write or create access on the related document if model, res_id 

378 - otherwise: raise 

379 - unlink: if 

380 - uid has write or create access on the related document 

381 - otherwise: raise 

382 

383 Specific case: non employee users see only messages with subtype (aka do 

384 not see internal logs). 

385 """ 

386 result = super()._check_access(operation) 

387 if not self: 

388 return result 

389 

390 # discard forbidden records, and check remaining ones 

391 messages = self - result[0] if result else self 

392 if messages and (forbidden := messages._get_forbidden_access(operation)): 

393 if result: 

394 result = (result[0] + forbidden, result[1]) 

395 else: 

396 result = (forbidden, lambda: forbidden._make_access_error(operation)) 

397 return result

399 def _get_forbidden_access(self, operation: str) -> api.Self: 

400 """ Return the subset of ``self`` that does not satisfy the specific 

401 conditions for messages. 

402 """ 

403 forbidden = self.browse() 

404 

405 # Non employees see only messages with a subtype (aka, not internal logs) 

406 if not self.env.user._is_internal(): 

407 rows = self.env.execute_query(SQL( 

408 ''' SELECT message.id 

409 FROM "mail_message" AS message 

410 LEFT JOIN "mail_message_subtype" as subtype ON message.subtype_id = subtype.id 

411 WHERE message.id = ANY (%s) 

412 AND message.message_type = 'comment' 

413 AND (message.is_internal IS TRUE OR message.subtype_id IS NULL OR subtype.internal IS TRUE) 

414 ''', 

415 self.ids, 

416 )) 

417 if rows: 

418 internal = self.browse(id_ for [id_] in rows) 

419 forbidden += internal 

420 self -= internal # noqa: PLW0642 

421 if not self: 

422 return forbidden 

423 

424 # Read the value of messages in order to determine their accessibility. 

425 # The values are put in 'messages_to_check', and entries are popped 

426 # once we know they are accessible. At the end, the remaining entries 

427 # are the invalid ones. 

428 self.flush_recordset(['model', 'res_id', 'author_id', 'create_uid', 'parent_id', 'message_type', 'partner_ids']) 

429 self.env['mail.notification'].flush_model(['mail_message_id', 'res_partner_id']) 

430 

431 if operation in ('read', 'write'): 

432 query = SQL( 

433 """ SELECT m.id, m.model, m.res_id, m.author_id, m.create_uid, m.parent_id, 

434 bool_or(partner_rel.res_partner_id IS NOT NULL OR needaction_rel.res_partner_id IS NOT NULL) AS notified, 

435 m.message_type 

436 FROM "mail_message" m 

437 LEFT JOIN "mail_message_res_partner_rel" partner_rel 

438 ON partner_rel.mail_message_id = m.id AND partner_rel.res_partner_id = %(pid)s 

439 LEFT JOIN "mail_notification" needaction_rel 

440 ON needaction_rel.mail_message_id = m.id AND needaction_rel.res_partner_id = %(pid)s 

441 WHERE m.id = ANY(%(ids)s) 

442 GROUP BY m.id 

443 """, 

444 pid=self.env.user.partner_id.id, ids=self.ids, 

445 ) 

446 elif operation in ('create', 'unlink'): 

447 query = SQL( 

448 """ SELECT id, model, res_id, author_id, parent_id, message_type 

449 FROM "mail_message" 

450 WHERE id = ANY(%s) 

451 """, self.ids, 

452 ) 

453 else: 

454 raise ValueError(_('Wrong operation name (%s)', operation)) 

455 

456 # trick: messages_to_check doesn't contain missing records from messages 

457 messages_to_check = { 

458 values['id']: values 

459 for values in self.env.execute_query_dict(query) 

460 } 

461 

462 # Author condition (READ, WRITE, CREATE (private)) 

463 partner_id = self.env.user.partner_id.id 

464 if operation == 'read': 

465 for mid, message in list(messages_to_check.items()): 

466 if (message.get('author_id') == partner_id 

467 or message.get('create_uid') == self.env.uid): 

468 messages_to_check.pop(mid) 

469 elif operation == 'write': 

470 for mid, message in list(messages_to_check.items()): 

471 if message.get('author_id') == partner_id: 

472 messages_to_check.pop(mid) 

473 elif operation == 'create': 

474 for mid, message in list(messages_to_check.items()): 

475 if not self.is_thread_message(message): 

476 messages_to_check.pop(mid) 

477 

478 if not messages_to_check: 

479 return forbidden 

480 

481 # Recipients condition, for read and write (partner_ids) 

482 # keep on top, usefull for systray notifications 

483 if operation in ('read', 'write'): 

484 for mid, message in list(messages_to_check.items()): 

485 if message.get('notified'): 

486 messages_to_check.pop(mid) 

487 if not messages_to_check: 

488 return forbidden 

489 

490 # CRUD: Access rights related to the document 

491 # {document_model_name: {document_id: message_ids}} 

492 model_docid_msgids = defaultdict(lambda: defaultdict(list)) 

493 for mid, message in messages_to_check.items(): 

494 if (message.get('model') and message.get('res_id') and 

495 message.get('message_type') != 'user_notification'): 

496 model_docid_msgids[message['model']][message['res_id']].append(mid) 

497 

498 for model, docid_msgids in model_docid_msgids.items(): 

499 documents = self.env[model].browse(docid_msgids) 

500 if hasattr(documents, '_get_mail_message_access'): 

501 doc_operation = documents._get_mail_message_access(docid_msgids, operation) # why not giving model here? 

502 else:

512 return forbidden

532 

533 if not messages_to_check: 

534 return forbidden

536 # Recipients condition for create (message_follower_ids) 

537 for model, docid_msgids in model_docid_msgids.items(): 

538 domain = [ 

539 ('res_model', '=', model), 

540 ('res_id', 'in', list(docid_msgids)), 

541 ('partner_id', '=', self.env.user.partner_id.id), 

542 ]

546 messages_to_check.pop(mid) 

547 

548 if not messages_to_check: 

549 return forbidden

612 def base64_to_boundary(match):

614 if not data_to_url.get(key): 

615 name = match.group(4) if match.group(4) else 'image%s' % len(data_to_url)

622 }) 

623 except binascii_error: 

624 _logger.warning("Impossible to create an attachment out of badly formated base64 embedded image. Image has been removed.")

627 attachment.generate_access_token() 

628 values['attachment_ids'].append((4, attachment.id)) 

629 data_to_url[key] = ['/web/image/%s?access_token=%s' % (attachment.id, attachment.access_token), name]

719 def unlink(self): 

720 # cascade-delete attachments that are directly attached to the message (should only happen 

721 # for mail.messages that act as parent for a standalone mail.mail record). 

722 # the cache of the related document doesn't need to be invalidate (see @_invalidate_documents) 

723 # because the unlink method invalidates the whole cache anyway 

724 if not self: 

725 return True 

726 self.check_access('unlink') 

727 self.mapped('attachment_ids').filtered( 

728 lambda attach: attach.res_model == self._name and (attach.res_id in self.ids or attach.res_id == 0) 

729 ).unlink() 

730 messages_by_partner = defaultdict(lambda: self.env['mail.message']) 

731 partners_with_user = self.partner_ids.filtered('user_ids') 

732 for elem in self: 

733 for partner in (

735 ): 

736 messages_by_partner[partner] |= elem 

737 # Notify front-end of messages deletion for partners having a user 

738 for partner, messages in messages_by_partner.items(): 

739 partner._bus_send("mail.message/delete", {"message_ids": messages.ids}) 

740 return super(Message, self).unlink()

1100 res["count"] = self.search_count(domain) 

1101 if around is not None: 

1102 messages_before = self.search(domain=[*domain, ('id', '<=', around)], limit=limit // 2, order="id DESC") 

1103 messages_after = self.search(domain=[*domain, ('id', '>', around)], limit=limit // 2, order='id ASC') 

1104 return {**res, "messages": (messages_after + messages_before).sorted('id', reverse=True)} 

1105 if before: 

1106 domain = expression.AND([domain, [('id', '<', before)]]) 

1107 if after: 

1108 domain = expression.AND([domain, [('id', '>', after)]]) 

1109 res["messages"] = self.search(domain, limit=limit, order='id ASC' if after else 'id DESC') 

1110 if after: 

1111 res["messages"] = res["messages"].sorted('id', reverse=True)
