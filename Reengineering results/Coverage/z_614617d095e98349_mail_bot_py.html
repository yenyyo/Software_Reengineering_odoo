15 def _apply_logic(self, record, values, command=None): 

16 """ Apply bot logic to generate an answer (or not) for the user 

17 The logic will only be applied if odoobot is in a chat with a user or 

18 if someone pinged odoobot. 

19 

20 :param record: the discuss channel where the user message was posted/odoobot will answer. 

21 :param values: msg_values of the message_post or other values needed by logic 

22 :param command: the name of the called command if the logic is not triggered by a message_post 

23 """ 

24 record.ensure_one() 

25 odoobot_id = self.env['ir.model.data']._xmlid_to_res_id("base.partner_root") 

26 if values.get("author_id") == odoobot_id or values.get("message_type") != "comment" and not command: 

27 return 

28 body = values.get("body", "").replace("\xa0", " ").strip().lower().strip(".!") 

29 if answer := self._get_answer(record, body, values, command): 

30 record.sudo().message_post( 

31 author_id=odoobot_id, 

32 body=answer, 

33 message_type="comment", 

34 silent=True, 

35 subtype_xmlid="mail.mt_comment", 

36 )

53 def _get_answer(self, record, body, values, command=False): 

54 odoobot = self.env.ref("base.partner_root") 

55 # onboarding 

56 odoobot_state = self.env.user.odoobot_state 

57 

58 if record.channel_type == "chat" and odoobot in record.channel_member_ids.partner_id: 

59 # main flow 

60 source = _("Thanks") 

61 description = _("This is a temporary canned response to see how canned responses work.") 

62 if odoobot_state == 'onboarding_emoji' and self._body_contains_emoji(body): 

63 self.env.user.odoobot_state = "onboarding_command" 

64 self.env.user.odoobot_failed = False 

65 return html_escape( 

66 _("Great! 👍%(new_line)sTo access special commands, %(bold_start)sstart your sentence " 

67 "with%(bold_end)s %(command_start)s/%(command_end)s. Try getting help." 

68 )) % self._get_style_dict() 

69 elif odoobot_state == 'onboarding_command' and command == 'help': 

70 self.env.user.odoobot_state = "onboarding_ping" 

71 self.env.user.odoobot_failed = False 

72 return html_escape( 

73 _("Wow you are a natural!%(new_line)sPing someone with @username to grab their attention. " 

74 "%(bold_start)sTry to ping me using%(bold_end)s %(command_start)s@OdooBot%(command_end)s" 

75 " in a sentence.")) % self._get_style_dict() 

76 elif odoobot_state == "onboarding_ping" and odoobot.id in values.get("partner_ids", []): 

77 self.env.user.odoobot_state = "onboarding_attachement" 

78 self.env.user.odoobot_failed = False 

79 return html_escape( 

80 _("Yep, I am here! 🎉 %(new_line)sNow, try %(bold_start)ssending an attachment%(bold_end)s," 

81 " like a picture of your cute dog...")) % self._get_style_dict() 

82 elif odoobot_state == "onboarding_attachement" and values.get("attachment_ids"): 

83 self.env["mail.canned.response"].create({ 

84 "source": source, 

85 "substitution": _("Thanks for your feedback. Goodbye!"), 

86 "description": description, 

87 }) 

88 self.env.user.odoobot_failed = False 

89 self.env.user.odoobot_state = "onboarding_canned" 

90 return html_escape( 

91 _("Wonderful! 😇%(new_line)sTry typing %(command_start)s:%(command_end)s to use canned responses." 

92 " I've created a temporary one for you.")) % self._get_style_dict() 

93 elif odoobot_state == "onboarding_canned" and self.env.context.get("canned_response_ids"): 

94 self.env["mail.canned.response"].search([ 

95 ("create_uid", "=", self.env.user.id), 

96 ("source", "=", source), 

97 ("description", "=", description), 

98 ]).unlink() 

99 self.env.user.odoobot_failed = False 

100 self.env.user.odoobot_state = "idle" 

101 return html_escape( 

102 _("Good, you can customize canned responses in the Discuss application." 

103 "%(new_line)s%(new_line)s%(bold_start)sIt's the end of this overview%(bold_end)s," 

104 " you can now %(bold_start)sclose this conversation%(bold_end)s or start the tour again with" 

105 " typing %(command_start)sstart the tour%(command_end)s." 

106 " Enjoy discovering Odoo!")) % self._get_style_dict() 

107 # repeat question if needed 

108 elif odoobot_state == 'onboarding_canned' and not self._is_help_requested(body): 

109 self.env.user.odoobot_failed = True 

110 return html_escape( 

111 _("Not sure what you are doing. Please, type %(command_start)s:%(command_end)s and wait for the" 

112 " propositions. Select one of them and press enter.")) % self._get_style_dict() 

113 elif odoobot_state in (False, "idle", "not_initialized") and (_('start the tour') in body.lower()): 

114 self.env.user.odoobot_state = "onboarding_emoji" 

115 return _("To start, try to send me an emoji :)") 

116 # easter eggs 

117 elif odoobot_state == "idle" and body in ['❤️', _('i love you'), _('love')]: 

118 return _("Aaaaaw that's really cute but, you know, bots don't work that way. You're too human for me! Let's keep it professional ❤️") 

119 elif _('fuck') in body or "fuck" in body: 

120 return _("That's not nice! I'm a bot but I have feelings... 💔") 

121 # help message 

122 elif self._is_help_requested(body) or odoobot_state == 'idle': 

123 return html_escape( 

124 _("Unfortunately, I'm just a bot 😞 I don't understand! If you need help discovering our product, " 

125 "please check %(document_link_start)sour documentation%(document_link_end)s or" 

126 " %(slides_link_start)sour videos%(slides_link_end)s.")) % self._get_style_dict() 

127 else: 

128 # repeat question 

129 if odoobot_state == 'onboarding_emoji': 

130 self.env.user.odoobot_failed = True 

131 return (html_escape( 

132 _("Not exactly. To continue the tour, send an emoji:" 

133 " %(bold_start)stype%(bold_end)s%(command_start)s :)%(command_end)s and press enter.")) 

134 % self._get_style_dict()) 

135 elif odoobot_state == 'onboarding_attachement': 

136 self.env.user.odoobot_failed = True 

137 return html_escape( 

138 _("To %(bold_start)ssend an attachment%(bold_end)s, click on the %(paperclip_icon)s icon and" 

139 " select a file.")) % self._get_style_dict() 

140 elif odoobot_state == 'onboarding_command': 

141 self.env.user.odoobot_failed = True 

142 return html_escape( 

143 _("Not sure what you are doing. Please, type %(command_start)s/%(command_end)s and wait for the propositions." 

144 " Select %(command_start)shelp%(command_end)s and press enter.")) % self._get_style_dict() 

145 elif odoobot_state == 'onboarding_ping': 

146 self.env.user.odoobot_failed = True 

147 return html_escape( 

148 _("Sorry, I am not listening. To get someone's attention, %(bold_start)sping him%(bold_end)s." 

149 " Write %(command_start)s@OdooBot%(command_end)s and select me.")) % self._get_style_dict() 

150 return random.choice([ 

151 html_escape( 

152 _("I'm not smart enough to answer your question.%(new_line)sTo follow my guide," 

153 " ask: %(command_start)sstart the tour%(command_end)s.")) % self._get_style_dict(), 

154 _("Hmmm..."), 

155 _("I'm afraid I don't understand. Sorry!"), 

156 html_escape( 

157 _("Sorry I'm sleepy. Or not! Maybe I'm just trying to hide my unawareness of human language..." 

158 "%(new_line)sI can show you features if you write: %(command_start)sstart the" 

159 " tour%(command_end)s.")) % self._get_style_dict() 

160 ]) 

161 return False

300 def _is_bot_in_private_channel(self, record): 

301 odoobot_id = self.env['ir.model.data']._xmlid_to_res_id("base.partner_root") 

302 if record._name == 'discuss.channel' and record.channel_type == 'chat': 

303 return odoobot_id in record.with_context(active_test=False).channel_partner_ids.ids
