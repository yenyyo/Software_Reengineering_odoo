159 def _get_suggested_ubl_cii_edi_format(self): 

160 self.ensure_one() 

161 format_mapping = self._get_ubl_cii_formats_by_country() 

162 country_code = self.commercial_partner_id._deduce_country_code() 

163 if country_code in format_mapping: 

164 formats_by_country = format_mapping[country_code] 

165 # return the format with the smallest sequence 

166 if len(formats_by_country) == 1: 

167 return formats_by_country[0] 

168 else: 

169 formats_info = self._get_ubl_cii_formats_info() 

170 return min(formats_by_country, key=lambda e: formats_info[e].get('sequence', 100)) # we use a sequence of 100 by default 

171 return False

209 @api.depends(lambda self: self._peppol_eas_endpoint_depends() + ['peppol_eas']) 

210 def _compute_peppol_endpoint(self): 

211 """ If the EAS changes and a valid endpoint is available, set it. Otherwise, keep the existing value.""" 

212 for partner in self: 

213 partner.peppol_endpoint = partner.peppol_endpoint 

214 country_code = partner._deduce_country_code() 

215 if country_code in EAS_MAPPING: 

216 field = EAS_MAPPING[country_code].get(partner.peppol_eas) 

217 if field \ 

218 and field in partner._fields \ 

219 and partner[field] \ 

220 and not partner._build_error_peppol_endpoint(partner.peppol_eas, partner[field]): 

221 partner.peppol_endpoint = partner[field] 

222 

223 @api.depends(lambda self: self._peppol_eas_endpoint_depends()) 

224 def _compute_peppol_eas(self): 

225 """ 

226 If the country_code changes, recompute the EAS only if there is a country_code, it exists in the 

227 EAS_MAPPING, and the current EAS is not consistent with the new country_code. 

228 """ 

229 for partner in self: 

230 partner.peppol_eas = partner.peppol_eas 

231 country_code = partner._deduce_country_code() 

232 if country_code in EAS_MAPPING: 

233 eas_to_field = EAS_MAPPING[country_code] 

234 if partner.peppol_eas not in eas_to_field.keys(): 

235 new_eas = next(iter(EAS_MAPPING[country_code].keys())) 

236 # Iterate on the possible EAS until a valid one is found 

237 for eas, field in eas_to_field.items(): 

238 if field and field in partner._fields and partner[field]: 

239 if not partner._build_error_peppol_endpoint(eas, partner[field]): 

240 new_eas = eas 

241 break 

242 partner.peppol_eas = new_eas

244 def _build_error_peppol_endpoint(self, eas, endpoint): 

245 """ This function contains all the rules regarding the peppol_endpoint.""" 

246 if eas == '0208' and not re.match(r"^\d{10}$", endpoint): 

247 return _("The Peppol endpoint is not valid. The expected format is: 0239843188") 

248 if eas == '0009' and not siret.is_valid(endpoint): 

249 return _("The Peppol endpoint is not valid. The expected format is: 73282932000074") 

250 if eas == '0007' and not re.match(r"^\d{10}$", endpoint): 

251 return _("The Peppol endpoint is not valid. " 

252 "It should contain exactly 10 digits (Company Registry number)." 

253 "The expected format is: 1234567890")
