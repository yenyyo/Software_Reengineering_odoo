7class IrConfigParameter(models.Model):

11 # While being there, let us document quickly mail ICP.

13 # Emailing 

14 # * 'mail.mail.queue.batch.size': used in MailMail.process_email_queue() 

15 # to limit maximum number of mail.mail managed by each cron call to VALUE. 

16 # 1000 by default; 

17 # * 'mail.session.batch.size': used in MailMail._split_by_mail_configuration() 

18 # to prepare batches of maximum VALUE mails to give at '_send()' at each 

19 # iteration. For each iteration an SMTP server is opened and closed. It 

20 # prepares data for 'send' in conjunction with auto_commit=True in order 

21 # to avoid repeating batches in case of failure). 1000 by default; 

22 # * 'mail.mail.force.send.limit': used in 

23 # - MailThread._notify_thread_by_email(): notification emails flow 

24 # - MailComposer._action_send_mail_mass_mail(): mail composer in mass mail mode 

25 # to force the cron queue usage and avoid sending too much emails in a given 

26 # transaction. When 0 is set flows based on it are always using the email 

27 # queue, no direct send is performed. Default value is 100; 

28 # * 'mail.batch_size': used in 

29 # - MailComposer._action_send_mail_mass_mail(): mails generation based on records 

30 # - MailThread._notify_thread_by_email(): mails generation for notification emails 

31 # - MailTemplate.send_mail_batch(): mails generation done directly from templates 

32 # to split mail generation in batches; 

33 # - EventMail._execute_attendee_based() and EventMail._execute_event_based(): 

34 # mails (+ sms, whatsapp) generation for each attendee of en event; 

35 # 50 by default; 

36 # * 'mail.render.cron.limit': used in cron involving rendering of content 

37 # and/or templates, like event mail scheduler cron. Defaults to 1000; 

38 

39 # Mail Gateway 

40 # * 'mail.gateway.loop.minutes' and 'mail.gateway.loop.threshold': block 

41 # emails with same email_from if gateway received more than THRESHOLD 

42 # in MINUTES. This is used to break loops e.g. when email servers bounce 

43 # each other. 20 emails / 120 minutes by default; 

44 # * 'mail.default.from_filter': default from_filter used when there is 

45 # no specific outgoing mail server used to send emails; 

46 # * 'mail.catchall.domain.allowed': optional list of email domains that 

47 # restricts right-part of aliases when used in pre-17 compatibility 

48 # mode (see MailAlias.alias_incoming_local); 

49 

50 # Activities 

51 # * 'mail.activity.gc.delete_overdue_years': if set, activities outdated 

52 # for more than VALUE years are gc. 0 (skipped) by default; 

53 # * 'mail.activity.systray.limit': number of activities fetched by the 

54 # systray, to avoid performance issues notably with technical users that 

55 # rarely connect. 1000 by default;

83 group_user = self.env.ref('base.group_user') 

84 group_mail_template_editor = self.env.ref('mail.group_mail_template_editor') 

85

90 # remove existing users, including inactive template user

93 # sanitize and normalize allowed catchall domains 

94 elif key == 'mail.catchall.domain.allowed' and value:
