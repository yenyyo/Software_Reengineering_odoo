11 @http.route(_onboarding_return_url, type='http', methods=['GET'], auth='user') 

12 def stripe_return_from_onboarding(self, provider_id, menu_id): 

13 """ Redirect the user to the provider form of the onboarded Stripe account. 

14 

15 The user is redirected to this route by Stripe after or during (if the user clicks on a 

16 dedicated button) the onboarding. 

17 

18 :param str provider_id: The provider linked to the Stripe account being onboarded, as a 

19 `payment.provider` id 

20 :param str menu_id: The menu from which the user started the onboarding step, as an 

21 `ir.ui.menu` id 

22 """ 

23 stripe_provider = request.env['payment.provider'].browse(int(provider_id)) 

24 request.env['onboarding.onboarding.step'].with_company( 

25 stripe_provider.company_id 

26 ).action_validate_step_payment_provider() 

27 url = f"/odoo/action-payment_stripe.action_payment_provider_onboarding/{provider_id}?menu_id={menu_id}" 

28 return request.redirect(url)

30 @http.route(_onboarding_refresh_url, type='http', methods=['GET'], auth='user') 

31 def stripe_refresh_onboarding(self, provider_id, account_id, menu_id): 

32 """ Redirect the user to a new Stripe Connect onboarding link. 

33 

34 The user is redirected to this route by Stripe if the onboarding link they used was expired. 

35 

36 :param str provider_id: The provider linked to the Stripe account being onboarded, as a 

37 `payment.provider` id 

38 :param str account_id: The id of the connected account 

39 :param str menu_id: The menu from which the user started the onboarding step, as an 

40 `ir.ui.menu` id 

41 """ 

42 stripe_provider = request.env['payment.provider'].browse(int(provider_id)) 

43 account_link = stripe_provider._stripe_create_account_link(account_id, int(menu_id))
