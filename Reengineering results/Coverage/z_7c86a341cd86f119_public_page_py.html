60 def _response_discuss_channel_from_token(self, create_token, channel_name=None, default_display_mode=False): 

61 # sudo: ir.config_parameter - reading hard-coded key and using it in a simple condition 

62 if not request.env["ir.config_parameter"].sudo().get_param("mail.chat_from_token"): 

63 raise NotFound() 

64 # sudo: discuss.channel - channel access is validated with invitation_token 

65 channel_sudo = request.env["discuss.channel"].sudo().search([("uuid", "=", create_token)]) 

66 if not channel_sudo: 

67 try: 

68 channel_sudo = channel_sudo.create( 

69 { 

70 "channel_type": "channel", 

71 "default_display_mode": default_display_mode, 

72 "group_public_id": None, 

73 "name": channel_name or create_token, 

74 "uuid": create_token, 

75 } 

76 ) 

77 except psycopg2.errors.UniqueViolation: 

78 # concurrent insert attempt: another request created the channel. 

79 # commit the current transaction and get the channel. 

80 request.env.cr.commit() 

81 channel_sudo = channel_sudo.search([("uuid", "=", create_token)]) 

82 store = Store({"isChannelTokenSecret": False}) 

83 return self._response_discuss_channel_invitation(store, channel_sudo.sudo(False))
