17class GoogleGmailController(http.Controller): 

18 @http.route('/google_gmail/confirm', type='http', auth='user') 

19 def google_gmail_callback(self, code=None, state=None, error=None, **kwargs): 

20 """Callback URL during the OAuth process. 

21 

22 Gmail redirects the user browser to this endpoint with the authorization code. 

23 We will fetch the refresh token and the access token thanks to this authorization 

24 code and save those values on the given mail server. 

25 """ 

26 if not request.env.user.has_group('base.group_system'): 

27 _logger.error('Google Gmail: non-system user trying to link an Gmail account.') 

28 raise Forbidden() 

29 

30 if error: 

31 return _('An error occur during the authentication process.') 

32 

33 try: 

34 state = json.loads(state) 

35 model_name = state['model'] 

36 rec_id = state['id'] 

37 csrf_token = state['csrf_token'] 

38 except Exception: 

39 _logger.error('Google Gmail: Wrong state value %r.', state) 

40 raise Forbidden() 

41 

42 model = request.env[model_name] 

43 

44 if not isinstance(model, request.env.registry['google.gmail.mixin']): 

45 # The model must inherits from the "google.gmail.mixin" mixin 

46 raise Forbidden() 

47 

48 record = model.browse(rec_id).exists() 

49 if not record: 

50 raise Forbidden() 

51 

52 if not csrf_token or not consteq(csrf_token, record._get_gmail_csrf_token()): 

53 _logger.error('Google Gmail: Wrong CSRF token during Gmail authentication.') 

54 raise Forbidden() 

55 

56 try: 

57 refresh_token, access_token, expiration = record._fetch_gmail_refresh_token(code) 

58 except UserError: 

59 return _('An error occur during the authentication process.') 

60 

61 record.write({ 

62 'google_gmail_access_token': access_token, 

63 'google_gmail_access_token_expiration': expiration, 

64 'google_gmail_authorization_code': code, 

65 'google_gmail_refresh_token': refresh_token, 

66 }) 

67 

68 url = f'/odoo/{model_name}/{rec_id}' 

69 return request.redirect(url)
