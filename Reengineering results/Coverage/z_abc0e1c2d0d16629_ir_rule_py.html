78 def _get_failing(self, for_records, mode='read'): 

79 """ Returns the rules for the mode for the current user which fail on 

80 the specified records. 

81 

82 Can return any global rule and/or all local rules (since local rules 

83 are OR-ed together, the entire group succeeds or fails, while global 

84 rules get AND-ed and can each fail) 

85 """ 

86 Model = for_records.browse(()).sudo() 

87 eval_context = self._eval_context() 

88 

89 all_rules = self._get_rules(Model._name, mode=mode).sudo() 

90 

91 # first check if the group rules fail for any record (aka if 

92 # searching on (records, group_rules) filters out some of the records) 

93 group_rules = all_rules.filtered(lambda r: r.groups and r.groups & self.env.user.groups_id) 

94 group_domains = expression.OR([ 

95 safe_eval(r.domain_force, eval_context) if r.domain_force else [] 

96 for r in group_rules 

97 ]) 

98 # if all records get returned, the group rules are not failing 

99 if Model.search_count(expression.AND([[('id', 'in', for_records.ids)], group_domains])) == len(for_records): 

100 group_rules = self.browse(()) 

101 

102 # failing rules are previously selected group rules or any failing global rule 

103 def is_failing(r, ids=for_records.ids): 

104 dom = safe_eval(r.domain_force, eval_context) if r.domain_force else [] 

105 return Model.search_count(expression.AND([ 

106 [('id', 'in', ids)], 

107 expression.normalize_domain(dom) 

108 ])) < len(ids) 

109 

110 return all_rules.filtered(lambda r: r in group_rules or (not r.groups and is_failing(r))).with_user(self.env.user)

112 def _get_rules(self, model_name, mode='read'): 

113 """ Returns all the rules matching the model for the mode for the 

114 current user. 

115 """ 

116 if mode not in self._MODES: 

117 raise ValueError('Invalid mode: %r' % (mode,)) 

118 

119 if self.env.su: 

120 return self.browse(()) 

121 

122 sql = SQL(""" 

123 SELECT r.id FROM ir_rule r 

124 JOIN ir_model m ON (r.model_id=m.id) 

125 WHERE m.model = %s AND r.active AND r.perm_%s 

126 AND (r.global OR r.id IN ( 

127 SELECT rule_group_id FROM rule_group_rel rg 

128 WHERE rg.group_id IN %s 

129 )) 

130 ORDER BY r.id 

131 """, model_name, SQL(mode), tuple(self.env.user._get_group_ids()) or (None,)) 

132 return self.browse(v for v, in self.env.execute_query(sql))

140 def _compute_domain(self, model_name, mode="read"): 

141 global_domains = [] # list of domains 

142 

143 # add rules for parent models 

144 for parent_model_name, parent_field_name in self.env[model_name]._inherits.items(): 

145 if domain := self._compute_domain(parent_model_name, mode): 

146 global_domains.append([(parent_field_name, 'any', domain)]) 

147 

148 rules = self._get_rules(model_name, mode=mode) 

149 if not rules: 

150 return expression.AND(global_domains) if global_domains else [] 

151 

152 # browse user and rules with sudo to avoid access errors! 

153 eval_context = self._eval_context() 

154 user_groups = self.env.user.groups_id 

155 group_domains = [] # list of domains 

156 for rule in rules.sudo(): 

157 # evaluate the domain for the current user 

158 dom = safe_eval(rule.domain_force, eval_context) if rule.domain_force else [] 

159 dom = expression.normalize_domain(dom) 

160 if not rule.groups: 

161 global_domains.append(dom) 

162 elif rule.groups & user_groups: 

163 group_domains.append(dom) 

164 

165 # combine global domains and group domains 

166 if not group_domains: 

167 return expression.AND(global_domains) 

168 return expression.AND(global_domains + [expression.OR(group_domains)])
