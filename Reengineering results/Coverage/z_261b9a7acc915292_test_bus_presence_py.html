17class TestBusPresence(WebsocketCase, MailCommon): 

18 def _receive_presence(self, sender, recipient): 

19 self._reset_bus() 

20 sent_from_user = isinstance(sender, self.env.registry["res.users"]) 

21 receive_to_user = isinstance(recipient, self.env.registry["res.users"]) 

22 if receive_to_user: 

23 session = self.authenticate(recipient.login, recipient.login) 

24 auth_cookie = f"session_id={session.sid};" 

25 else: 

26 self.authenticate(None, None) 

27 auth_cookie = f"{recipient._cookie_name}={recipient._format_auth_cookie()};" 

28 websocket = self.websocket_connect(cookie=auth_cookie, timeout=1) 

29 sender_bus_target = sender.partner_id if sent_from_user else sender 

30 self.subscribe( 

31 websocket, 

32 [f"odoo-presence-{sender_bus_target._name}_{sender_bus_target.id}"], 

33 self.env["bus.bus"]._bus_last_id(), 

34 ) 

35 self.env["bus.presence"].create( 

36 {"user_id" if sent_from_user else "guest_id": sender.id, "status": "online"} 

37 ) 

38 self.trigger_notification_dispatching([(sender_bus_target, "presence")]) 

39 notifications = json.loads(websocket.recv()) 

40 self._close_websockets() 

41 bus_record = self.env["bus.bus"].search([("id", "=", int(notifications[0]["id"]))]) 

42 self.assertEqual( 

43 bus_record.channel, 

44 json_dump(channel_with_db(self.env.cr.dbname, (sender_bus_target, "presence"))), 

45 ) 

46 self.assertEqual(notifications[0]["message"]["type"], "bus.bus/im_status_updated") 

47 self.assertEqual(notifications[0]["message"]["payload"]["im_status"], "online") 

48 self.assertEqual( 

49 notifications[0]["message"]["payload"]["partner_id" if sent_from_user else "guest_id"], 

50 sender_bus_target.id, 

51 )

53 def test_receive_presences_as_guest(self): 

54 guest = self.env["mail.guest"].create({"name": "Guest"}) 

55 bob = new_test_user(self.env, login="bob_user", groups="base.group_user") 

56 # Guest should not receive users's presence: no common channel. 

57 with self.assertRaises(ws._exceptions.WebSocketTimeoutException): 

58 self._receive_presence(sender=bob, recipient=guest) 

59 channel = self.env["discuss.channel"].channel_create(group_id=None, name="General") 

60 channel.add_members(guest_ids=[guest.id], partner_ids=[bob.partner_id.id]) 

61 # Now that they share a channel, guest should receive users's presence. 

62 self._receive_presence(sender=bob, recipient=guest) 

63 

64 other_guest = self.env["mail.guest"].create({"name": "OtherGuest"}) 

65 # Guest should not receive guest's presence: no common channel. 

66 with self.assertRaises(ws._exceptions.WebSocketTimeoutException): 

67 self._receive_presence(sender=other_guest, recipient=guest) 

68 channel.add_members(guest_ids=[other_guest.id]) 

69 # Now that they share a channel, guest should receive guest's presence. 

70 self._receive_presence(sender=other_guest, recipient=guest)
