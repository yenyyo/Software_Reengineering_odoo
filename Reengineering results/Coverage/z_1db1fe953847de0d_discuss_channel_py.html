507 def _notify_get_recipients(self, message, msg_vals, **kwargs): 

508 """ Override recipients computation as channel is not a standard 

509 mail.thread document. Indeed there are no followers on a channel. 

510 Instead of followers it has members that should be notified. 

511 

512 :param message: see ``MailThread._notify_get_recipients()``; 

513 :param msg_vals: see ``MailThread._notify_get_recipients()``; 

514 :param kwargs: see ``MailThread._notify_get_recipients()``; 

515 

516 :return recipients: structured data holding recipients data. See 

517 ``MailThread._notify_thread()`` for more details about its content 

518 and use; 

519 """ 

520 # get values from msg_vals or from message if msg_vals doen't exists 

521 message_type = msg_vals.get('message_type', 'comment') if msg_vals else message.message_type 

522 pids = msg_vals.get('partner_ids', []) if msg_vals else message.partner_ids.ids 

523 

524 # notify only user input (comment, whatsapp messages or incoming / outgoing emails) 

525 if message_type not in ('comment', 'email', 'email_outgoing', 'whatsapp_message'): 

526 return [] 

527 

528 recipients_data = [] 

529 author_id = msg_vals.get("author_id") or message.author_id.id 

530 if pids: 

531 email_from = tools.email_normalize(msg_vals.get('email_from') or message.email_from) 

532 self.env['res.partner'].flush_model(['active', 'email', 'partner_share']) 

533 self.env['res.users'].flush_model(['notification_type', 'partner_id']) 

534 sql_query = """ 

535 SELECT DISTINCT ON (partner.id) partner.id, 

536 partner.lang, 

537 partner.partner_share, 

538 users.id as uid, 

539 COALESCE(users.notification_type, 'email') as notif, 

540 COALESCE(users.share, FALSE) as ushare 

541 FROM res_partner partner 

542 LEFT JOIN res_users users on partner.id = users.partner_id 

543 WHERE partner.active IS TRUE 

544 AND partner.email != %s 

545 AND partner.id = ANY(%s) AND partner.id != ANY(%s)""" 

546 self.env.cr.execute( 

547 sql_query, 

548 (email_from or '', list(pids), [author_id] if author_id else [], ) 

549 ) 

550 for partner_id, lang, partner_share, uid, notif, ushare in self._cr.fetchall(): 

551 # ocn_client: will add partners to recipient recipient_data. more ocn notifications. We neeed to filter them maybe 

552 recipients_data.append({ 

553 'active': True, 

554 'id': partner_id, 

555 'is_follower': False, 

556 'groups': [], 

557 'lang': lang, 

558 'notif': notif, 

559 'share': partner_share, 

560 'type': 'user' if not partner_share and notif else 'customer', 

561 'uid': uid, 

562 'ushare': ushare, 

563 }) 

564 

565 domain = expression.AND([ 

566 [("channel_id", "=", self.id)], 

567 [("partner_id", "!=", author_id)], 

568 [("partner_id.active", "=", True)], 

569 [("mute_until_dt", "=", False)], 

570 [("partner_id.user_ids.res_users_settings_ids.mute_until_dt", "=", False)], 

571 expression.OR([ 

572 [("channel_id.channel_type", "!=", "channel")], 

573 expression.AND([ 

574 [("channel_id.channel_type", "=", "channel")], 

575 expression.OR([ 

576 [("custom_notifications", "=", "all")], 

577 expression.AND([ 

578 [("custom_notifications", "=", False)], 

579 [("partner_id.user_ids.res_users_settings_ids.channel_notifications", "=", "all")], 

580 ]), 

581 expression.AND([ 

582 [("custom_notifications", "=", "mentions")], 

583 [("partner_id", "in", pids)], 

584 ]), 

585 expression.AND([ 

586 [("custom_notifications", "=", False)], 

587 [("partner_id.user_ids.res_users_settings_ids.channel_notifications", "=", False)], 

588 [("partner_id", "in", pids)], 

589 ]), 

590 ]), 

591 ]), 

592 ]), 

593 ]) 

594 # sudo: discuss.channel.member - read to get the members of the channel and res.users.settings of the partners 

595 members = self.env["discuss.channel.member"].sudo().search(domain) 

596 for member in members: 

597 recipients_data.append({ 

598 "active": True, 

599 "id": member.partner_id.id, 

600 "is_follower": False, 

601 "groups": [], 

602 "lang": member.partner_id.lang, 

603 "notif": "web_push", 

604 "share": member.partner_id.partner_share, 

605 "type": "customer", 

606 "uid": False, 

607 "ushare": False, 

608 }) 

609 return recipients_data

887 def _to_store(self, store: Store): 

888 """Adds channel data to the given store.""" 

889 if not self: 

890 return [] 

891 # sudo: bus.bus: reading non-sensitive last id 

892 bus_last_id = self.env["bus.bus"].sudo()._bus_last_id() 

893 current_partner, current_guest = self.env["res.partner"]._get_current_persona() 

894 self.env['discuss.channel'].flush_model() 

895 self.env['discuss.channel.member'].flush_model() 

896 # Query instead of ORM for performance reasons: "LEFT JOIN" is more 

897 # efficient than "id IN" for the cross-table condition between channel 

898 # (for channel_type) and member (for other fields). 

899 self.env.cr.execute(""" 

900 SELECT discuss_channel_member.id 

901 FROM discuss_channel_member 

902 LEFT JOIN discuss_channel 

903 ON discuss_channel.id = discuss_channel_member.channel_id 

904 AND discuss_channel.channel_type != 'channel' 

905 WHERE discuss_channel_member.channel_id in %(channel_ids)s 

906 AND ( 

907 discuss_channel.id IS NOT NULL 

908 OR discuss_channel_member.rtc_inviting_session_id IS NOT NULL 

909 OR discuss_channel_member.partner_id = %(current_partner_id)s 

910 OR discuss_channel_member.guest_id = %(current_guest_id)s 

911 ) 

912 ORDER BY discuss_channel_member.id ASC 

913 """, {'channel_ids': tuple(self.ids), 'current_partner_id': current_partner.id or None, 'current_guest_id': current_guest.id or None}) 

914 all_needed_members = self.env['discuss.channel.member'].browse([m['id'] for m in self.env.cr.dictfetchall()]) 

915 Store(all_needed_members) # prefetch in batch 

916 members_by_channel = defaultdict(lambda: self.env['discuss.channel.member']) 

917 invited_members_by_channel = defaultdict(lambda: self.env['discuss.channel.member']) 

918 member_of_current_user_by_channel = defaultdict(lambda: self.env['discuss.channel.member']) 

919 for member in all_needed_members: 

920 members_by_channel[member.channel_id] += member 

921 if member.rtc_inviting_session_id: 

922 invited_members_by_channel[member.channel_id] += member 

923 if (current_partner and member.partner_id == current_partner) or (current_guest and member.guest_id == current_guest): 

924 member_of_current_user_by_channel[member.channel_id] = member 

925 for channel in self: 

926 member = member_of_current_user_by_channel.get(channel, self.env['discuss.channel.member']).with_prefetch([m.id for m in member_of_current_user_by_channel.values()]) 

927 info = channel._channel_basic_info() 

928 info["is_editable"] = channel.is_editable 

929 info["fetchChannelInfoState"] = "fetched" 

930 info["parent_channel_id"] = Store.one(channel.parent_channel_id) 

931 info["from_message_id"] = Store.one(channel.from_message_id) 

932 info["group_public_id"] = channel.group_public_id.id or False 

933 # find the channel member state 

934 if current_partner or current_guest: 

935 info['message_needaction_counter'] = channel.message_needaction_counter 

936 info["message_needaction_counter_bus_id"] = bus_last_id 

937 if member: 

938 store.add( 

939 member, 

940 extra_fields={ 

941 "last_interest_dt": True, 

942 "message_unread_counter": True, 

943 "message_unread_counter_bus_id": bus_last_id, 

944 "new_message_separator": True 

945 }, 

946 ) 

947 info['state'] = member.fold_state or 'closed' 

948 info['custom_notifications'] = member.custom_notifications 

949 info['mute_until_dt'] = fields.Datetime.to_string(member.mute_until_dt) 

950 info['custom_channel_name'] = member.custom_channel_name 

951 info['is_pinned'] = member.is_pinned 

952 if member.rtc_inviting_session_id: 

953 # sudo: discuss.channel.rtc.session - reading sessions of accessible channel is acceptable 

954 info["rtcInvitingSession"] = Store.one(member.rtc_inviting_session_id.sudo()) 

955 # add members info 

956 if channel.channel_type != 'channel': 

957 # avoid sending potentially a lot of members for big channels 

958 # exclude chat and other small channels from this optimization because they are 

959 # assumed to be smaller and it's important to know the member list for them 

960 store.add(members_by_channel[channel] - member)

962 invited_members = invited_members_by_channel[channel]

973 def channel_get(self, partners_to, pin=True, force_open=False): 

974 """ Get the canonical private channel between some partners, create it if needed. 

975 To reuse an old channel (conversation), this one must be private, and contains 

976 only the given partners. 

977 :param partners_to : list of res.partner ids to add to the conversation 

978 :param pin : True if getting the channel should pin it for the current user 

979 :param force_open : True if getting the channel should open it for the current user 

980 :returns: channel_info of the created or existing channel 

981 :rtype: dict 

982 """ 

983 if self.env.user.partner_id.id not in partners_to: 

984 partners_to.append(self.env.user.partner_id.id) 

985 if len(partners_to) > 2: 

986 raise UserError(_("A chat should not be created with more than 2 persons. Create a group instead.")) 

987 # determine type according to the number of partner in the channel 

988 self.flush_model() 

989 self.env['discuss.channel.member'].flush_model() 

990 self.env.cr.execute(""" 

991 SELECT M.channel_id 

992 FROM discuss_channel C, discuss_channel_member M 

993 WHERE M.channel_id = C.id 

994 AND M.partner_id IN %s 

995 AND C.channel_type LIKE 'chat' 

996 AND NOT EXISTS ( 

997 SELECT 1 

998 FROM discuss_channel_member M2 

999 WHERE M2.channel_id = C.id 

1000 AND M2.partner_id NOT IN %s 

1001 ) 

1002 GROUP BY M.channel_id 

1003 HAVING ARRAY_AGG(DISTINCT M.partner_id ORDER BY M.partner_id) = %s 

1004 LIMIT 1 

1005 """, (tuple(partners_to), tuple(partners_to), sorted(list(partners_to)),)) 

1006 result = self.env.cr.dictfetchall() 

1007 if result: 

1008 # get the existing channel between the given partners 

1009 channel = self.browse(result[0].get('channel_id')) 

1010 # pin or open the channel for the current partner 

1011 if pin or force_open: 

1012 member = self.env['discuss.channel.member'].search([('partner_id', '=', self.env.user.partner_id.id), ('channel_id', '=', channel.id)]) 

1013 vals = {'last_interest_dt': fields.Datetime.now()} 

1014 if pin: 

1015 vals['unpin_dt'] = False 

1016 if force_open: 

1017 vals['fold_state'] = "open" 

1018 member.write(vals) 

1019 channel._broadcast(self.env.user.partner_id.ids) 

1020 else: 

1021 # create a new one 

1022 channel = self.create({ 

1023 'channel_member_ids': [ 

1024 Command.create({ 

1025 'partner_id': partner_id, 

1026 # only pin for the current user, so the chat does not show up for the correspondent until a message has been sent 

1027 # manually set the last_interest_dt to make sure that it works well with the default last_interest_dt (datetime.now()) 

1028 'unpin_dt': False if partner_id == self.env.user.partner_id.id else fields.Datetime.now(), 

1029 'last_interest_dt': fields.Datetime.now() if partner_id == self.env.user.partner_id.id else fields.Datetime.now() - timedelta(seconds=30), 

1030 }) for partner_id in partners_to 

1031 ], 

1032 'channel_type': 'chat', 

1033 'name': ', '.join(self.env['res.partner'].browse(partners_to).mapped('name')), 

1034 }) 

1035 channel._broadcast(partners_to) 

1036 return channel
