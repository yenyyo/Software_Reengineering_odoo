11class Followers(models.Model): 

12 """ mail_followers holds the data related to the follow mechanism inside 

13 Odoo. Partners can choose to follow documents (records) of any kind 

14 that inherits from mail.thread. Following documents allow to receive 

15 notifications for new messages. A subscription is characterized by: 

16 

17 :param: res_model: model of the followed objects 

18 :param: res_id: ID of resource (may be 0 for every objects) 

19 """

21 _rec_name = 'partner_id' 

22 _log_access = False 

23 _description = 'Document Followers'

41 def _invalidate_documents(self, vals_list=None): 

42 """ Invalidate the cache of the documents followed by ``self``. 

43 

44 Modifying followers change access rights to individual documents. As the 

45 cache may contain accessible/inaccessible data, one has to refresh it. 

46 """ 

47 to_invalidate = defaultdict(list) 

48 for record in (vals_list or [{'res_model': rec.res_model, 'res_id': rec.res_id} for rec in self]): 

49 if record.get('res_id'): 

50 to_invalidate[record.get('res_model')].append(record.get('res_id'))

105 def _get_recipient_data(self, records, message_type, subtype_id, pids=None): 

106 """ Private method allowing to fetch recipients data based on a subtype. 

107 Purpose of this method is to fetch all data necessary to notify recipients 

108 in a single query. It fetches data from 

109 

110 * followers (partners and channels) of records that follow the given 

111 subtype if records and subtype are set; 

112 * partners if pids is given; 

113 

114 :param records: fetch data from followers of ``records`` that follow 

115 ``subtype_id``; 

116 :param str message_type: mail.message.message_type in order to allow custom 

117 behavior depending on it (SMS for example); 

118 :param int subtype_id: mail.message.subtype to check against followers; 

119 :param pids: additional set of partner IDs from which to fetch recipient 

120 data independently from following status; 

121 

122 :return dict: recipients data based on record.ids if given, else a generic 

123 '0' key to keep a dict-like return format. Each item is a dict based on 

124 recipients partner ids formatted like 

125 {'active': whether partner is active; 

126 'id': res.partner ID; 

127 'is_follower': True if linked to a record and if partner is a follower; 

128 'lang': lang of the partner; 

129 'groups': groups of the partner's user. If several users exist preference 

130 is given to internal user, then share users. In case of multiples 

131 users of same kind groups are unioned; 

132 'notif': notification type ('inbox' or 'email'). Overrides may change 

133 this value (e.g. 'sms' in sms module); 

134 'share': if partner is a customer (no user or share user); 

135 'ushare': if partner has users, whether all are shared (public or portal); 

136 'type': summary of partner 'usage' (portal, customer, internal user); 

137 } 

138 """ 

139 self.env['mail.followers'].flush_model(['partner_id', 'subtype_ids']) 

140 self.env['mail.message.subtype'].flush_model(['internal']) 

141 self.env['res.users'].flush_model(['notification_type', 'active', 'partner_id', 'groups_id']) 

142 self.env['res.partner'].flush_model(['active', 'partner_share']) 

143 self.env['res.groups'].flush_model(['users']) 

144 # if we have records and a subtype: we have to fetch followers, unless being 

145 # in user notification mode (contact only pids) 

146 if message_type != 'user_notification' and records and subtype_id: 

147 query = """ 

148 WITH sub_followers AS ( 

149 SELECT fol.partner_id AS pid, 

150 fol.id AS fid, 

151 fol.res_id AS res_id, 

152 TRUE as is_follower, 

153 COALESCE(subrel.follow, FALSE) AS subtype_follower, 

154 COALESCE(subrel.internal, FALSE) AS internal 

155 FROM mail_followers fol 

156 LEFT JOIN LATERAL ( 

157 SELECT TRUE AS follow, 

158 subtype.internal AS internal 

159 FROM mail_followers_mail_message_subtype_rel m 

160 LEFT JOIN mail_message_subtype subtype ON subtype.id = m.mail_message_subtype_id 

161 WHERE m.mail_followers_id = fol.id AND m.mail_message_subtype_id = %s 

162 ) subrel ON TRUE 

163 WHERE fol.res_model = %s 

164 AND fol.res_id IN %s 

165 

166 UNION ALL 

167 

168 SELECT res_partner.id AS pid, 

169 0 AS fid, 

170 0 AS res_id, 

171 FALSE as is_follower, 

172 FALSE as subtype_follower, 

173 FALSE as internal 

174 FROM res_partner 

175 WHERE res_partner.id = ANY(%s) 

176 ) 

177 SELECT partner.id as pid, 

178 partner.active as active, 

179 partner.lang as lang, 

180 partner.partner_share as pshare, 

181 sub_user.uid as uid, 

182 COALESCE(sub_user.share, FALSE) as ushare, 

183 COALESCE(sub_user.notification_type, 'email') as notif, 

184 sub_user.groups as groups, 

185 sub_followers.res_id as res_id, 

186 sub_followers.is_follower as _insert_followerslower 

187 FROM res_partner partner 

188 JOIN sub_followers ON sub_followers.pid = partner.id 

189 AND (sub_followers.internal IS NOT TRUE OR partner.partner_share IS NOT TRUE) 

190 LEFT JOIN LATERAL ( 

191 SELECT users.id AS uid, 

192 users.share AS share, 

193 users.notification_type AS notification_type, 

194 ARRAY_AGG(groups_rel.gid) FILTER (WHERE groups_rel.gid IS NOT NULL) AS groups 

195 FROM res_users users 

196 LEFT JOIN res_groups_users_rel groups_rel ON groups_rel.uid = users.id 

197 WHERE users.partner_id = partner.id AND users.active 

198 GROUP BY users.id, 

199 users.share, 

200 users.notification_type 

201 ORDER BY users.share ASC NULLS FIRST, users.id ASC 

202 FETCH FIRST ROW ONLY 

203 ) sub_user ON TRUE 

204 

205 WHERE sub_followers.subtype_follower OR partner.id = ANY(%s) 

206""" 

207 params = [subtype_id, records._name, tuple(records.ids), list(pids or []), list(pids or [])]

211 elif pids and records:

335 def _get_subscription_data(self, doc_data, pids, include_pshare=False, include_active=False): 

336 """ Private method allowing to fetch follower data from several documents of a given model. 

337 Followers can be filtered given partner IDs and channel IDs. 

338 

339 :param doc_data: list of pair (res_model, res_ids) that are the documents from which we 

340 want to have subscription data; 

341 :param pids: optional partner to filter; if None take all, otherwise limitate to pids 

342 :param include_pshare: optional join in partner to fetch their share status 

343 :param include_active: optional join in partner to fetch their active flag 

344 

345 :return: list of followers data which is a list of tuples containing 

346 follower ID, 

347 document ID, 

348 partner ID, 

349 followed subtype IDs, 

350 share status of partner (returned only if include_pshare is True) 

351 active flag status of partner (returned only if include_active is True) 

352 """ 

353 self.env['mail.followers'].flush_model(['partner_id', 'res_id', 'res_model', 'subtype_ids']) 

354 self.env['res.partner'].flush_model(['active', 'partner_share']) 

355 # base query: fetch followers of given documents 

356 where_clause = ' OR '.join(['fol.res_model = %s AND fol.res_id IN %s'] * len(doc_data)) 

357 where_params = list(itertools.chain.from_iterable((rm, tuple(rids)) for rm, rids in doc_data)) 

358 

359 # additional: filter on optional pids 

360 sub_where = [] 

361 if pids: 

362 sub_where += ["fol.partner_id IN %s"] 

363 where_params.append(tuple(pids)) 

364 elif pids is not None: 

365 sub_where += ["fol.partner_id IS NULL"] 

366 if sub_where: 

367 where_clause += "AND (%s)" % " OR ".join(sub_where) 

368 

369 query = """ 

370SELECT fol.id, fol.res_id, fol.partner_id, array_agg(subtype.id)%s%s 

371FROM mail_followers fol 

372%s 

373LEFT JOIN mail_followers_mail_message_subtype_rel fol_rel ON fol_rel.mail_followers_id = fol.id 

374LEFT JOIN mail_message_subtype subtype ON subtype.id = fol_rel.mail_message_subtype_id 

375WHERE %s 

376GROUP BY fol.id%s%s""" % ( 

377 ', partner.partner_share' if include_pshare else '', 

378 ', partner.active' if include_active else '', 

379 'LEFT JOIN res_partner partner ON partner.id = fol.partner_id' if (include_pshare or include_active) else '', 

380 where_clause, 

381 ', partner.partner_share' if include_pshare else '', 

382 ', partner.active' if include_active else '' 

383 ) 

384 self.env.cr.execute(query, tuple(where_params)) 

385 return self.env.cr.fetchall()

391 def _insert_followers(self, res_model, res_ids, 

392 partner_ids, subtypes=None, 

393 customer_ids=None, check_existing=True, existing_policy='skip'): 

394 """ Main internal method allowing to create or update followers for documents, given a 

395 res_model and the document res_ids. This method does not handle access rights. This is the 

396 role of the caller to ensure there is no security breach. 

397 

398 :param subtypes: see ``_add_followers``. If not given, default ones are computed. 

399 :param customer_ids: see ``_add_default_followers`` 

400 :param check_existing: see ``_add_followers``; 

401 :param existing_policy: see ``_add_followers``; 

402 """ 

403 sudo_self = self.sudo().with_context(default_partner_id=False) 

404 if not subtypes: # no subtypes -> default computation, no force, skip existing 

405 new, upd = self._add_default_followers( 

406 res_model, res_ids, partner_ids, 

407 customer_ids=customer_ids, 

408 check_existing=check_existing, 

409 existing_policy=existing_policy) 

410 else: 

411 new, upd = self._add_followers( 

412 res_model, res_ids, 

413 partner_ids, subtypes, 

414 check_existing=check_existing, 

415 existing_policy=existing_policy) 

416 if new: 

417 sudo_self.create([ 

418 dict(values, res_id=res_id) 

419 for res_id, values_list in new.items() 

420 for values in values_list 

421 ]) 

422 for fol_id, values in upd.items(): 

423 sudo_self.browse(fol_id).write(values)

425 def _add_default_followers(self, res_model, res_ids, partner_ids, customer_ids=None, 

426 check_existing=True, existing_policy='skip'): 

427 """ Shortcut to ``_add_followers`` that computes default subtypes. Existing 

428 followers are skipped as their subscription is considered as more important 

429 compared to new default subscription. 

430 

431 :param customer_ids: optional list of partner ids that are customers. It is used if computing 

432 default subtype is necessary and allow to avoid the check of partners being customers (no 

433 user or share user). It is just a matter of saving queries if the info is already known; 

434 :param check_existing: see ``_add_followers``; 

435 :param existing_policy: see ``_add_followers``; 

436 

437 :return: see ``_add_followers`` 

438 """ 

439 if not partner_ids: 

440 return dict(), dict() 

441 

442 default, _, external = self.env['mail.message.subtype'].default_subtypes(res_model) 

443 if partner_ids and customer_ids is None: 

444 customer_ids = self.env['res.partner'].sudo().search([('id', 'in', partner_ids), ('partner_share', '=', True)]).ids 

445 

446 p_stypes = dict((pid, external.ids if pid in customer_ids else default.ids) for pid in partner_ids) 

447 

448 return self._add_followers(res_model, res_ids, partner_ids, p_stypes, check_existing=check_existing, existing_policy=existing_policy)

450 def _add_followers(self, res_model, res_ids, partner_ids, subtypes, 

451 check_existing=False, existing_policy='skip'): 

452 """ Internal method that generates values to insert or update followers. Callers have to 

453 handle the result, for example by making a valid ORM command, inserting or updating directly 

454 follower records, ... This method returns two main data 

455 

456 * first one is a dict which keys are res_ids. Value is a list of dict of values valid for 

457 creating new followers for the related res_id; 

458 * second one is a dict which keys are follower ids. Value is a dict of values valid for 

459 updating the related follower record; 

460 

461 :param subtypes: optional subtypes for new partner followers. This 

462 is a dict whose keys are partner IDs and value subtype IDs for that 

463 partner. 

464 :param channel_subtypes: optional subtypes for new channel followers. This 

465 is a dict whose keys are channel IDs and value subtype IDs for that 

466 channel. 

467 :param check_existing: if True, check for existing followers for given 

468 documents and handle them according to existing_policy parameter. 

469 Setting to False allows to save some computation if caller is sure 

470 there are no conflict for followers; 

471 :param existing policy: if check_existing, tells what to do with already 

472 existing followers: 

473 

474 * skip: simply skip existing followers, do not touch them; 

475 * force: update existing with given subtypes only; 

476 * replace: replace existing with new subtypes (like force without old / new follower); 

477 * update: gives an update dict allowing to add missing subtypes (no subtype removal); 

478 """ 

479 _res_ids = res_ids or [0] 

480 data_fols, doc_pids = dict(), dict((i, set()) for i in _res_ids) 

481 

482 if check_existing and res_ids: 

483 for fid, rid, pid, sids in self._get_subscription_data([(res_model, res_ids)], partner_ids or None): 

484 if existing_policy != 'force': 

485 if pid: 

486 doc_pids[rid].add(pid) 

487 data_fols[fid] = (rid, pid, sids) 

488 

489 if existing_policy == 'force': 

490 self.sudo().browse(data_fols.keys()).unlink() 

491 

492 new, update = dict(), dict() 

493 for res_id in _res_ids: 

494 for partner_id in set(partner_ids or []): 

495 if partner_id not in doc_pids[res_id]: 

496 new.setdefault(res_id, list()).append({ 

497 'res_model': res_model, 

498 'partner_id': partner_id, 

499 'subtype_ids': [Command.set(subtypes[partner_id])], 

500 }) 

501 elif existing_policy in ('replace', 'update'): 

502 fol_id, sids = next(((key, val[2]) for key, val in data_fols.items() if val[0] == res_id and val[1] == partner_id), (False, [])) 

503 new_sids = set(subtypes[partner_id]) - set(sids) 

504 old_sids = set(sids) - set(subtypes[partner_id]) 

505 update_cmd = [] 

506 if fol_id and new_sids: 

507 update_cmd += [Command.link(sid) for sid in new_sids] 

508 if fol_id and old_sids and existing_policy == 'replace': 

509 update_cmd += [Command.unlink(sid) for sid in old_sids] 

510 if update_cmd: 

511 update[fol_id] = {'subtype_ids': update_cmd} 

512 

513 return new, update
