119 def action_stripe_connect_account(self, menu_id=None): 

120 """ Create a Stripe Connect account and redirect the user to the next onboarding step. 

121 

122 If the provider is already enabled, close the current window. Otherwise, generate a Stripe 

123 Connect onboarding link and redirect the user to it. If provided, the menu id is included in 

124 the URL the user is redirected to when coming back on Odoo after the onboarding. If the link 

125 generation failed, redirect the user to the provider form. 

126 

127 Note: This method serves as a hook for modules that would fully implement Stripe Connect. 

128 Note: self.ensure_one() 

129 

130 :param int menu_id: The menu from which the user started the onboarding step, as an 

131 `ir.ui.menu` id. 

132 :return: The next step action 

133 :rtype: dict 

134 """ 

135 self.ensure_one() 

136 

137 if self.env.company.country_id.code not in const.SUPPORTED_COUNTRIES: 

138 raise RedirectWarning( 

139 _( 

140 "Stripe Connect is not available in your country, please use another payment" 

141 " provider." 

142 ), 

143 self.env.ref('payment.action_payment_provider').id, 

144 _("Other Payment Providers"), 

145 ) 

146 

147 if self.state == 'enabled': 

148 self.env['onboarding.onboarding.step'].action_validate_step_payment_provider() 

149 action = {'type': 'ir.actions.act_window_close'} 

150 else: 

151 # Account creation 

152 connected_account = self._stripe_fetch_or_create_connected_account() 

153 

154 # Link generation 

155 if not menu_id: 

156 # Fall back on `account_payment`'s menu if it is installed. If not, the user is 

157 # redirected to the provider's form view but without any menu in the breadcrumb. 

158 menu = self.env.ref('account_payment.payment_provider_menu', False) 

159 menu_id = menu and menu.id # Only set if `account_payment` is installed. 

160 

161 account_link_url = self._stripe_create_account_link(connected_account['id'], menu_id) 

162 if account_link_url: 

163 action = { 

164 'type': 'ir.actions.act_url', 

165 'url': account_link_url, 

166 'target': 'self', 

167 } 

168 else: 

169 action = { 

170 'type': 'ir.actions.act_window', 

171 'model': 'payment.provider', 

172 'views': [[False, 'form']], 

173 'res_id': self.id, 

174 } 

175 

176 return action

217 def action_stripe_verify_apple_pay_domain(self): 

218 """ Verify the web domain with Stripe to enable Apple Pay. 

219 

220 The domain is sent to Stripe API for them to verify that it is valid by making a request to 

221 the `/.well-known/apple-developer-merchantid-domain-association` route. If the domain is 

222 valid, it is registered to use with Apple Pay. 

223 See https://stripe.com/docs/stripe-js/elements/payment-request-button#verifying-your-domain-with-apple-pay. 

224 

225 :return dict: A client action with a success message. 

226 :raise UserError: If test keys are used to make the request. 

227 """ 

228 self.ensure_one() 

229 

230 web_domain = url_parse(self.get_base_url()).netloc 

231 response_content = self._stripe_make_request('apple_pay/domains', payload={ 

232 'domain_name': web_domain 

233 }) 

234 if not response_content['livemode']: 

235 # If test keys are used to make the request, Stripe will respond with an HTTP 200 but 

236 # will not register the domain. Ask the user to use live credentials. 

237 raise UserError(_("Please use live credentials to enable Apple Pay.")) 

238 

239 return { 

240 'type': 'ir.actions.client', 

241 'tag': 'display_notification', 

242 'params': { 

243 'message': _("Your web domain was successfully verified."), 

244 'type': 'success', 

245 }, 

246 }

253 def _stripe_make_request( 

254 self, endpoint, payload=None, method='POST', offline=False, idempotency_key=None 

255 ): 

256 """ Make a request to Stripe API at the specified endpoint. 

257 

258 Note: self.ensure_one() 

259 

260 :param str endpoint: The endpoint to be reached by the request 

261 :param dict payload: The payload of the request 

262 :param str method: The HTTP method of the request 

263 :param bool offline: Whether the operation of the transaction being processed is 'offline' 

264 :param str idempotency_key: The idempotency key to pass in the request. 

265 :return The JSON-formatted content of the response 

266 :rtype: dict 

267 :raise: ValidationError if an HTTP error occurs 

268 """ 

269 self.ensure_one() 

270 

271 url = url_join('https://api.stripe.com/v1/', endpoint) 

272 headers = { 

273 'AUTHORIZATION': f'Bearer {stripe_utils.get_secret_key(self)}', 

274 'Stripe-Version': const.API_VERSION, # SetupIntent requires a specific version. 

275 **self._get_stripe_extra_request_headers(), 

276 } 

277 if method == 'POST' and idempotency_key: 

278 headers['Idempotency-Key'] = idempotency_key 

279 try: 

280 response = requests.request(method, url, data=payload, headers=headers, timeout=60) 

281 # Stripe can send 4XX errors for payment failures (not only for badly-formed requests). 

282 # Check if an error code is present in the response content and raise only if not. 

283 # See https://stripe.com/docs/error-codes. 

284 # If the request originates from an offline operation, don't raise to avoid a cursor 

285 # rollback and return the response as-is for flow-specific handling. 

286 if not response.ok \ 

287 and not offline \ 

288 and 400 <= response.status_code < 500 \ 

289 and response.json().get('error'): # The 'code' entry is sometimes missing 

290 try: 

291 response.raise_for_status() 

292 except requests.exceptions.HTTPError: 

293 _logger.exception("invalid API request at %s with data %s", url, payload) 

294 error_msg = response.json().get('error', {}).get('message', '') 

295 raise ValidationError( 

296 "Stripe: " + _( 

297 "The communication with the API failed.\n" 

298 "Stripe gave us the following info about the problem:\n'%s'", error_msg 

299 ) 

300 ) 

301 except requests.exceptions.ConnectionError: 

302 _logger.exception("unable to reach endpoint at %s", url) 

303 raise ValidationError("Stripe: " + _("Could not establish the connection to the API.")) 

304 return response.json()

356 def _stripe_create_account_link(self, connected_account_id, menu_id): 

357 """ Create an account link and return its URL. 

358 

359 An account link url is the beginning URL of Stripe Onboarding. 

360 This URL is only valid once, and can only be used once. 

361 

362 Note: self.ensure_one() 

363 

364 :param str connected_account_id: The id of the connected account. 

365 :param int menu_id: The menu from which the user started the onboarding step, as an 

366 `ir.ui.menu` id 

367 :return: The account link URL 

368 :rtype: str 

369 """ 

370 self.ensure_one() 

371 

372 base_url = self.company_id.get_base_url() 

373 return_url = OnboardingController._onboarding_return_url 

374 refresh_url = OnboardingController._onboarding_refresh_url 

375 return_params = dict(provider_id=self.id, menu_id=menu_id) 

376 refresh_params = dict(**return_params, account_id=connected_account_id) 

377 

378 account_link = self._stripe_make_proxy_request('account_links', payload={ 

379 'account': connected_account_id, 

380 'return_url': f'{url_join(base_url, return_url)}?{url_encode(return_params)}', 

381 'refresh_url': f'{url_join(base_url, refresh_url)}?{url_encode(refresh_params)}', 

382 'type': 'account_onboarding', 

383 }) 

384 return account_link['url']

386 def _stripe_make_proxy_request(self, endpoint, payload=None, version=1): 

387 """ Make a request to the Stripe proxy at the specified endpoint. 

388 

389 :param str endpoint: The proxy endpoint to be reached by the request 

390 :param dict payload: The payload of the request 

391 :param int version: The proxy version used 

392 :return The JSON-formatted content of the response 

393 :rtype: dict 

394 :raise: ValidationError if an HTTP error occurs 

395 """ 

396 proxy_payload = { 

397 'jsonrpc': '2.0', 

398 'id': uuid.uuid4().hex, 

399 'method': 'call', 

400 'params': { 

401 'payload': payload, # Stripe data. 

402 'proxy_data': self._stripe_prepare_proxy_data(stripe_payload=payload), 

403 }, 

404 } 

405 url = url_join(const.PROXY_URL, f'{version}/{endpoint}') 

406 try: 

407 response = requests.post(url=url, json=proxy_payload, timeout=60) 

408 response.raise_for_status() 

409 except requests.exceptions.ConnectionError: 

410 _logger.exception("unable to reach endpoint at %s", url) 

411 raise ValidationError(_("Stripe Proxy: Could not establish the connection.")) 

412 except requests.exceptions.HTTPError: 

413 _logger.exception("invalid API request at %s with data %s", url, payload) 

414 raise ValidationError( 

415 _("Stripe Proxy: An error occurred when communicating with the proxy.") 

416 ) 

417 

418 # Stripe proxy endpoints always respond with HTTP 200 as they implement JSON-RPC 2.0 

419 response_content = response.json() 

420 if response_content.get('error'): # An exception was raised on the proxy 

421 error_data = response_content['error']['data'] 

422 _logger.warning("request forwarded with error: %s", error_data['message']) 

423 raise ValidationError(_("Stripe Proxy error: %(error)s", error=error_data['message'])) 

424 

425 return response_content.get('result', {})
