14 def _compute_duration_tracking(self): 

15 """ 

16 Computes duration_tracking, a Json field stored as { <many2one_id (str)>: <duration_spent_in_seconds (int)> } 

17 

18 e.g. {"1": 1230, "2": 2220, "5": 14} 

19 

20 `_track_duration_field` must be present in the model that uses the mixin to specify on what 

21 field to compute time spent. Besides, tracking must be activated for that field. 

22 

23 e.g. 

24 class MyModel(models.Model): 

25 _track_duration_field = "tracked_field" 

26 

27 tracked_field = fields.Many2one('tracked.model', tracking=True) 

28 """ 

29 

30 field = self.env['ir.model.fields'].sudo().search_fetch([ 

31 ('model', '=', self._name), 

32 ('name', '=', self._track_duration_field), 

33 ], ['id'], limit=1) 

34 

35 if ( 

36 self._track_duration_field not in self._track_get_fields() 

37 or self._fields[self._track_duration_field].type != 'many2one' 

38 ): 

39 self.duration_tracking = False 

40 raise ValueError(_( 

41 'Field “%(field)s” on model “%(model)s” must be of type Many2one and have tracking=True for the computation of duration.', 

42 field=self._track_duration_field, model=self._name 

43 )) 

44 

45 self.env['mail.tracking.value'].flush_model() 

46 self.env['mail.message'].flush_model()

49 v.create_date, 

50 v.old_value_integer

52 LEFT JOIN mail_message m 

53 ON m.id = v.mail_message_id 

54 AND v.field_id = %(field_id)s 

55 WHERE m.model = %(model_name)s

66 def _get_duration_from_tracking(self, trackings): 

67 """ 

68 Calculates the duration spent in each value based on the provided list of trackings. 

69 It adds a "fake" tracking at the end of the trackings list to account for the time spent in the current value. 

70 

71 Args: 

72 trackings (list): A list of dictionaries representing the trackings with: 

73 - 'create_date': The date and time of the tracking. 

74 - 'old_value_integer': The ID of the previous value. 

75 

76 Returns: 

77 dict: A dictionary where the keys are the IDs of the values, and the values are the durations in seconds 

78 """ 

79 self.ensure_one() 

80 json = defaultdict(lambda: 0) 

81 previous_date = self.create_date 

82 

83 # If there is a tracking value to be created, but still in the 

84 # precommit values, create a fake one to take it into account. 

85 # Otherwise, the duration_tracking value will add time spent on 

86 # previous tracked field value to the time spent in the new value 

87 # (after writing the stage on the record) 

88 if f'mail.tracking.{self._name}' in self.env.cr.precommit.data: 

89 if data := self.env.cr.precommit.data.get(f'mail.tracking.{self._name}', {}).get(self.id): 

90 new_id = data.get(self._track_duration_field, self.env[self._name]).id 

91 if new_id and new_id != self[self._track_duration_field].id: 

92 trackings.append({ 

93 'create_date': self.env.cr.now(), 

94 'old_value_integer': data[self._track_duration_field].id, 

95 }) 

96 

97 # add "fake" tracking for time spent in the current value 

98 trackings.append({ 

99 'create_date': self.env.cr.now(), 

100 'old_value_integer': self[self._track_duration_field].id, 

101 }) 

102 

103 for tracking in trackings: 

104 json[tracking['old_value_integer']] += int((tracking['create_date'] - previous_date).total_seconds()) 

105 previous_date = tracking['create_date'] 

106 

107 return json
