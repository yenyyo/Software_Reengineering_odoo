53 def _check_currency_table_monocurrency(self, companies): 

54 """ Returns whether displaying the data of the provided companies can be done with a monocurrency currency table. 

55 If it can, calling _get_monocurrency_currency_table_sql is enough to join the currency table (which actually consists of a bunch of VALUES 

56 directly injected in the join). 

57 Else, a full-flegdge temporary table will be needed, that will have to be generated by a call to _create_currency_table. 

58 """ 

59 return len(companies.currency_id) == 1

61 def _get_monocurrency_currency_table_sql(self, companies, use_cta_rates=False): 

62 """ Returns a simplified currency table, faster to generate, for cases were all the data to convert are expressed in the same currency, 

63 to be use in a JOIN. It actually just consists of a few VALUES ; no temporary table is created in this case. 

64 

65 All the rates in this currency table are equal to 1 (since everything is in the same currency). This is useful so that the queries can 

66 be written exactly in the same way, joining the currency table returned by some function, for both mono and multi currency cases. 

67 """ 

68 unit_rates = [ 

69 SQL("(%(company_id)s, CAST(NULL AS VARCHAR), CAST(NULL AS DATE), CAST(NULL AS DATE), %(rate_type)s, 1)", company_id=company.id, rate_type=rate_type) 

70 for company in companies 

71 for rate_type in (('historical', 'current', 'average') if use_cta_rates else ('current',)) 

72 ] 

73 return SQL('(VALUES %s) AS account_currency_table(company_id, period_key, date_from, date_next, rate_type, rate)', SQL(',').join(unit_rates))

75 def _create_currency_table(self, companies, date_periods, use_cta_rates=False): 

76 """ Creates a temporary table containing the currency rates to be used in order to aggregate amounts belonging to companies 

77 with different main currencies in a reporting query. 

78 These rates are computed from the res.currrency.rate objects defined for self.env.company. 

79 

80 The currency table consists of the following columns: 

81 - company_id: The id of the company whose amounts can be converted with this rate. 

82 - period_key: The key corresponding to the period this rate is valid for. (see params list) 

83 - date_from: Only set for rate_type 'historical'. The starting date for this rate. 

84 - date_next: Only set for rate_type 'historical'. The date of the next rate. So, the rate applies until one day before date_next. 

85 - rate_type: 'historical', 'current' or 'average' 

86 - 'historical' means the rate is to be used to convert operations at the date they were made; they each 

87 directly correspond to the res.currency.rate objects of the active company 

88 - 'current' means this rate is the most recent rate within the period. This rate is unique per (company_id, period_key). 

89 - 'average' means this rate is the average rate for the period. This rate is unique per (company_id, period_key). 

90 - rate: The rate to apply, as a decimal factor to apply directly to the value to convert, provided it is expressed in the 

91 main currency of the company referred to by company_id. 

92 

93 

94 :param companies: The res.company objects to generate rates for. 

95 :param date_periods: List of tuples in the form (period_key, date_from, date_to), containing each of the periods to generate rates for, where: 

96 - period_key is a unique string identifier used to differentiate the periods 

97 - date_from is the date the period starts at ; it can be None if the period want to consider everything from the beginning 

98 - date_to is the date the periods ends at 

99 :param use_cta_rates: Boolean parameter, enabling the computation of CTA rates. If True, 'current', 'average' and 'historical' rates will be 

100 computed for all companies, for all periods. Else, only 'current' will be computed. 

101 """ 

102 main_company = self.env.company 

103 domestic_currency_companies = companies.filtered(lambda x: x.currency_id == main_company.currency_id) 

104 other_companies = companies - domestic_currency_companies 

105 

106 table_builders = [] 

107 if domestic_currency_companies: 

108 table_builders += [self._get_table_builder_domestic_currency(domestic_currency_companies, use_cta_rates)] 

109 

110 last_date_to = None 

111 for period_key, date_from, date_to in date_periods: 

112 main_company_unit_factor = main_company.currency_id._get_rates(main_company, date_to)[main_company.currency_id.id] 

113 

114 if use_cta_rates: 

115 table_builders += [ 

116 self._get_table_builder_closing(period_key, main_company, other_companies, date_to, main_company_unit_factor), 

117 self._get_table_builder_historical(main_company, other_companies, date_to, main_company_unit_factor, last_date_to), 

118 self._get_table_builder_average(period_key, main_company, other_companies, date_from, date_to, main_company_unit_factor), 

119 ] 

120 else: 

121 table_builders += [self._get_table_builder_current(period_key, main_company, other_companies, date_to, main_company_unit_factor)] 

122 

123 last_date_to = date_to 

124 

125 self._cr.execute(SQL( 

126 """ 

127 -- Tests may call this function multiple times within the same transaction; we then need to delete an regenerate the currency table 

128 DROP TABLE IF EXISTS account_currency_table; 

129 

130 -- Create a temporary table 

131 CREATE TEMPORARY TABLE 

132 account_currency_table (company_id, period_key, date_from, date_next, rate_type, rate) 

133 ON COMMIT DROP 

134 AS (%(currency_table_build_query)s); 

135 

136 -- Create a supporting index to avoid seq.scans 

137 CREATE INDEX account_currency_table_index ON account_currency_table (company_id, rate_type, date_from, date_next); 

138 -- Update statistics for correct planning 

139 ANALYZE account_currency_table; 

140 """, 

141 currency_table_build_query=SQL(" UNION ALL ").join(SQL('(%s)', builder) for builder in table_builders), 

142 ))

169 def _get_table_builder_current(self, period_key, main_company, other_companies, date_to, main_company_unit_factor) -> SQL: 

170 return SQL( 

171 """ 

172 SELECT DISTINCT ON (other_company.id) 

173 other_company.id, 

174 %(period_key)s, 

175 CAST(NULL AS DATE), 

176 CAST(NULL AS DATE), 

177 'current', 

178 CASE WHEN rate.id IS NOT NULL THEN %(main_company_unit_factor)s / rate.rate ELSE 1 END 

179 FROM res_company other_company 

180 LEFT JOIN res_currency_rate rate 

181 ON rate.currency_id = other_company.currency_id 

182 AND rate.name <= %(date_to)s 

183 AND rate.company_id = %(main_company_id)s 

184 WHERE 

185 other_company.id IN %(other_company_ids)s 

186 ORDER BY other_company.id, rate.name DESC 

187 """, 

188 period_key=period_key, 

189 main_company_id=main_company.root_id.id, 

190 other_company_ids=tuple(other_companies.ids), 

191 date_to=date_to, 

192 main_company_unit_factor=main_company_unit_factor,

195 def _get_table_builder_closing(self, period_key, main_company, other_companies, date_to, main_company_unit_factor) -> SQL: 

196 fiscal_year_bounds = self._get_currency_table_fiscal_year_bounds(main_company) 

197 

198 return SQL( 

199 """ 

200 SELECT DISTINCT ON (other_company.id, fiscal_year_bounds.date_from, fiscal_year_bounds.date_to) 

201 other_company.id, 

202 %(period_key)s, 

203 fiscal_year_bounds.date_from, 

204 CAST(fiscal_year_bounds.date_to::TIMESTAMP + INTERVAL '1' DAY AS DATE), 

205 'closing', 

206 CASE WHEN rate.id IS NOT NULL THEN %(main_company_unit_factor)s / rate.rate ELSE 1 END 

207 FROM res_company other_company 

208 LEFT JOIN res_currency_rate rate 

209 ON rate.currency_id = other_company.currency_id 

210 AND rate.name <= %(date_to)s 

211 AND rate.company_id = %(main_company_id)s 

212 JOIN (VALUES %(fiscal_year_bounds_values)s) AS fiscal_year_bounds(date_from, date_to) 

213 ON fiscal_year_bounds.date_to IS NULL 

214 OR fiscal_year_bounds.date_to >= rate.name 

215 WHERE 

216 other_company.id IN %(other_company_ids)s 

217 ORDER BY other_company.id, fiscal_year_bounds.date_from, fiscal_year_bounds.date_to, rate.name DESC 

218 """, 

219 period_key=period_key, 

220 main_company_id=main_company.root_id.id, 

221 fiscal_year_bounds_values=SQL(",").join(SQL("(%(fy_from)s::date,%(fy_to)s::date)", fy_from=fy_from, fy_to=fy_to) for fy_from, fy_to in fiscal_year_bounds), 

222 other_company_ids=tuple(other_companies.ids), 

223 date_to=date_to, 

224 main_company_unit_factor=main_company_unit_factor, 

225 )

227 def _get_currency_table_fiscal_year_bounds(self, main_company): 

228 today_fiscal_year = main_company.compute_fiscalyear_dates(fields.Date.today()) 

229 first_rate = self.env['res.currency.rate'].search(self.env['res.currency.rate']._check_company_domain(main_company), order="name ASC", limit=1) 

230 fiscal_year_bounds = [] 

231 if first_rate: 

232 first_rate_fiscal_year = main_company.compute_fiscalyear_dates(first_rate.name) 

233 fiscal_year_bounds = [(None, first_rate_fiscal_year['date_from'] - relativedelta(days=1))] # Initialized to have a value for everything before the first rate 

234 for civil_year in range(first_rate_fiscal_year['date_from'].year, today_fiscal_year['date_from'].year): 

235 year_delta = relativedelta(years=civil_year - first_rate_fiscal_year['date_from'].year) 

236 fiscal_year_bounds.append((first_rate_fiscal_year['date_from'] + year_delta, first_rate_fiscal_year['date_to'] + year_delta)) 

237 

238 # The current fiscal year is not closed yet, so we need to use its rates for everything after it 

239 fiscal_year_bounds.append((today_fiscal_year['date_from'], None)) 

240 

241 return fiscal_year_bounds

269 def _get_table_builder_average(self, period_key, main_company, other_companies, date_from, date_to, main_company_unit_factor) -> SQL: 

270 if not date_from: 

271 # When there is no start date, we want to compute the average rate on the current year only 

272 date_from = date_utils.start_of(fields.Date.from_string(date_to), 'year')

274 return SQL( 

275 """ 

276 SELECT 

277 rate_with_days.other_company_id, 

278 %(period_key)s, 

279 CAST(NULL AS DATE), 

280 CAST(NULL AS DATE), 

281 'average',

283 FROM ( 

284 SELECT 

285 other_company.id as other_company_id, 

286 rate.rate AS rate, 

287 EXTRACT ( 

288 'Day' FROM COALESCE( 

289 LEAD(rate.name, 1) OVER (PARTITION BY other_company.id, rate.currency_id ORDER BY rate.name ASC)::TIMESTAMP, 

290 %(date_to)s::TIMESTAMP + INTERVAL '1' DAY 

291 ) - rate.name::TIMESTAMP 

292 ) AS number_of_days 

293 FROM res_company other_company 

294 JOIN res_currency_rate rate 

295 ON rate.currency_id = other_company.currency_id 

296 WHERE 

297 rate.name <= %(date_to)s

299 AND other_company.id IN %(other_company_ids)s 

300 AND rate.company_id = %(main_company_id)s 

301 

302 UNION ALL 

303 

304 ( 

305 SELECT DISTINCT ON (other_company.id) 

306 other_company.id as other_company_id, 

307 COALESCE(out_period_rate.rate, 1.0) AS rate, 

308 EXTRACT('Day' FROM COALESCE(in_period_rate.name::TIMESTAMP, %(date_to)s::TIMESTAMP + INTERVAL '1' DAY) - %(date_from)s::TIMESTAMP) AS number_of_days 

309 

310 FROM res_company other_company 

311 

312 LEFT JOIN res_currency_rate in_period_rate 

313 ON in_period_rate.currency_id = other_company.currency_id 

314 AND in_period_rate.name <= %(date_to)s 

315 AND in_period_rate.name >= %(date_from)s 

316 AND in_period_rate.company_id = %(main_company_id)s 

317 

318 LEFT JOIN res_currency_rate out_period_rate 

319 ON out_period_rate.currency_id = other_company.currency_id 

320 AND out_period_rate.company_id = %(main_company_id)s

323 WHERE 

324 other_company.id IN %(other_company_ids)s
