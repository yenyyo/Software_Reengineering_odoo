3import logging 

4from odoo import models 

5from odoo.http import request 

6 

7_logger = logging.getLogger(__name__) 

8BAD_REQUEST = """Missing request.is_frontend attribute. 

9 

10The request.is_frontend attribute is missing, this means that although 

11http_routing is installed and that all incoming requests SHOULD be 

12going through ir.http._match (which sets that attribute), 

13there are some rogue requests which do not. This is likely due to a 

14@route(auth='none') controller which creates its own registry and attempts 

15to render a template (e.g. odoo/odoo#99667). 

16 

17The following expectations MUST hold: 

18 

19When: 

20* there is an incoming http request (request is truthy) 

21* there is a registry loaded (models are in use) 

22* http_routing is installed (dependency of both portal and website) 

23 

24Then: 

25* request.is_frontend is set 

26 

27Failure to meet this expectation can lead to downstream problems, e.g. 

28here inside of http_routing's ir.qweb. Solutions vary, the one used 

29inside of #99667 is to use the request.borrow_request context manager to 

30temporary hide the incoming http request. 

31""" 

32 

33class IrQweb(models.AbstractModel): 

34 _inherit = "ir.qweb" 

35 

36 def _prepare_environment(self, values): 

37 irQweb = super()._prepare_environment(values) 

38 values['slug'] = self.env['ir.http']._slug 

39 values['unslug_url'] = self.env['ir.http']._unslug_url 

40 

41 if not irQweb.env.context.get('minimal_qcontext') and request: 

42 if not hasattr(request, 'is_frontend'): 

43 _logger.warning(BAD_REQUEST, stack_info=True) 

44 elif request.is_frontend: 

45 return irQweb._prepare_frontend_environment(values) 

46 

47 return irQweb 

48 

49 def _prepare_frontend_environment(self, values): 

50 values['url_for'] = self.env['ir.http']._url_for 

51 values['url_localized'] = self.env['ir.http']._url_localized 

52 return self
