294 @api.constrains('account_opening_move_id', 'fiscalyear_last_day', 'fiscalyear_last_month') 

295 def _check_fiscalyear_last_day(self): 

296 # if the user explicitly chooses the 29th of February we allow it: 

297 # there is no "fiscalyear_last_year" so we do not know his intentions. 

298 for rec in self: 

299 if rec.fiscalyear_last_day == 29 and rec.fiscalyear_last_month == '2': 

300 continue 

301 

302 if rec.account_opening_date: 

303 year = rec.account_opening_date.year 

304 else: 

305 year = datetime.now().year 

306 

307 max_day = calendar.monthrange(year, int(rec.fiscalyear_last_month))[1] 

308 if rec.fiscalyear_last_day > max_day: 

309 raise ValidationError(_("Invalid fiscal year last day"))

466 def _validate_locks(self, values): 

467 """Check that the lock date changes are valid. 

468 * Check that we do not decrease or remove the hard lock dates. 

469 * Check there are no unreconciled bank statement lines in the period we want to lock. 

470 * Check there are no unhashed journal entires in the period we want to lock. 

471 :param vals: The values passed to the write method. 

472 """ 

473 new_locks = {field: fields.Date.to_date(values[field])for field in LOCK_DATE_FIELDS if field in values} 

474 

475 fiscalyear_lock_date = new_locks.get('fiscalyear_lock_date') 

476 hard_lock_date = new_locks.get('hard_lock_date') 

477 sale_lock_date = new_locks.get('sale_lock_date') 

478 purchase_lock_date = new_locks.get('purchase_lock_date') 

479 fiscal_lock_date = None 

480 if fiscalyear_lock_date or hard_lock_date: 

481 fiscal_lock_date = max(fiscalyear_lock_date or date.min, hard_lock_date or date.min) 

482 

483 if 'hard_lock_date' in new_locks: 

484 for company in self: 

485 if not company.hard_lock_date: 

486 continue 

487 if not hard_lock_date: 

488 raise UserError(_("The Hard Lock Date cannot be removed.")) 

489 if hard_lock_date < company.hard_lock_date: 

490 raise UserError(_("A new Hard Lock Date must be posterior (or equal) to the previous one.")) 

491 

492 if hard_lock_date: 

493 draft_entries = self.env['account.move'].search([ 

494 ('company_id', 'child_of', self.ids),

521 def _get_user_lock_date(self, soft_lock_date_field, ignore_exceptions=False): 

522 """Get the lock date called `soft_lock_date_field` for this company depending on the user. 

523 We consider the field and exceptions (except if `ignore_exceptions`) for it in this company and the parent companies. 

524 :param str soft_lock_date_field: One of the lock date fields (except 'hard_lock_date'; see SOFT_LOCK_DATE_FIELDS) 

525 :param bool ignore_exceptions: Whether we ignore exceptions or not 

526 :return the user lock date 

527 """ 

528 self.ensure_one() 

529 soft_lock_date = date.min 

530 # We need to use sudo, since we might not have access to a parent company. 

531 for company in self.sudo().parent_ids: 

532 if company[soft_lock_date_field]: 

533 if ignore_exceptions: 

534 exception = None 

535 else: 

536 exception = self.env['account.lock_exception'].search( 

537 [ 

538 ('state', '=', 'active'), # checks the datetime 

539 '|', 

540 ('user_id', '=', None), 

541 ('user_id', '=', self.env.user.id), 

542 (soft_lock_date_field, '<', company[soft_lock_date_field]), 

543 ('company_id', '=', company.id), 

544 ], 

545 order='lock_date asc NULLS FIRST', 

546 limit=1, 

547 ) 

548 if exception: 

549 # The search domain of the exception ensures `exception[soft_lock_date_field] < company[soft_lock_date_field]` 

550 # or `exception[soft_lock_date_field] is False` 

551 soft_lock_date = max(soft_lock_date, exception[soft_lock_date_field] or date.min) 

552 else: 

553 soft_lock_date = max(soft_lock_date, company[soft_lock_date_field]) 

554 return soft_lock_date

570 def _get_violated_soft_lock_date(self, soft_lock_date_field, date): 

571 """ 

572 Check whether `date` violates the lock date called `soft_lock_date_field`. 

573 :param str soft_lock_date_field: One of the lock date fields (except 'hard_lock_date'; see SOFT_LOCK_DATE_FIELDS) 

574 :param date: We check whether this date is prior or equal to the lock date. 

575 :return the violated lock date as a date (or `None`) 

576 """ 

577 violated_date = None 

578 if not self: 

579 return violated_date 

580 self.ensure_one() 

581 user_lock_date_field = f'user_{soft_lock_date_field}' 

582 regular_lock_date = self.with_context(ignore_exceptions=True)[user_lock_date_field] 

583 if date <= regular_lock_date: 

584 violated_date = regular_lock_date 

585 user_lock_date = self.with_context(ignore_exceptions=False)[user_lock_date_field] 

586 violated_date = None if date > user_lock_date else user_lock_date 

587 return violated_date

589 def _get_lock_date_violations(self, accounting_date, fiscalyear=True, sale=True, purchase=True, tax=True, hard=True): 

590 """Get all the lock dates affecting the current accounting_date. 

591 :param accounting_date: The accounting date 

592 :param bool fiscalyear: Whether we should check the `fiscalyear_lock_date` 

593 :param bool sale: Whether we should check the `sale_lock_date` 

594 :param bool purchase: Whether we should check the `purchase_lock_date` 

595 :param bool tax: Whether we should check the `tax_lock_date` 

596 :param bool hard: Whether we should check the `hard_lock_date` 

597 :return: a list of tuples containing the lock dates (not ordered chronologically). 

598 """ 

599 self.ensure_one() 

600 locks = [] 

601 

602 if not accounting_date: 

603 return locks 

604 

605 soft_lock_date_fields_to_check = [ 

606 # (field, "to check") 

607 ('fiscalyear_lock_date', fiscalyear), 

608 ('sale_lock_date', sale), 

609 ('purchase_lock_date', purchase), 

610 ('tax_lock_date', tax), 

611 ] 

612 for field, to_check in soft_lock_date_fields_to_check: 

613 if not to_check: 

614 continue 

615 violated_date = self._get_violated_soft_lock_date(field, accounting_date) 

616 if violated_date: 

617 locks.append((violated_date, field)) 

618 

619 if hard: 

620 hard_lock_date = self.user_hard_lock_date 

621 if accounting_date <= hard_lock_date: 

622 locks.append((hard_lock_date, 'hard_lock_date')) 

623 

624 return locks

781 def del_lines(lines): 

782 nonlocal open_balance 

783 for line in lines: 

784 open_balance -= line.balance 

785 yield Command.delete(line.id) 

786 

787 def update_vals(account, side, balance, balancing=False): 

788 nonlocal open_balance 

789 corresponding_lines = corresponding_lines_per_account[(account, side)] 

790 currency = account.currency_id or self.currency_id 

791 amount_currency = balance if balancing else self.currency_id._convert(balance, currency, date=conversion_date) 

792 open_balance += balance 

793 if self.currency_id.is_zero(balance): 

794 yield from del_lines(corresponding_lines) 

795 elif corresponding_lines: 

796 line_to_update = corresponding_lines[0] 

797 open_balance -= line_to_update.balance 

798 yield Command.update(line_to_update.id, { 

799 'balance': balance, 

800 'amount_currency': amount_currency, 

801 }) 

802 yield from del_lines(corresponding_lines[1:]) 

803 else: 

804 yield Command.create({ 

805 'name':_("Automatic Balancing Line") if balancing else _("Opening balance"), 

806 'account_id': account.id, 

807 'balance': balance, 

808 'amount_currency': amount_currency, 

809 'currency_id': currency.id, 

810 }) 

811 

812 # Decode the existing opening move. 

813 corresponding_lines_per_account = defaultdict(lambda: self.env['account.move.line']) 

814 corresponding_lines_per_account.update(opening_move.line_ids.grouped(lambda line: ( 

815 line.account_id, 

816 'debit' if line.balance > 0.0 or line.amount_currency > 0.0 else 'credit', 

817 ))) 

818 

819 # Update the opening move's lines. 

820 balancing_account = self.get_unaffected_earnings_account() 

821 open_balance = ( 

822 sum(corresponding_lines_per_account[(balancing_account, 'credit')].mapped('credit')) 

823 -sum(corresponding_lines_per_account[(balancing_account, 'debit')].mapped('debit')) 

824 ) 

825 commands = [] 

826 move_values = {'line_ids': commands} 

827 if opening_move: 

828 conversion_date = opening_move.date 

829 else: 

830 move_values.update(self._get_default_opening_move_values()) 

831 conversion_date = move_values['date'] 

832 for account, (debit, credit) in to_update.items(): 

833 if debit is not None: 

834 commands.extend(update_vals(account, 'debit', debit)) 

835 if credit is not None: 

836 commands.extend(update_vals(account, 'credit', -credit)) 

837 

838 commands.extend(update_vals(balancing_account, 'debit', max(-open_balance, 0), balancing=True)) 

839 commands.extend(update_vals(balancing_account, 'credit', -max(open_balance, 0), balancing=True)) 

840 

841 # Nothing to do. 

842 if not commands: 

843 return 

844 

845 if opening_move: 

846 opening_move.write(move_values) 

847 else: 

848 self.account_opening_move_id = self.env['account.move'].create(move_values)

902 def _check_hash_integrity(self): 

903 """Checks that all hashed moves have still the same data as when they were hashed 

904 and raises an error with the result. 

905 """ 

906 if not self.env.user.has_group('account.group_account_user'): 

907 raise UserError(_('Please contact your accountant to print the Hash integrity result.')) 

908 

909 journals = self.env['account.journal'].search(self.env['account.journal']._check_company_domain(self)) 

910 results = [] 

911 

912 for journal in journals: 

913 restricted_by_hash_table_flag = 'V' if journal.restrict_mode_hash_table else 'X' 

914 # We need the `sudo()` to ensure that all the moves are searched, no matter the user's access rights. 

915 # This is required in order to generate consistent hashes. 

916 # It is not an issue, since the data is only used to compute a hash and not to return the actual values. 

917 query = self.env['account.move'].sudo()._search( 

918 domain=[ 

919 ('journal_id', '=', journal.id), 

920 ('inalterable_hash', '!=', False), 

921 ], 

922 order="secure_sequence_number ASC NULLS LAST, sequence_prefix, sequence_number ASC", 

923 ) 

924 prefix2result = defaultdict(lambda: { 

925 'first_move': self.env['account.move'], 

926 'last_move': self.env['account.move'], 

927 'corrupted_move': self.env['account.move'], 

928 }) 

929 last_move = self.env['account.move'] 

930 self.env.execute_query(SQL("DECLARE hashed_moves CURSOR FOR %s", query.select())) 

931 while move_ids := self.env.execute_query(SQL("FETCH %s FROM hashed_moves", INTEGRITY_HASH_BATCH_SIZE)): 

932 self.env.invalidate_all() 

933 moves = self.env['account.move'].browse(move_id[0] for move_id in move_ids) 

934 if not moves and not last_move: 

935 results.append({ 

936 'journal_name': journal.name, 

937 'restricted_by_hash_table': restricted_by_hash_table_flag, 

938 'status': 'no_data', 

939 'msg_cover': _('There is no journal entry flagged for accounting data inalterability yet.'), 

940 }) 

941 continue 

942 

943 current_hash_version = 1 

944 for move in moves: 

945 prefix_result = prefix2result[move.sequence_prefix] 

946 if prefix_result['corrupted_move']: 

947 continue 

948 previous_move = prefix_result['last_move'] if not move.secure_sequence_number else last_move 

949 previous_hash = previous_move.inalterable_hash or "" 

950 computed_hash = move.with_context(hash_version=current_hash_version)._calculate_hashes(previous_hash)[move] 

951 while move.inalterable_hash != computed_hash and current_hash_version < MAX_HASH_VERSION: 

952 current_hash_version += 1 

953 computed_hash = move.with_context(hash_version=current_hash_version)._calculate_hashes(previous_hash)[move] 

954 if move.inalterable_hash != computed_hash: 

955 prefix_result['corrupted_move'] = move 

956 continue 

957 if not prefix_result['first_move']: 

958 prefix_result['first_move'] = move 

959 prefix_result['last_move'] = move 

960 last_move = move 

961 

962 self.env.execute_query(SQL("CLOSE hashed_moves")) 

963 

964 for prefix, prefix_result in prefix2result.items(): 

965 if corrupted_move := prefix_result['corrupted_move']: 

966 results.append({ 

967 'restricted_by_hash_table': restricted_by_hash_table_flag, 

968 'journal_name': f"{journal.name} ({prefix}...)", 

969 'status': 'corrupted', 

970 'msg_cover': _( 

971 "Corrupted data on journal entry with id %(id)s (%(name)s).", 

972 id=corrupted_move.id, 

973 name=corrupted_move.name, 

974 ), 

975 }) 

976 else: 

977 results.append({ 

978 'restricted_by_hash_table': restricted_by_hash_table_flag, 

979 'journal_name': f"{journal.name} ({prefix}...)", 

980 'status': 'verified', 

981 'msg_cover': _("Entries are correctly hashed"), 

982 'first_move_name': prefix_result['first_move'].name, 

983 'first_hash': prefix_result['first_move'].inalterable_hash, 

984 'first_move_date': format_date(self.env, prefix_result['first_move'].date), 

985 'last_move_name': prefix_result['last_move'].name, 

986 'last_hash': prefix_result['last_move'].inalterable_hash, 

987 'last_move_date': format_date(self.env, prefix_result['last_move'].date), 

988 }) 

989 

990 return { 

991 'results': results, 

992 'printing_date': format_date(self.env, fields.Date.context_today(self)), 

993 }
