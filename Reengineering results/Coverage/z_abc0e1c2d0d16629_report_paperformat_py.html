166class report_paperformat(models.Model): 

167 _name = "report.paperformat" 

168 _description = "Paper Format Config" 

169 

170 name = fields.Char('Name', required=True) 

171 default = fields.Boolean('Default paper format?') 

172 format = fields.Selection([(ps['key'], ps['description']) for ps in PAPER_SIZES], 'Paper size', default='A4', help="Select Proper Paper size") 

173 margin_top = fields.Float('Top Margin (mm)', default=40) 

174 margin_bottom = fields.Float('Bottom Margin (mm)', default=20) 

175 margin_left = fields.Float('Left Margin (mm)', default=7) 

176 margin_right = fields.Float('Right Margin (mm)', default=7) 

177 page_height = fields.Integer('Page height (mm)', default=False) 

178 page_width = fields.Integer('Page width (mm)', default=False) 

179 orientation = fields.Selection([ 

180 ('Landscape', 'Landscape'), 

181 ('Portrait', 'Portrait') 

182 ], 'Orientation', default='Landscape') 

183 header_line = fields.Boolean('Display a header line', default=False) 

184 header_spacing = fields.Integer('Header spacing', default=35) 

185 disable_shrinking = fields.Boolean('Disable smart shrinking') 

186 dpi = fields.Integer('Output DPI', required=True, default=90) 

187 report_ids = fields.One2many('ir.actions.report', 'paperformat_id', 'Associated reports', help="Explicitly associated reports") 

188 print_page_width = fields.Float('Print page width (mm)', compute='_compute_print_page_size') 

189 print_page_height = fields.Float('Print page height (mm)', compute='_compute_print_page_size') 

190 css_margins = fields.Boolean('Use css margins', default=False) 

191 

192 @api.constrains('format') 

193 def _check_format_or_page(self): 

194 if self.filtered(lambda x: x.format != 'custom' and (x.page_width or x.page_height)): 

195 raise ValidationError(_('You can select either a format or a specific page width/height, but not both.')) 

196 

197 def _compute_print_page_size(self): 

198 for record in self: 

199 width = height = 0.0 

200 if record.format: 

201 if record.format == 'custom': 

202 width = record.page_width 

203 height = record.page_height 

204 else: 

205 paper_size = next(ps for ps in PAPER_SIZES if ps['key'] == record.format) 

206 width = paper_size['width'] 

207 height = paper_size['height'] 

208 

209 if record.orientation == 'Landscape': 

210 # swap sizes 

211 width, height = height, width 

212 

213 record.print_page_width = width 

214 record.print_page_height = height
